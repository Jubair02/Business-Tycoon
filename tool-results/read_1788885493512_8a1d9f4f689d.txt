    81→// ---- Tick Concurrency Lock ----
    82→
    83→/**
    84→ * Atomically attempt to acquire the tick lock using the GameState key-value store.
    85→ * Returns true if the lock was acquired, false if it was already held.
    86→ *
    87→ * Stale lock recovery: if the lock has been held for longer than
    88→ * STALE_TICK_TIMEOUT_MS, the lock is stolen and a warning is logged.
    89→ */
    90→export async function acquireTickLock(): Promise<boolean> {
    91→  const now = new Date();
    92→
    93→  return db.$transaction(async (tx) => {
    94→    // Read current lock state
    95→    const lockState = await tx.gameState.findUnique({ where: { key: 'tickInProgress' } });
    96→    const startedAtState = await tx.gameState.findUnique({ where: { key: 'tickStartedAt' } });
    97→
    98→    const isLocked = lockState?.value === 'true';
    99→    const startedAt = startedAtState?.value ? new Date(startedAtState.value) : null;
   100→
   101→    if (isLocked && startedAt) {
   102→      const elapsed = now.getTime() - startedAt.getTime();
   103→      if (elapsed < STALE_TICK_TIMEOUT_MS) {
   104→        return false; // Lock is legitimately held
   105→      }
   106→      console.warn(`[GameEngine] Stale tick lock detected (held for ${Math.round(elapsed / 1000)}s). Stealing lock.`);
   107→    } else if (isLocked) {
   108→      console.warn('[GameEngine] Tick lock held with no timestamp. Stealing lock.');
   109→    }
   110→
   111→    // Acquire the lock
   112→    const nowISO = now.toISOString();
   113→    await tx.gameState.upsert({
   114→      where: { key: 'tickInProgress' },
   115→      create: { key: 'tickInProgress', value: 'true' },
   116→      update: { value: 'true' },
   117→    });
   118→    await tx.gameState.upsert({
   119→      where: { key: 'tickStartedAt' },
   120→      create: { key: 'tickStartedAt', value: nowISO },
   121→      update: { value: nowISO },
   122→    });
   123→
   124→    // Increment tick version
   125→    const versionState = await tx.gameState.findUnique({ where: { key: 'tickVersion' } });
   126→    const currentVersion = parseInt(versionState?.value || '0', 10);
   127→    await tx.gameState.upsert({
   128→      where: { key: 'tickVersion' },
   129→      create: { key: 'tickVersion', value: '1' },
   130→      update: { value: String(currentVersion + 1) },
   131→    });
   132→
   133→    return true;
   134→  });
   135→}

   145→// ---- Net Worth Recalculation ----
   146→async function recalculateNetWorth(tx: PrismaTx, playerId: string): Promise<void> {
   147→  const player = await tx.player.findUnique({
   148→    where: { id: playerId },
   149→    select: { id: true, cash: true },
   150→  });
   151→  if (!player) {
   152→    console.error(`[GameEngine] recalculateNetWorth: Player ${playerId} not found`);
   153→    return;
   154→  }
   155→
   156→  const allBusinesses = await tx.business.findMany({
   157→    where: { playerId },
   158→    select: { cash: true },
   159→  });
   160→  const totalBusinessCash = allBusinesses.reduce((sum, b) => sum + b.cash, 0);
   161→
   162→  const allInventory = await tx.inventory.findMany({
   163→    where: { business: { playerId } },
   164→    select: { quantity: true, purchasePrice: true },
   165→  });
   166→  const totalInventoryValue = allInventory.reduce((sum, inv) => sum + inv.quantity * inv.purchasePrice, 0);
   167→
   168→  const activeLoans = await tx.loan.findMany({
   169→    where: { playerId, status: 'ACTIVE' },
   170→    select: { remainingDebt: true },
   171→  });
   172→  const outstandingDebt = activeLoans.reduce((sum, loan) => sum + loan.remainingDebt, 0);
   173→
   174→  const netWorth = player.cash + totalBusinessCash + totalInventoryValue - outstandingDebt;
   175→  await tx.player.update({ where: { id: playerId }, data: { netWorth } });
   176→}

   179→export async function generateEvent(): Promise<void> {
   180→  const totalWeight = EVENT_TEMPLATES.reduce((sum, e) => sum + e.weight, 0);
   181→  let random = Math.random() * totalWeight;
   182→  let selected: EventTemplate | null = null;
   183→  for (const event of EVENT_TEMPLATES) {
   184→    random -= event.weight;
   185→    if (random <= 0) {
   186→      selected = event;
   187→      break;
   188→    }
   189→  }
   190→  if (!selected) selected = EVENT_TEMPLATES[0];
   191→
   192→  const now = new Date();
   193→  const endAt = new Date(now.getTime() + selected.durationDays * 60 * 1000);
   194→  await db.gameEvent.create({
   195→    data: {
   196→      title: selected.title,
   197→      description: selected.description,
   198→      type: selected.type,
   199→      effects: JSON.stringify(selected.effects),
   200→      active: true,
   201→      startsAt: now,
   202→      endsAt: endAt,
   203→    },
   204→  });
   205→}

   213→export async function updateMarketPrices(): Promise<void> {
   214→  const allProducts = Object.values(PRODUCTS).flat();
   215→  const cities = CITIES.map(c => c.id);
   216→
   217→  // Get active events for price modifiers
   218→  const activeEvents = await db.gameEvent.findMany({ where: { active: true } });
   219→  const categoryPriceMod: Record<string, number> = {};
   220→  let allPriceMod = 0;
   221→  for (const event of activeEvents) {
   222→    const effects = JSON.parse(event.effects) as Record<string, number>;
   223→    for (const [key, value] of Object.entries(effects)) {
   224→      if (key === 'all_price' || key === 'all_imported') allPriceMod += value;
   225→      if (key.endsWith('_price')) {
   226→        const cat = key.replace('_price', '');
   227→        categoryPriceMod[cat] = (categoryPriceMod[cat] || 0) + value;
   228→      }
   229→    }
   230→  }
   231→
   232→  const updates: { productName: string; city: string; priceMultiplier: number; demandMultiplier: number }[] = [];
   233→
   234→  for (const city of cities) {
   235→    for (const product of allProducts) {
   236→      // Get current market price (for retention)
   237→      const existing = await db.marketPrice.findUnique({
   238→        where: { productName_city: { productName: product.name, city } },
   239→      });
   240→      const previousPriceMult = existing?.priceMultiplier ?? 1;
   241→
   242→      // Calculate event price modifier for this product
   243→      const eventPriceMod = allPriceMod + (categoryPriceMod[product.category] || 0);
   244→
   245→      // Use Phase 1 retention-based update
   246→      const { priceMultiplier, demandMultiplier } = calculateMarketPriceUpdate(
   247→        previousPriceMult,
   248→        eventPriceMod
   249→      );
   250→
   251→      updates.push({ productName: product.name, city, priceMultiplier, demandMultiplier });
   252→    }
   253→  }
   254→
   255→  // Batch update in a transaction
   256→  await db.$transaction(async (tx) => {
   257→    for (const upd of updates) {
   258→      await tx.marketPrice.upsert({
   259→        where: { productName_city: { productName: upd.productName, city: upd.city } },
   260→        create: upd,
   261→        update: {
   262→          priceMultiplier: upd.priceMultiplier,
   263→          demandMultiplier: upd.demandMultiplier,
   264→          updatedAt: new Date(),
   265→        },
   266→      });
   267→    }
   268→  });
   269→}

   290→async function processMarketingTick(
   291→  businessId: string,
   292→  businessLevel: number,
   293→  satisfactionScore: number,
   294→  loyaltyScore: number,
   295→  repeatCustomerRate: number,
   296→  currentBrandAwareness: number,
   297→  segmentDemands: { segment: string; demandMultiplier: number }[],
   298→  gameDay: number,
   299→): Promise<{
   300→  combinedDemandModifier: number;
   301→  totalMarketingSpend: number;
   302→  newBrandAwareness: number;
   303→  totalConversions: number;
   304→  totalReach: number;
   305→  campaignIds: string[];  // IDs of campaigns processed this tick (for revenue attribution)
   306→}> {
   307→  // Fetch active campaigns
   308→  const activeCampaigns = await db.marketingCampaign.findMany({
   309→    where: { businessId, status: 'ACTIVE' },
   310→  });
   311→
   312→  if (activeCampaigns.length === 0) {
   313→    // No campaigns — just decay brand awareness
   314→    const newAwareness = calculateBrandAwareness(currentBrandAwareness, 0);
   315→    return {
   316→      combinedDemandModifier: 1.0,
   317→      totalMarketingSpend: 0,
   318→      newBrandAwareness: newAwareness,
   319→      totalConversions: 0,
   320→      totalReach: 0,
   321→      campaignIds: [],
   322→    };
   323→  }
   324→
   325→  const campaignModifiers: number[] = [];
   326→  const processedCampaignIds: string[] = [];  // Track which campaigns were processed
   327→  let totalSpend = 0;
   328→  let totalReach = 0;
   329→  let totalConversions = 0;
   330→
   331→  for (const campaign of activeCampaigns) {
   332→    // Calculate daily spend (clamped to remaining budget)
   333→    const dailySpend = calculateCampaignDailySpend(
   334→      campaign.dailyBudget,
   335→      campaign.totalSpend,
   336→      campaign.totalBudget,
   337→    );
   338→

   340→      // Budget exhausted — complete the campaign
   341→      await db.marketingCampaign.update({
   342→        where: { id: campaign.id },
   343→        data: { status: 'COMPLETED' },
   344→      });
   345→      continue;
   346→    }
   347→
   348→    // Calculate reach
   349→    const reach = calculateCampaignReach(
   350→      campaign.channel as any,
   351→      dailySpend,
   352→      businessLevel,
   353→    );
   354→
   355→    // Calculate conversions
   356→    const conversions = calculateCampaignConversions(
   357→      campaign.channel as any,
   358→      reach,
   359→      satisfactionScore,
   360→      campaign.targetSegment as any,
   361→      segmentDemands,
   362→    );
   363→
   364→    // Calculate this campaign's demand modifier
   365→    // Use a reasonable baseline for scaling (business type base customers × city multiplier)
   366→    const baselineForScaling = 50; // Reasonable default
   367→    const demandMod = calculateCampaignDemandModifier(conversions, baselineForScaling);
   368→    campaignModifiers.push(demandMod);
   369→
   370→    // Calculate effectiveness
   371→    const newTotalSpend = campaign.totalSpend + dailySpend;
   372→    const newTotalConversions = campaign.totalConversions + conversions;
   373→    const newTotalReach = campaign.totalReach + reach;
   374→    const effectiveness = calculateCampaignEffectiveness(
   375→      newTotalConversions,
   376→      newTotalReach,
   377→      newTotalSpend,
   378→      campaign.revenueInfluenced, // Will be updated later when we know revenue
   379→    );
   380→
   381→    // Update campaign
   382→    const newDaysRun = campaign.daysRun + 1;
   383→    const isExpired = newDaysRun >= campaign.duration;
   384→
   385→    await db.marketingCampaign.update({
   386→      where: { id: campaign.id },
   387→      data: {
   388→        daysRun: newDaysRun,
   389→        totalSpend: newTotalSpend,
   390→        totalReach: newTotalReach,
   391→        totalConversions: newTotalConversions,
   392→        effectiveness,
   393→        status: isExpired ? 'COMPLETED' : 'ACTIVE',
   394→      },
   395→    });
   396→
   397→    // Create daily metric
   398→    await db.campaignMetric.create({
   399→      data: {
   400→        campaignId: campaign.id,
   401→        gameDay,
   402→        dailySpend,
   403→        dailyReach: reach,
   404→        dailyConversions: conversions,
   405→        demandModifier: demandMod,
   406→      },
   407→    });
   408→
   409→    totalSpend += dailySpend;
   410→    totalReach += reach;
   411→    totalConversions += conversions;
   412→    processedCampaignIds.push(campaign.id);
   413→  }
   414→
   415→  // Combined demand modifier with stacking diminishing returns
   416→  const combinedDemandModifier = calculateCombinedMarketingModifier(campaignModifiers);
   417→
   418→  // Brand awareness update
   419→  const newBrandAwareness = calculateBrandAwareness(currentBrandAwareness, totalReach);
   420→
   421→  return {
   422→    combinedDemandModifier,
   423→    totalMarketingSpend: totalSpend,
   424→    newBrandAwareness,
   425→    totalConversions,
   426→    totalReach,
   427→    campaignIds: processedCampaignIds,
   428→  };
   429→}

   452→export async function simulateBusinessTick(businessId: string): Promise<void> {
   453→  const business = await db.business.findUnique({
   454→    where: { id: businessId },
   455→    include: {
   456→      inventories: true,
   457→      employees: true,
   458→      player: { select: { id: true, cash: true, netWorth: true } },
   459→    },
   460→  });
   461→
   462→  if (!business) return;
   463→
   464→  const businessType = getBusinessType(business.type);
   465→  const city = getCity(business.city);
   466→  if (!businessType || !city) return;
   467→
   468→  const economyConfig = getBusinessEconomyConfig(business.type);
   469→
   470→  // Get active event effects (combined)
   471→  const activeEvents = await db.gameEvent.findMany({
   472→    where: { active: true },
   473→    select: { effects: true },
   474→  });
   475→  const combinedEffects: Record<string, number> = {};
   476→  for (const event of activeEvents) {
   477→    const effects = JSON.parse(event.effects) as Record<string, number>;
   478→    for (const [key, value] of Object.entries(effects)) {
   479→      combinedEffects[key] = (combinedEffects[key] || 0) + value;
   480→    }
   481→  }
   482→
   483→  // Get market prices for this city
   484→  const marketPrices = await db.marketPrice.findMany({
   485→    where: { city: business.city },
   486→  });
   487→  const marketPriceMap: Record<string, { priceMultiplier: number; demandMultiplier: number }> = {};
   488→  for (const mp of marketPrices) {
   489→    marketPriceMap[mp.productName] = {
   490→      priceMultiplier: mp.priceMultiplier,
   491→      demandMultiplier: mp.demandMultiplier,
   492→    };
   493→  }
   494→
   495→  // ---- Step 0.5: CX Demand Modifier (from previous tick) ----
   496→  // Use the business's existing satisfaction/loyalty/reputation to calculate
   497→  // how CX affects THIS tick's customer count. This avoids circular dependency
   498→  // because we use previous-tick CX values to influence current-tick demand.
   499→  const previousCxDemandModifier = calculateCXDemandModifier(
   500→    business.satisfactionScore,
   501→    business.repeatCustomerRate,
   502→    getLoyaltyTier(business.loyaltyScore).multiplier,
   503→  );
   504→
   505→  // Calculate segment demand modifier from previous-tick data
   506→  // Segments react differently to quality, service, reputation
   507→  // NOTE: priceCompetitiveness is NOT passed to segment demands because price is already
   508→  // handled by per-product priceDemandMultiplier AND satisfaction's priceCompetitiveness.
   509→  // Passing it here would triple-count price effects.
   510→  // Similarly, reputation is already in calculatePotentialCustomers Layer 3 AND satisfaction's
   511→  // atmosphere factor, so we use a moderate value to avoid quadruple-counting.
   512→  const previousCxServiceQuality = calculateServiceQuality(
   513→    business.employees.length,
   514→    business.employees.length > 0
   515→      ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   516→      : 0,
   517→    2 + business.level,
   518→  );
   519→  const segmentDemands = calculateSegmentDemands({
   520→    priceCompetitiveness: 1.0,  // Price already handled by priceDemandMultiplier + satisfaction
   521→    productQuality: Math.max(0, Math.min(1, business.healthScore / 100)), // Use health as quality proxy from previous tick
   522→    serviceQuality: previousCxServiceQuality,
   523→    reputation: Math.min(100, business.reputation * 0.5 + 50),  // Dampened: avg 50 ± 25 from reputation
   524→  });
   525→  // Aggregate segment demand: sum of demand multipliers (NOT weighted by share again)
   526→  // demandMultiplier already includes baseShare, so we just sum them
   527→  const segmentDemandModifier = segmentDemands.reduce(
   528→    (sum, seg) => sum + seg.demandMultiplier, 0
   529→  );
   530→
   531→  // ---- Step 0.6: Marketing Demand Modifier (Phase 4) ----
   532→  // Fetch gameDay early (needed for marketing metrics and later for business metrics)
   533→  const gameDayState = await db.gameState.findUnique({ where: { key: 'gameDay' } });
   534→  const gameDay = parseInt(gameDayState?.value || '1', 10);
   535→
   536→  const marketingEffect = await processMarketingTick(
   537→    business.id,
   538→    business.level,
   539→    business.satisfactionScore,
   540→    business.loyaltyScore,
   541→    business.repeatCustomerRate,
   542→    business.brandAwareness,
   543→    segmentDemands,
   544→    gameDay,
   545→  );
   546→
   547→  // ---- Step 1: Calculate potential customers ----
   548→  const totalStock = business.inventories.reduce((sum, inv) => sum + inv.quantity, 0);
   549→  // Max stock capacity: sum of product maxStock values for this business type
   550→  const productDefs = PRODUCTS[business.type] || [];
   551→  const maxStockCapacity = productDefs.reduce((sum, p) => sum + p.maxStock, 0);
   552→  const avgEmployeeSkill = business.employees.length > 0
   553→    ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   554→    : 0;
   555→
   556→  const basePotentialCustomers = calculatePotentialCustomers({
   557→    baseCustomers: businessType.baseCustomers,
   558→    cityMultiplier: city.customerMultiplier,
   559→    level: business.level,
   560→    reputation: business.reputation,
   561→    totalStock,
   562→    maxStockCapacity,
   563→    employeeCount: business.employees.length,
   564→    avgEmployeeSkill,
   565→    eventCustomerEffect: combinedEffects['all_customers'] || 0,
   566→    businessDemandEffect: combinedEffects[`${businessType.id}_demand`] || 0,
   567→    businessTypeId: business.type,
   568→  });
   569→
   570→  // Apply CX demand modifier, segment demand modifier, and marketing modifiers
   571→  // CX modifier: satisfaction/loyalty affect how many customers visit (0.2-2.0 range)
   572→  // Segment modifier: different customer segments find the business more/less attractive
   573→  // Phase 4: Marketing demand modifier and brand awareness bonus
   574→  const marketingDemandModifier = marketingEffect.combinedDemandModifier;
   575→  const brandAwarenessBonus = calculateBrandAwarenessDemandBonus(marketingEffect.newBrandAwareness);
   576→
   577→  const potentialCustomers = Math.max(0, Math.floor(
   578→    basePotentialCustomers * previousCxDemandModifier * segmentDemandModifier * marketingDemandModifier * brandAwarenessBonus
   579→  ));
   580→
   581→  // ---- Step 2: Product-level sales simulation ----
   582→  let totalRevenue = 0;
   583→  let totalCOGS = 0;
   584→  const inventoryUpdates: { id: string; quantityDecrement: number }[] = [];
   585→  const salesResults: ProductSalesResult[] = [];
   586→
   587→  for (const inventory of business.inventories) {
   588→    if (inventory.quantity <= 0) continue;
   589→
   590→    const productDef = productDefs.find(p => p.name === inventory.productName);
   591→    if (!productDef) continue;
   592→
   593→    // Product demand config
   594→    const prodDemandConfig = getProductDemandConfig(inventory.productName);
   595→
   596→    // Market reference price = typical retail price (basePrice × (1 + suggestedMarkup) × priceMultiplier)
   597→    // This represents what the MARKET typically charges, not the wholesale cost.
   598→    // Players compare their price to market retail price, not to their own cost.
   599→    const mp = marketPriceMap[inventory.productName];
   600→    const typicalRetailPrice = productDef.basePrice * (1 + productDef.suggestedMarkup);
   601→    const marketReferencePrice = Math.round(
   602→      typicalRetailPrice * (mp?.priceMultiplier || 1)
   603→    );
   604→
   605→    // Market demand multiplier from events
   606→    const demandMult = mp?.demandMultiplier || 1;
   607→
   608→    // Event demand effect for this product's category
   609→    const eventDemandEffect = combinedEffects[`${productDef.category}_demand`] || 0;
   610→
   611→    // Run the complete sales pipeline
   612→    const salesResult = simulateProductSales({
   613→      inventoryId: inventory.id,
   614→      productName: inventory.productName,
   615→      quantity: inventory.quantity,
   616→      purchasePrice: inventory.purchasePrice,
   617→      sellPrice: inventory.sellPrice,
   618→      marketReferencePrice,
   619→      baseDemand: prodDemandConfig.baseDemand,
   620→      demandMultiplier: demandMult,
   621→      eventDemandEffect,
   622→      productPriceSensitivity: prodDemandConfig.priceSensitivity,
   623→      productVolatility: prodDemandConfig.volatility,
   624→      businessPriceSensitivity: economyConfig.priceSensitivity,
   625→      potentialCustomers,
   626→    });
   627→
   628→    if (salesResult.itemsSold > 0) {
   629→      totalRevenue += salesResult.revenue;
   630→      totalCOGS += salesResult.costOfGoodsSold;
   631→      inventoryUpdates.push({
   632→        id: inventory.id,
   633→        quantityDecrement: salesResult.itemsSold,
   634→      });
   635→    }
   636→
   637→    salesResults.push(salesResult);
   638→  }
   639→
   640→  // ---- Step 3: Calculate expenses ----
   641→  const totalMonthlySalaries = business.employees.reduce((sum, emp) => sum + emp.salary, 0);
   642→  const grossProfit = totalRevenue - totalCOGS;
   643→
   644→  const expenses = calculateBusinessExpenses({
   645→    baseRent: businessType.rent,
   646→    level: business.level,
   647→    cityRentMultiplier: city.rentMultiplier,
   648→    totalMonthlySalaries,
   649→    businessLevel: business.level,
   650→    businessTypeId: business.type,
   651→    revenue: totalRevenue,
   652→    grossProfit,
   653→  });
   654→
   655→  // ---- Step 4: Calculate net profit ----
   656→  // Phase 4: Include marketing spend in total expenses
   657→  const totalExpenseWithMarketing = expenses.totalExpense + marketingEffect.totalMarketingSpend;
   658→  const dailyProfit = calculateNetProfit(totalRevenue, totalCOGS, totalExpenseWithMarketing);
   659→
   660→  // ---- Step 5: Calculate reputation change ----
   661→  const outOfStockRatio = business.inventories.length > 0
   662→    ? business.inventories.filter(inv => inv.quantity <= 0).length / business.inventories.length
   663→    : 0;
   664→  const managers = business.employees.filter(e => e.role === 'MANAGER');
   665→  const cleaners = business.employees.filter(e => e.role === 'CLEANER');
   666→
   667→  const reputationChange = calculateReputationChange({
   668→    dailyProfit,
   669→    outOfStockRatio,
   670→    managers: managers.map(m => ({ skill: m.skill })),
   671→    cleaners: cleaners.map(c => ({ skill: c.skill })),
   672→  });
   673→
   674→  const newReputation = Math.max(
   675→    ECONOMY_CONFIG.minReputation,
   676→    Math.min(ECONOMY_CONFIG.maxReputation, business.reputation + reputationChange)
   677→  );
   678→
   679→  // ---- Step 6: Calculate business health score ----
   680→  const healthResult = calculateBusinessHealth({
   681→    dailyProfit,
   682→    dailyRevenue: totalRevenue,
   683→    businessCash: business.cash + dailyProfit,
   684→    totalStock,
   685→    maxStockCapacity,
   686→    reputation: newReputation,
   687→    businessTypeId: business.type,
   688→  });
   689→
   690→  // ---- Step 6.5: Customer Experience (Phase 3) ----
   691→  // Calculate price competitiveness across all products
   692→  const cxPriceCompetitiveness = calculatePriceCompetitiveness(
   693→    business.inventories.map(inv => ({ productName: inv.productName, sellPrice: inv.sellPrice })),
   694→    productDefs.map(p => ({ name: p.name, basePrice: p.basePrice, suggestedMarkup: p.suggestedMarkup })),
   695→    marketPriceMap,
   696→  );
   697→
   698→  // Calculate service quality from employees
   699→  const idealEmployeeCount = 2 + business.level; // Base 2 + 1 per level
   700→  const cxServiceQuality = calculateServiceQuality(
   701→    business.employees.length,
   702→    avgEmployeeSkill,
   703→    idealEmployeeCount,
   704→  );
   705→
   706→  // Calculate product quality from margin (higher margin = better products)
   707→  const cxProductQuality = totalRevenue > 0
   708→    ? Math.min(1, Math.max(0, (totalRevenue - totalCOGS) / totalRevenue / 0.3))
   709→    : 0.5;
   710→
   711→  // Calculate satisfaction
   712→  const cxSatisfaction = calculateSatisfaction({
   713→    priceCompetitiveness: cxPriceCompetitiveness,
   714→    stockAvailability: maxStockCapacity > 0 ? totalStock / maxStockCapacity : 0,
   715→    serviceQuality: cxServiceQuality,
   716→    productQuality: cxProductQuality,
   717→    atmosphere: newReputation / 100,
   718→    currentSatisfaction: business.satisfactionScore,
   719→  });
   720→
   721→  // Determine if this was a positive experience
   722→  const wasPositiveExperience = cxSatisfaction.overall >= 50 && dailyProfit >= 0;
   723→
   724→  // Calculate loyalty
   725→  const cxLoyalty = calculateLoyalty({
   726→    currentLoyaltyScore: business.loyaltyScore,
   727→    currentRepeatRate: business.repeatCustomerRate,
   728→    wasPositiveExperience,
   729→    totalCustomers: potentialCustomers,
   730→    numberOfReviews: business.totalReviews,
   731→    averageRating: business.avgReviewRating,
   732→    loyaltyMultiplier: getLoyaltyTier(business.loyaltyScore).multiplier,
   733→  });
   734→
   735→  // ---- Step 7: Get game day for metrics ----
   736→  // (gameDay already fetched at Step 0.6 for marketing processing)
   737→
   738→  // ---- Step 8: Execute all updates in a single transaction ----
   739→  await db.$transaction(async (tx) => {
   740→    // Update inventory quantities
   741→    for (const upd of inventoryUpdates) {
   742→      await tx.inventory.update({
   743→        where: { id: upd.id },
   744→        data: { quantity: { decrement: upd.quantityDecrement } },
   745→      });
   746→    }
   747→
   748→    // Phase 1 FIX: Distribute business profit to player cash
   749→    // Previously, profit accumulated in business.cash but was never accessible.
   750→    // Now: business profit is transferred to the player each tick.
   751→    const profitToDistribute = dailyProfit;
   752→
   753→    // Update business (including Phase 3 CX fields and Phase 4 Marketing fields)
   754→    await tx.business.update({
   755→      where: { id: business.id },
   756→      data: {
   757→        dailyRevenue: roundTaka(totalRevenue),
   758→        dailyExpense: roundTaka(totalExpenseWithMarketing),
   759→        dailyProfit: roundTaka(dailyProfit),
   760→        dailyCOGS: roundTaka(totalCOGS),
   761→        dailyCustomers: potentialCustomers,
   762→        cash: business.cash + roundTaka(dailyProfit),
   763→        totalRevenue: business.totalRevenue + roundTaka(totalRevenue),
   764→        totalProfit: business.totalProfit + roundTaka(dailyProfit),
   765→        reputation: newReputation,
   766→        healthScore: healthResult.score,
   767→        // Phase 3: CX fields
   768→        satisfactionScore: cxSatisfaction.overall,
   769→        loyaltyScore: cxLoyalty.loyaltyScore,
   770→        repeatCustomerRate: cxLoyalty.repeatCustomerRate,
   771→        // Phase 4: Marketing fields
   772→        brandAwareness: marketingEffect.newBrandAwareness,
   773→      },
   774→    });
   775→
   776→    // Distribute profit to player cash
   777→    await tx.player.update({
   778→      where: { id: business.playerId },
   779→      data: { cash: { increment: roundTaka(profitToDistribute) } },
   780→    });
   781→
   782→    // Phase 4: Attribute revenue to campaigns for ROI analytics
   783→    // NOTE: Marketing spend is already included in totalExpenseWithMarketing
   784→    // which reduced dailyProfit, so it's already deducted from player cash
   785→    // via the profitToDistribute above. Do NOT deduct again.
   786→    if (marketingEffect.totalMarketingSpend > 0 && totalRevenue > 0 && marketingEffect.campaignIds.length > 0) {
   787→      // Fetch campaigns that were processed this tick (using IDs from processMarketingTick)
   788→      const processedCampaigns = await tx.marketingCampaign.findMany({
   789→        where: { id: { in: marketingEffect.campaignIds } },
   790→        select: { id: true, totalSpend: true },
   791→      });
   792→      const totalCampaignSpend = processedCampaigns.reduce((sum, c) => sum + c.totalSpend, 0);
   793→      if (totalCampaignSpend > 0) {
   794→        // Only attribute the "extra" revenue from marketing (not base revenue)
   795→        // Correct formula: total boost = marketingDemandModifier × brandAwarenessBonus
   796→        // Extra fraction = 1 - 1/totalBoost (how much of revenue is due to marketing)
   797→        const totalMarketingBoost = marketingDemandModifier * brandAwarenessBonus;
   798→        const extraFraction = totalMarketingBoost > 1 ? (1 - 1 / totalMarketingBoost) : 0;
   799→        const marketingRevenue = totalRevenue * extraFraction;
   800→
   801→        for (const campaign of processedCampaigns) {
   802→          const spendShare = campaign.totalSpend / totalCampaignSpend;
   803→          const attributedRevenue = Math.max(0, marketingRevenue * spendShare);
   804→          if (attributedRevenue > 0) {
   805→            await tx.marketingCampaign.update({
   806→              where: { id: campaign.id },
   807→              data: { revenueInfluenced: { increment: roundTaka(attributedRevenue) } },
   808→            });
   809→          }
   810→        }
   811→      }
   812→    }
   813→
   814→    // Save daily metrics for analytics (including Phase 3 CX metrics)
   815→    await tx.businessMetric.upsert({
   816→      where: { businessId_gameDay: { businessId: business.id, gameDay } },
   817→      create: {
   818→        businessId: business.id,
   819→        gameDay,
   820→        revenue: roundTaka(totalRevenue),
   821→        expenses: roundTaka(totalExpenseWithMarketing),
   822→        profit: roundTaka(dailyProfit),
   823→        cogs: roundTaka(totalCOGS),
   824→        customers: potentialCustomers,
   825→        reputation: newReputation,
   826→        // Phase 3: CX metrics
   827→        satisfaction: cxSatisfaction.overall,
   828→        loyalty: cxLoyalty.loyaltyScore,
   829→        nps: business.npsScore,
   830→      },
   831→      update: {
   832→        revenue: roundTaka(totalRevenue),
   833→        expenses: roundTaka(totalExpenseWithMarketing),
   834→        profit: roundTaka(dailyProfit),
   835→        cogs: roundTaka(totalCOGS),
   836→        customers: potentialCustomers,
   837→        reputation: newReputation,
   838→        // Phase 3: CX metrics
   839→        satisfaction: cxSatisfaction.overall,
   840→        loyalty: cxLoyalty.loyaltyScore,
   841→        nps: business.npsScore,
   842→      },
   843→    });
   844→
   845→    // Log (with Phase 3 CX info)
   846→    const cogsStr = totalCOGS > 0 ? ` | COGS ৳${roundTaka(totalCOGS).toLocaleString()}` : '';
   847→    await tx.gameLog.create({
   848→      data: {
   849→        playerId: business.playerId,
   850→        businessId: business.id,
   851→        type: dailyProfit >= 0 ? 'PROFIT' : 'LOSS',
   852→        message: `${business.name}: Rev ৳${roundTaka(totalRevenue).toLocaleString()}${cogsStr} | Exp ৳${roundTaka(totalExpenseWithMarketing).toLocaleString()} | ${dailyProfit >= 0 ? 'Profit' : 'Loss'} ৳${Math.abs(roundTaka(dailyProfit)).toLocaleString()} | ${potentialCustomers} customers | Health ${healthResult.score} | Sat ${Math.round(cxSatisfaction.overall)} | Loy ${cxLoyalty.tier} | Mkt ৳${roundTaka(marketingEffect.totalMarketingSpend).toLocaleString()}`,
   853→        amount: dailyProfit,
   854→      },
   855→    });
   856→
   857→    // Phase 3: Generate and persist customer reviews
   858→    const cxReviews = generateReviews(business.id, gameDay, cxSatisfaction, REVIEW_CONFIG.maxPerTick);
   859→    for (const review of cxReviews) {
   860→      await tx.customerReview.create({
   861→        data: {
   862→          businessId: review.businessId,
   863→          gameDay: review.gameDay,
   864→          rating: review.rating,
   865→          sentiment: review.sentiment,
   866→          category: review.category,
   867→          comment: review.comment,
   868→          segment: review.segment,
   869→        },
   870→      });
   871→    }
   872→
   873→    // Phase 3: Update NPS and review stats
   874→    if (cxReviews.length > 0) {
   875→      const allReviews = await tx.customerReview.findMany({
   876→        where: { businessId: business.id },
   877→        select: { rating: true },
   878→        orderBy: { createdAt: 'desc' },
   879→        take: REVIEW_CONFIG.keepLast,
   880→      });
   881→      const npsResult = calculateNPS(allReviews);
   882→      const totalReviews = allReviews.length;
   883→      const avgRating = totalReviews > 0
   884→        ? allReviews.reduce((sum, r) => sum + r.rating, 0) / totalReviews
   885→        : 0;
   886→
   887→      await tx.business.update({
   888→        where: { id: business.id },
   889→        data: {
   890→          npsScore: npsResult.nps,
   891→          totalReviews,
   892→          avgReviewRating: Math.round(avgRating * 100) / 100,
   893→        },
   894→      });
   895→
   896→      // Prune old reviews (keep only last N)
   897→      const reviewCount = await tx.customerReview.count({
   898→        where: { businessId: business.id },
   899→      });
   900→      if (reviewCount > REVIEW_CONFIG.keepLast) {
   901→        const oldReviews = await tx.customerReview.findMany({
   902→          where: { businessId: business.id },
   903→          orderBy: { createdAt: 'asc' },
   904→          take: reviewCount - REVIEW_CONFIG.keepLast,
   905→          select: { id: true },
   906→        });
   907→        for (const old of oldReviews) {
   908→          await tx.customerReview.delete({ where: { id: old.id } });
   909→        }
   910→      }
   911→    }
   912→
   913→    // Recalculate net worth within the same transaction
   914→    await recalculateNetWorth(tx, business.playerId);
   915→  });
   916→}

   918→// ---- Loan Payment Processing ----
   919→
   920→async function processLoanPayments(): Promise<void> {
   921→  const activeLoans = await db.loan.findMany({
   922→    where: { status: 'ACTIVE' },
   923→  });
   924→
   925→  if (activeLoans.length === 0) return;
   926→
   927→  for (const loan of activeLoans) {
   928→    const payment = loan.dailyPayment;
   929→    const isPaidOff = loan.daysRemaining - 1 <= 0;
   930→    const actualPayment = isPaidOff ? loan.remainingDebt : payment;
   931→
   932→    await db.$transaction(async (tx) => {
   933→      const player = await tx.player.findUnique({
   934→        where: { id: loan.playerId },
   935→        select: { cash: true },
   936→      });
   937→      if (!player) {
   938→        console.error(`[GameEngine] processLoanPayments: Player ${loan.playerId} not found for loan ${loan.id}`);
   939→        return;
   940→      }
   941→
   942→      const deductAmount = Math.min(actualPayment, Math.max(0, player.cash));
   943→      const newRemainingDebt = Math.max(loan.remainingDebt - deductAmount, 0);
   944→
   945→      await tx.loan.update({
   946→        where: { id: loan.id },
   947→        data: {
   948→          remainingDebt: newRemainingDebt,
   949→          daysRemaining: Math.max(loan.daysRemaining - 1, 0),
   950→          status: newRemainingDebt <= 0 ? 'PAID_OFF' : 'ACTIVE',
   951→        },
   952→      });
   953→
   954→      await tx.player.update({
   955→        where: { id: loan.playerId },
   956→        data: { cash: { decrement: deductAmount } },
   957→      });
   958→
   959→      await tx.gameLog.create({
   960→        data: {
   961→          playerId: loan.playerId,
   962→          type: newRemainingDebt <= 0 ? 'LOAN_PAID' : 'LOAN_PAYMENT',
   963→          message: newRemainingDebt <= 0
   964→            ? `Loan of ৳${loan.amount.toLocaleString()} fully repaid! Final deduction: ৳${Math.round(deductAmount).toLocaleString()}`
   965→            : `Loan payment: ৳${Math.round(deductAmount).toLocaleString()} deducted. Remaining debt: ৳${Math.round(newRemainingDebt).toLocaleString()} (${loan.daysRemaining - 1} days left)`,
   966→          amount: -deductAmount,
   967→        },
   968→      });
   969→    });
   970→  }
   971→}

   975→export async function gameTick(): Promise<void> {
   976→  // NOTE: The tick lock is managed by the API route handler.
   977→  // gameTick() is the pure simulation function and does NOT acquire/release the lock.
   978→
   979→  // Log tick start
   980→  await db.gameLog.create({
   981→    data: {
   982→      type: 'GAME_TICK_STARTED',
   983→      message: `Game tick started at ${new Date().toISOString()}`,
   984→    },
   985→  });
   986→
   987→  // 0. Game state housekeeping
   988→  const tickState = await db.gameState.findUnique({ where: { key: 'tickCount' } });
   989→  const tickNum = parseInt(tickState?.value || '0');
   990→
   991→  await db.$transaction(async (tx) => {
   992→    // Expire old events
   993→    await tx.gameEvent.updateMany({
   994→      where: { active: true, endsAt: { lt: new Date() } },
   995→      data: { active: false },
   996→    });
   997→
   998→    // Update game state
   999→    const now = new Date().toISOString();
  1000→    await tx.gameState.upsert({
  1001→      where: { key: 'lastTick' },
  1002→      create: { key: 'lastTick', value: now },
  1003→      update: { value: now },
  1004→    });
  1005→    await tx.gameState.upsert({
  1006→      where: { key: 'tickCount' },
  1007→      create: { key: 'tickCount', value: '1' },
  1008→      update: { value: String(tickNum + 1) },
  1009→    });
  1010→    const dayState = await tx.gameState.findUnique({ where: { key: 'gameDay' } });
  1011→    await tx.gameState.upsert({
  1012→      where: { key: 'gameDay' },
  1013→      create: { key: 'gameDay', value: '1' },
  1014→      update: { value: String(parseInt(dayState?.value || '0') + 1) },
  1015→    });
  1016→  });
  1017→
  1018→  // Generate new events (20% chance)
  1019→  if (Math.random() < 0.2) {
  1020→    try {
  1021→      await generateEvent();
  1022→    } catch (err) {
  1023→      console.error('[GameEngine] Failed to generate event:', err);
  1024→    }
  1025→  }
  1026→
  1027→  // Update market prices (every 3 ticks)
  1028→  // Phase 1: This now uses retention-based updates with event effects integrated
  1029→  if (tickNum % 3 === 0) {
  1030→    try {
  1031→      await updateMarketPrices();
  1032→    } catch (err) {
  1033→      console.error('[GameEngine] Failed to update market prices:', err);
  1034→    }
  1035→    // applyEventEffects() is now a no-op — events handled in updateMarketPrices()
  1036→  }
  1037→
  1038→  // 1. Simulate all businesses
  1039→  const businesses = await db.business.findMany({ select: { id: true } });
  1040→  for (const business of businesses) {
  1041→    try {
  1042→      await simulateBusinessTick(business.id);
  1043→    } catch (err) {
  1044→      console.error(`[GameEngine] Failed to simulate business ${business.id}:`, err);
  1045→    }
  1046→  }
  1047→
  1048→  // 2. Process loan payments
  1049→  try {
  1050→    await processLoanPayments();
  1051→  } catch (err) {
  1052→    console.error('[GameEngine] Failed to process loan payments:', err);
  1053→  }
  1054→
  1055→  // 3. Simulate AI player activity (Phase 2: Real AI decisions)
  1056→  try {
  1057→    const gameDayState2 = await db.gameState.findUnique({ where: { key: 'gameDay' } });
  1058→    const currentGameDay = parseInt(gameDayState2?.value || '1', 10);
  1059→    await simulateAIPlayersTick(currentGameDay);
  1060→  } catch (err) {
  1061→    console.error('[GameEngine] Failed to simulate AI tick:', err);
  1062→  }
  1063→
  1064→  // 4. Generate news (50% chance)
  1065→  if (Math.random() < 0.5) {
  1066→    try {
  1067→      await generateNews();
  1068→    } catch (err) {
  1069→      console.error('[GameEngine] Failed to generate news:', err);
  1070→    }
  1071→  }
  1072→
  1073→  // Log tick completion
  1074→  await db.gameLog.create({
  1075→    data: {
  1076→      type: 'GAME_TICK_COMPLETED',
  1077→      message: `Game tick completed at ${new Date().toISOString()}`,
  1078→    },
  1079→  });
  1080→}
