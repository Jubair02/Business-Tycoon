    48→    48→// ---- Tick Concurrency Lock Constants ----
    49→    49→const STALE_TICK_TIMEOUT_MS = 2 * 60 * 1000; // 2 minutes
    50→    50→
    51→    51→// ---- Tick Concurrency Lock ----
    52→    52→
    53→    53→/**
    54→    54→ * Atomically attempt to acquire the tick lock using the GameState key-value store.
    55→    55→ * Returns true if the lock was acquired, false if it was already held.
    56→    56→ *
    57→    57→ * Stale lock recovery: if the lock has been held for longer than
    58→    58→ * STALE_TICK_TIMEOUT_MS, the lock is stolen and a warning is logged.
    59→    59→ */
    60→    60→export async function acquireTickLock(): Promise<boolean> {
    61→    61→  const now = new Date();
    62→    62→
    63→    63→  return db.$transaction(async (tx) => {
    64→    64→    // Read current lock state
    65→    65→    const lockState = await tx.gameState.findUnique({ where: { key: 'tickInProgress' } });
    66→    66→    const startedAtState = await tx.gameState.findUnique({ where: { key: 'tickStartedAt' } });
    67→    67→
    68→    68→    const isLocked = lockState?.value === 'true';
    69→    69→    const startedAt = startedAtState?.value ? new Date(startedAtState.value) : null;
    70→    70→
    71→    71→    if (isLocked && startedAt) {
    72→    72→      const elapsed = now.getTime() - startedAt.getTime();
    73→    73→      if (elapsed < STALE_TICK_TIMEOUT_MS) {
    74→    74→        return false; // Lock is legitimately held
    75→    75→      }
    76→    76→      console.warn(`[GameEngine] Stale tick lock detected (held for ${Math.round(elapsed / 1000)}s). Stealing lock.`);
    77→    77→    } else if (isLocked) {
    78→    78→      console.warn('[GameEngine] Tick lock held with no timestamp. Stealing lock.');
    79→    79→    }
    80→    80→
    81→    81→    // Acquire the lock
    82→    82→    const nowISO = now.toISOString();
    83→    83→    await tx.gameState.upsert({
    84→    84→      where: { key: 'tickInProgress' },
    85→    85→      create: { key: 'tickInProgress', value: 'true' },
    86→    86→      update: { value: 'true' },
    87→    87→    });
    88→    88→    await tx.gameState.upsert({
    89→    89→      where: { key: 'tickStartedAt' },
    90→    90→      create: { key: 'tickStartedAt', value: nowISO },
    91→    91→      update: { value: nowISO },
    92→    92→    });
    93→    93→
    94→    94→    // Increment tick version
    95→    95→    const versionState = await tx.gameState.findUnique({ where: { key: 'tickVersion' } });
    96→    96→    const currentVersion = parseInt(versionState?.value || '0', 10);
    97→    97→    await tx.gameState.upsert({
    98→    98→      where: { key: 'tickVersion' },
    99→    99→      create: { key: 'tickVersion', value: '1' },
   100→   100→      update: { value: String(currentVersion + 1) },
   101→   101→    });
   102→   102→
   103→   103→    return true;
   104→   104→  });
   105→   105→}

   116→   116→async function recalculateNetWorth(tx: PrismaTx, playerId: string): Promise<void> {
   117→   117→  const player = await tx.player.findUnique({
   118→   118→    where: { id: playerId },
   119→   119→    select: { id: true, cash: true },
   120→   120→  });
   121→   121→  if (!player) {
   122→   122→    console.error(`[GameEngine] recalculateNetWorth: Player ${playerId} not found`);
   123→   123→    return;
   124→   124→  }
   125→   125→
   126→   126→  const allBusinesses = await tx.business.findMany({
   127→   127→    where: { playerId },
   128→   128→    select: { cash: true },
   129→   129→  });
   130→   130→  const totalBusinessCash = allBusinesses.reduce((sum, b) => sum + b.cash, 0);
   131→   131→
   132→   132→  const allInventory = await tx.inventory.findMany({
   133→   133→    where: { business: { playerId } },
   134→   134→    select: { quantity: true, purchasePrice: true },
   135→   135→  });
   136→   136→  const totalInventoryValue = allInventory.reduce((sum, inv) => sum + inv.quantity * inv.purchasePrice, 0);
   137→   137→
   138→   138→  const activeLoans = await tx.loan.findMany({
   139→   139→    where: { playerId, status: 'ACTIVE' },
   140→   140→    select: { remainingDebt: true },
   141→   141→  });
   142→   142→  const outstandingDebt = activeLoans.reduce((sum, loan) => sum + loan.remainingDebt, 0);
   143→   143→
   144→   144→  const netWorth = player.cash + totalBusinessCash + totalInventoryValue - outstandingDebt;
   145→   145→  await tx.player.update({ where: { id: playerId }, data: { netWorth } });
   146→   146→}

   149→   149→export async function generateEvent(): Promise<void> {
   150→   150→  const totalWeight = EVENT_TEMPLATES.reduce((sum, e) => sum + e.weight, 0);
   151→   151→  let random = Math.random() * totalWeight;
   152→   152→  let selected: EventTemplate | null = null;
   153→   153→  for (const event of EVENT_TEMPLATES) {
   154→   154→    random -= event.weight;
   155→   155→    if (random <= 0) {
   156→   156→      selected = event;
   157→   157→      break;
   158→   158→    }
   159→   159→  }
   160→   160→  if (!selected) selected = EVENT_TEMPLATES[0];
   161→   161→
   162→   162→  const now = new Date();
   163→   163→  const endAt = new Date(now.getTime() + selected.durationDays * 60 * 1000);
   164→   164→  await db.gameEvent.create({
   165→   165→    data: {
   166→   166→      title: selected.title,
   167→   167→      description: selected.description,
   168→   168→      type: selected.type,
   169→   169→      effects: JSON.stringify(selected.effects),
   170→   170→      active: true,
   171→   171→      startsAt: now,
   172→   172→      endsAt: endAt,
   173→   173→    },
   174→   174→  });
   175→   175→}

   183→   183→export async function updateMarketPrices(): Promise<void> {
   184→   184→  const allProducts = Object.values(PRODUCTS).flat();
   185→   185→  const cities = CITIES.map(c => c.id);
   186→   186→
   187→   187→  // Get active events for price modifiers
   188→   188→  const activeEvents = await db.gameEvent.findMany({ where: { active: true } });
   189→   189→  const categoryPriceMod: Record<string, number> = {};
   190→   190→  let allPriceMod = 0;
   191→   191→  for (const event of activeEvents) {
   192→   192→    const effects = JSON.parse(event.effects) as Record<string, number>;
   193→   193→    for (const [key, value] of Object.entries(effects)) {
   194→   194→      if (key === 'all_price' || key === 'all_imported') allPriceMod += value;
   195→   195→      if (key.endsWith('_price')) {
   196→   196→        const cat = key.replace('_price', '');
   197→   197→        categoryPriceMod[cat] = (categoryPriceMod[cat] || 0) + value;
   198→   198→      }
   199→   199→    }
   200→   200→  }
   201→   201→
   202→   202→  const updates: { productName: string; city: string; priceMultiplier: number; demandMultiplier: number }[] = [];
   203→   203→
   204→   204→  for (const city of cities) {
   205→   205→    for (const product of allProducts) {
   206→   206→      // Get current market price (for retention)
   207→   207→      const existing = await db.marketPrice.findUnique({
   208→   208→        where: { productName_city: { productName: product.name, city } },
   209→   209→      });
   210→   210→      const previousPriceMult = existing?.priceMultiplier ?? 1;
   211→   211→
   212→   212→      // Calculate event price modifier for this product
   213→   213→      const eventPriceMod = allPriceMod + (categoryPriceMod[product.category] || 0);
   214→   214→
   215→   215→      // Use Phase 1 retention-based update
   216→   216→      const { priceMultiplier, demandMultiplier } = calculateMarketPriceUpdate(
   217→   217→        previousPriceMult,
   218→   218→        eventPriceMod
   219→   219→      );
   220→   220→
   221→   221→      updates.push({ productName: product.name, city, priceMultiplier, demandMultiplier });
   222→   222→    }
   223→   223→  }
   224→   224→
   225→   225→  // Batch update in a transaction
   226→   226→  await db.$transaction(async (tx) => {
   227→   227→    for (const upd of updates) {
   228→   228→      await tx.marketPrice.upsert({
   229→   229→        where: { productName_city: { productName: upd.productName, city: upd.city } },
   230→   230→        create: upd,
   231→   231→        update: {
   232→   232→          priceMultiplier: upd.priceMultiplier,
   233→   233→          demandMultiplier: upd.demandMultiplier,
   234→   234→          updatedAt: new Date(),
   235→   235→        },
   236→   236→      });
   237→   237→    }
   238→   238→  });
   239→   239→}

   273→   273→export async function simulateBusinessTick(businessId: string): Promise<void> {
   274→   274→  const business = await db.business.findUnique({
   275→   275→    where: { id: businessId },
   276→   276→    include: {
   277→   277→      inventories: true,
   278→   278→      employees: true,
   279→   279→      player: { select: { id: true, cash: true, netWorth: true } },
   280→   280→    },
   281→   281→  });
   282→   282→
   283→   283→  if (!business) return;
   284→   284→
   285→   285→  const businessType = getBusinessType(business.type);
   286→   286→  const city = getCity(business.city);
   287→   287→  if (!businessType || !city) return;
   288→   288→
   289→   289→  const economyConfig = getBusinessEconomyConfig(business.type);
   290→   290→
   291→   291→  // Get active event effects (combined)
   292→   292→  const activeEvents = await db.gameEvent.findMany({
   293→   293→    where: { active: true },
   294→   294→    select: { effects: true },
   295→   295→  });
   296→   296→  const combinedEffects: Record<string, number> = {};
   297→   297→  for (const event of activeEvents) {
   298→   298→    const effects = JSON.parse(event.effects) as Record<string, number>;
   299→   299→    for (const [key, value] of Object.entries(effects)) {
   300→   300→      combinedEffects[key] = (combinedEffects[key] || 0) + value;
   301→   301→    }
   302→   302→  }
   303→   303→
   304→   304→  // Get market prices for this city
   305→   305→  const marketPrices = await db.marketPrice.findMany({
   306→   306→    where: { city: business.city },
   307→   307→  });
   308→   308→  const marketPriceMap: Record<string, { priceMultiplier: number; demandMultiplier: number }> = {};
   309→   309→  for (const mp of marketPrices) {
   310→   310→    marketPriceMap[mp.productName] = {
   311→   311→      priceMultiplier: mp.priceMultiplier,
   312→   312→      demandMultiplier: mp.demandMultiplier,
   313→   313→    };
   314→   314→  }
   315→   315→
   316→   316→  // ---- Step 1: Calculate potential customers ----
   317→   317→  const totalStock = business.inventories.reduce((sum, inv) => sum + inv.quantity, 0);
   318→   318→  // Max stock capacity: sum of product maxStock values for this business type
   319→   319→  const productDefs = PRODUCTS[business.type] || [];
   320→   320→  const maxStockCapacity = productDefs.reduce((sum, p) => sum + p.maxStock, 0);
   321→   321→  const avgEmployeeSkill = business.employees.length > 0
   322→   322→    ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   323→   323→    : 0;
   324→   324→
   325→   325→  const potentialCustomers = calculatePotentialCustomers({
   326→   326→    baseCustomers: businessType.baseCustomers,
   327→   327→    cityMultiplier: city.customerMultiplier,
   328→   328→    level: business.level,
   329→   329→    reputation: business.reputation,
   330→   330→    totalStock,
   331→   331→    maxStockCapacity,
   332→   332→    employeeCount: business.employees.length,
   333→   333→    avgEmployeeSkill,
   334→   334→    eventCustomerEffect: combinedEffects['all_customers'] || 0,
   335→   335→    businessDemandEffect: combinedEffects[`${businessType.id}_demand`] || 0,
   336→   336→    businessTypeId: business.type,
   337→   337→  });
   338→   338→
   339→   339→  // ---- Step 2: Product-level sales simulation ----
   340→   340→  let totalRevenue = 0;
   341→   341→  let totalCOGS = 0;
   342→   342→  const inventoryUpdates: { id: string; quantityDecrement: number }[] = [];
   343→   343→  const salesResults: ProductSalesResult[] = [];
   344→   344→
   345→   345→  for (const inventory of business.inventories) {
   346→   346→    if (inventory.quantity <= 0) continue;
   347→   347→
   348→   348→    const productDef = productDefs.find(p => p.name === inventory.productName);
   349→   349→    if (!productDef) continue;
   350→   350→
   351→   351→    // Product demand config
   352→   352→    const prodDemandConfig = getProductDemandConfig(inventory.productName);
   353→   353→
   354→   354→    // Market reference price = typical retail price (basePrice × (1 + suggestedMarkup) × priceMultiplier)
   355→   355→    // This represents what the MARKET typically charges, not the wholesale cost.
   356→   356→    // Players compare their price to market retail price, not to their own cost.
   357→   357→    const mp = marketPriceMap[inventory.productName];
   358→   358→    const typicalRetailPrice = productDef.basePrice * (1 + productDef.suggestedMarkup);
   359→   359→    const marketReferencePrice = Math.round(
   360→   360→      typicalRetailPrice * (mp?.priceMultiplier || 1)
   361→   361→    );
   362→   362→
   363→   363→    // Market demand multiplier from events
   364→   364→    const demandMult = mp?.demandMultiplier || 1;
   365→   365→
   366→   366→    // Event demand effect for this product's category
   367→   367→    const eventDemandEffect = combinedEffects[`${productDef.category}_demand`] || 0;
   368→   368→
   369→   369→    // Run the complete sales pipeline
   370→   370→    const salesResult = simulateProductSales({
   371→   371→      inventoryId: inventory.id,
   372→   372→      productName: inventory.productName,
   373→   373→      quantity: inventory.quantity,
   374→   374→      purchasePrice: inventory.purchasePrice,
   375→   375→      sellPrice: inventory.sellPrice,
   376→   376→      marketReferencePrice,
   377→   377→      baseDemand: prodDemandConfig.baseDemand,
   378→   378→      demandMultiplier: demandMult,
   379→   379→      eventDemandEffect,
   380→   380→      productPriceSensitivity: prodDemandConfig.priceSensitivity,
   381→   381→      productVolatility: prodDemandConfig.volatility,
   382→   382→      businessPriceSensitivity: economyConfig.priceSensitivity,
   383→   383→      potentialCustomers,
   384→   384→    });
   385→   385→
   386→   386→    if (salesResult.itemsSold > 0) {
   387→   387→      totalRevenue += salesResult.revenue;
   388→   388→      totalCOGS += salesResult.costOfGoodsSold;
   389→   389→      inventoryUpdates.push({
   390→   390→        id: inventory.id,
   391→   391→        quantityDecrement: salesResult.itemsSold,
   392→   392→      });
   393→   393→    }
   394→   394→
   395→   395→    salesResults.push(salesResult);
   396→   396→  }
   397→   397→
   398→   398→  // ---- Step 3: Calculate expenses ----
   399→   399→  const totalMonthlySalaries = business.employees.reduce((sum, emp) => sum + emp.salary, 0);
   400→   400→  const grossProfit = totalRevenue - totalCOGS;
   401→   401→
   402→   402→  const expenses = calculateBusinessExpenses({
   403→   403→    baseRent: businessType.rent,
   404→   404→    level: business.level,
   405→   405→    cityRentMultiplier: city.rentMultiplier,
   406→   406→    totalMonthlySalaries,
   407→   407→    businessLevel: business.level,
   408→   408→    businessTypeId: business.type,
   409→   409→    revenue: totalRevenue,
   410→   410→    grossProfit,
   411→   411→  });
   412→   412→
   413→   413→  // ---- Step 4: Calculate net profit ----
   414→   414→  const dailyProfit = calculateNetProfit(totalRevenue, totalCOGS, expenses.totalExpense);
   415→   415→
   416→   416→  // ---- Step 5: Calculate reputation change ----
   417→   417→  const outOfStockRatio = business.inventories.length > 0
   418→   418→    ? business.inventories.filter(inv => inv.quantity <= 0).length / business.inventories.length
   419→   419→    : 0;
   420→   420→  const managers = business.employees.filter(e => e.role === 'MANAGER');
   421→   421→  const cleaners = business.employees.filter(e => e.role === 'CLEANER');
   422→   422→
   423→   423→  const reputationChange = calculateReputationChange({
   424→   424→    dailyProfit,
   425→   425→    outOfStockRatio,
   426→   426→    managers: managers.map(m => ({ skill: m.skill })),
   427→   427→    cleaners: cleaners.map(c => ({ skill: c.skill })),
   428→   428→  });
   429→   429→
   430→   430→  const newReputation = Math.max(
   431→   431→    ECONOMY_CONFIG.minReputation,
   432→   432→    Math.min(ECONOMY_CONFIG.maxReputation, business.reputation + reputationChange)
   433→   433→  );
   434→   434→
   435→   435→  // ---- Step 6: Calculate business health score ----
   436→   436→  const healthResult = calculateBusinessHealth({
   437→   437→    dailyProfit,
   438→   438→    dailyRevenue: totalRevenue,
   439→   439→    businessCash: business.cash + dailyProfit,
   440→   440→    totalStock,
   441→   441→    maxStockCapacity,
   442→   442→    reputation: newReputation,
   443→   443→    businessTypeId: business.type,
   444→   444→  });
   445→   445→
   446→   446→  // ---- Step 7: Get game day for metrics ----
   447→   447→  const gameDayState = await db.gameState.findUnique({ where: { key: 'gameDay' } });
   448→   448→  const gameDay = parseInt(gameDayState?.value || '1', 10);
   449→   449→
   450→   450→  // ---- Step 8: Execute all updates in a single transaction ----
   451→   451→  await db.$transaction(async (tx) => {
   452→   452→    // Update inventory quantities
   453→   453→    for (const upd of inventoryUpdates) {
   454→   454→      await tx.inventory.update({
   455→   455→        where: { id: upd.id },
   456→   456→        data: { quantity: { decrement: upd.quantityDecrement } },
   457→   457→      });
   458→   458→    }
   459→   459→
   460→   460→    // Phase 1 FIX: Distribute business profit to player cash
   461→   461→    // Previously, profit accumulated in business.cash but was never accessible.
   462→   462→    // Now: business profit is transferred to the player each tick.
   463→   463→    const profitToDistribute = dailyProfit;
   464→   464→
   465→   465→    // Update business
   466→   466→    await tx.business.update({
   467→   467→      where: { id: business.id },
   468→   468→      data: {

   528→   528→// ---- Loan Payment Processing ----
   529→   529→
   530→   530→async function processLoanPayments(): Promise<void> {
   531→   531→  const activeLoans = await db.loan.findMany({
   532→   532→    where: { status: 'ACTIVE' },
   533→   533→  });
   534→   534→
   535→   535→  if (activeLoans.length === 0) return;
   536→   536→
   537→   537→  for (const loan of activeLoans) {
   538→   538→    const payment = loan.dailyPayment;
   539→   539→    const isPaidOff = loan.daysRemaining - 1 <= 0;
   540→   540→    const actualPayment = isPaidOff ? loan.remainingDebt : payment;
   541→   541→
   542→   542→    await db.$transaction(async (tx) => {
   543→   543→      const player = await tx.player.findUnique({
   544→   544→        where: { id: loan.playerId },
   545→   545→        select: { cash: true },
   546→   546→      });
   547→   547→      if (!player) {
   548→   548→        console.error(`[GameEngine] processLoanPayments: Player ${loan.playerId} not found for loan ${loan.id}`);
   549→   549→        return;
   550→   550→      }
   551→   551→
   552→   552→      const deductAmount = Math.min(actualPayment, Math.max(0, player.cash));
   553→   553→      const newRemainingDebt = Math.max(loan.remainingDebt - deductAmount, 0);
   554→   554→
   555→   555→      await tx.loan.update({
   556→   556→        where: { id: loan.id },
   557→   557→        data: {
   558→   558→          remainingDebt: newRemainingDebt,
   559→   559→          daysRemaining: Math.max(loan.daysRemaining - 1, 0),
   560→   560→          status: newRemainingDebt <= 0 ? 'PAID_OFF' : 'ACTIVE',
   561→   561→        },
   562→   562→      });
   563→   563→
   564→   564→      await tx.player.update({
   565→   565→        where: { id: loan.playerId },
   566→   566→        data: { cash: { decrement: deductAmount } },
   567→   567→      });
   568→   568→
   569→   569→      await tx.gameLog.create({
   570→   570→        data: {
   571→   571→          playerId: loan.playerId,
   572→   572→          type: newRemainingDebt <= 0 ? 'LOAN_PAID' : 'LOAN_PAYMENT',
   573→   573→          message: newRemainingDebt <= 0
   574→   574→            ? `Loan of ৳${loan.amount.toLocaleString()} fully repaid! Final deduction: ৳${Math.round(deductAmount).toLocaleString()}`
   575→   575→            : `Loan payment: ৳${Math.round(deductAmount).toLocaleString()} deducted. Remaining debt: ৳${Math.round(newRemainingDebt).toLocaleString()} (${loan.daysRemaining - 1} days left)`,
   576→   576→          amount: -deductAmount,
   577→   577→        },
   578→   578→      });
   579→   579→    });
   580→   580→  }
   581→   581→}

   585→   585→export async function gameTick(): Promise<void> {
   586→   586→  // NOTE: The tick lock is managed by the API route handler.
   587→   587→  // gameTick() is the pure simulation function and does NOT acquire/release the lock.
   588→   588→
   589→   589→  // Log tick start
   590→   590→  await db.gameLog.create({
   591→   591→    data: {
   592→   592→      type: 'GAME_TICK_STARTED',
   593→   593→      message: `Game tick started at ${new Date().toISOString()}`,
   594→   594→    },
   595→   595→  });
   596→   596→
   597→   597→  // 0. Game state housekeeping
   598→   598→  const tickState = await db.gameState.findUnique({ where: { key: 'tickCount' } });
   599→   599→  const tickNum = parseInt(tickState?.value || '0');
   600→   600→
   601→   601→  await db.$transaction(async (tx) => {
   602→   602→    // Expire old events
   603→   603→    await tx.gameEvent.updateMany({
   604→   604→      where: { active: true, endsAt: { lt: new Date() } },
   605→   605→      data: { active: false },
   606→   606→    });
   607→   607→
   608→   608→    // Update game state
   609→   609→    const now = new Date().toISOString();
   610→   610→    await tx.gameState.upsert({
   611→   611→      where: { key: 'lastTick' },
   612→   612→      create: { key: 'lastTick', value: now },
   613→   613→      update: { value: now },
   614→   614→    });
   615→   615→    await tx.gameState.upsert({
   616→   616→      where: { key: 'tickCount' },
   617→   617→      create: { key: 'tickCount', value: '1' },
   618→   618→      update: { value: String(tickNum + 1) },
   619→   619→    });
   620→   620→    const dayState = await tx.gameState.findUnique({ where: { key: 'gameDay' } });
   621→   621→    await tx.gameState.upsert({
   622→   622→      where: { key: 'gameDay' },
   623→   623→      create: { key: 'gameDay', value: '1' },
   624→   624→      update: { value: String(parseInt(dayState?.value || '0') + 1) },
   625→   625→    });
   626→   626→  });
   627→   627→
   628→   628→  // Generate new events (20% chance)
   629→   629→  if (Math.random() < 0.2) {
   630→   630→    try {
   631→   631→      await generateEvent();
   632→   632→    } catch (err) {
   633→   633→      console.error('[GameEngine] Failed to generate event:', err);
   634→   634→    }
   635→   635→  }
   636→   636→
   637→   637→  // Update market prices (every 3 ticks)
   638→   638→  // Phase 1: This now uses retention-based updates with event effects integrated
   639→   639→  if (tickNum % 3 === 0) {
   640→   640→    try {
   641→   641→      await updateMarketPrices();
   642→   642→    } catch (err) {
   643→   643→      console.error('[GameEngine] Failed to update market prices:', err);
   644→   644→    }
   645→   645→    // applyEventEffects() is now a no-op — events handled in updateMarketPrices()
   646→   646→  }
   647→   647→
   648→   648→  // 1. Simulate all businesses
   649→   649→  const businesses = await db.business.findMany({ select: { id: true } });
   650→   650→  for (const business of businesses) {
   651→   651→    try {
   652→   652→      await simulateBusinessTick(business.id);
   653→   653→    } catch (err) {
   654→   654→      console.error(`[GameEngine] Failed to simulate business ${business.id}:`, err);
   655→   655→    }
   656→   656→  }
   657→   657→
   658→   658→  // 2. Process loan payments
   659→   659→  try {
   660→   660→    await processLoanPayments();
   661→   661→  } catch (err) {
   662→   662→    console.error('[GameEngine] Failed to process loan payments:', err);
   663→   663→  }
   664→   664→
   665→   665→  // 3. Simulate AI player activity
   666→   666→  try {
   667→   667→    await simulateAITick();
   668→   668→  } catch (err) {
   669→   669→    console.error('[GameEngine] Failed to simulate AI tick:', err);
   670→   670→  }
   671→   671→
   672→   672→  // 4. Generate news (50% chance)
   673→   673→  if (Math.random() < 0.5) {
   674→   674→    try {
   675→   675→      await generateNews();
   676→   676→    } catch (err) {
   677→   677→      console.error('[GameEngine] Failed to generate news:', err);
   678→   678→    }
   679→   679→  }
   680→   680→
   681→   681→  // Log tick completion
   682→   682→  await db.gameLog.create({
   683→   683→    data: {
   684→   684→      type: 'GAME_TICK_COMPLETED',
   685→   685→      message: `Game tick completed at ${new Date().toISOString()}`,
   686→   686→    },
   687→   687→  });
   688→   688→}
