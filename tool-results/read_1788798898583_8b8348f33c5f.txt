    51→    51→// ---- Tick Concurrency Lock Constants ----
    52→    52→const STALE_TICK_TIMEOUT_MS = 2 * 60 * 1000; // 2 minutes
    53→    53→
    54→    54→// ---- Tick Concurrency Lock ----
    55→    55→
    56→    56→/**
    57→    57→ * Atomically attempt to acquire the tick lock using the GameState key-value store.
    58→    58→ * Returns true if the lock was acquired, false if it was already held.
    59→    59→ *
    60→    60→ * Stale lock recovery: if the lock has been held for longer than
    61→    61→ * STALE_TICK_TIMEOUT_MS, the lock is stolen and a warning is logged.
    62→    62→ */
    63→    63→export async function acquireTickLock(): Promise<boolean> {
    64→    64→  const now = new Date();
    65→    65→
    66→    66→  return db.$transaction(async (tx) => {
    67→    67→    // Read current lock state
    68→    68→    const lockState = await tx.gameState.findUnique({ where: { key: 'tickInProgress' } });
    69→    69→    const startedAtState = await tx.gameState.findUnique({ where: { key: 'tickStartedAt' } });
    70→    70→
    71→    71→    const isLocked = lockState?.value === 'true';
    72→    72→    const startedAt = startedAtState?.value ? new Date(startedAtState.value) : null;
    73→    73→
    74→    74→    if (isLocked && startedAt) {
    75→    75→      const elapsed = now.getTime() - startedAt.getTime();
    76→    76→      if (elapsed < STALE_TICK_TIMEOUT_MS) {
    77→    77→        return false; // Lock is legitimately held
    78→    78→      }
    79→    79→      console.warn(`[GameEngine] Stale tick lock detected (held for ${Math.round(elapsed / 1000)}s). Stealing lock.`);
    80→    80→    } else if (isLocked) {
    81→    81→      console.warn('[GameEngine] Tick lock held with no timestamp. Stealing lock.');
    82→    82→    }
    83→    83→
    84→    84→    // Acquire the lock
    85→    85→    const nowISO = now.toISOString();
    86→    86→    await tx.gameState.upsert({
    87→    87→      where: { key: 'tickInProgress' },
    88→    88→      create: { key: 'tickInProgress', value: 'true' },
    89→    89→      update: { value: 'true' },
    90→    90→    });
    91→    91→    await tx.gameState.upsert({
    92→    92→      where: { key: 'tickStartedAt' },
    93→    93→      create: { key: 'tickStartedAt', value: nowISO },
    94→    94→      update: { value: nowISO },
    95→    95→    });
    96→    96→
    97→    97→    // Increment tick version
    98→    98→    const versionState = await tx.gameState.findUnique({ where: { key: 'tickVersion' } });
    99→    99→    const currentVersion = parseInt(versionState?.value || '0', 10);
   100→   100→    await tx.gameState.upsert({
   101→   101→      where: { key: 'tickVersion' },
   102→   102→      create: { key: 'tickVersion', value: '1' },
   103→   103→      update: { value: String(currentVersion + 1) },
   104→   104→    });
   105→   105→
   106→   106→    return true;
   107→   107→  });
   108→   108→}

   120→   120→  const player = await tx.player.findUnique({
   121→   121→    where: { id: playerId },
   122→   122→    select: { id: true, cash: true },

   126→   126→    return;

   128→   128→
   129→   129→  const allBusinesses = await tx.business.findMany({
   130→   130→    where: { playerId },
   131→   131→    select: { cash: true },
   132→   132→  });
   133→   133→  const totalBusinessCash = allBusinesses.reduce((sum, b) => sum + b.cash, 0);
   134→   134→
   135→   135→  const allInventory = await tx.inventory.findMany({
   136→   136→    where: { business: { playerId } },
   137→   137→    select: { quantity: true, purchasePrice: true },
   138→   138→  });
   139→   139→  const totalInventoryValue = allInventory.reduce((sum, inv) => sum + inv.quantity * inv.purchasePrice, 0);
   140→   140→
   141→   141→  const activeLoans = await tx.loan.findMany({
   142→   142→    where: { playerId, status: 'ACTIVE' },
   143→   143→    select: { remainingDebt: true },
   144→   144→  });
   145→   145→  const outstandingDebt = activeLoans.reduce((sum, loan) => sum + loan.remainingDebt, 0);
   146→   146→
   147→   147→  const netWorth = player.cash + totalBusinessCash + totalInventoryValue - outstandingDebt;
   148→   148→  await tx.player.update({ where: { id: playerId }, data: { netWorth } });
   149→   149→}

   153→   153→  const totalWeight = EVENT_TEMPLATES.reduce((sum, e) => sum + e.weight, 0);
   154→   154→  let random = Math.random() * totalWeight;
   155→   155→  let selected: EventTemplate | null = null;
   156→   156→  for (const event of EVENT_TEMPLATES) {
   157→   157→    random -= event.weight;
   158→   158→    if (random <= 0) {
   159→   159→      selected = event;
   160→   160→      break;
   161→   161→    }
   162→   162→  }

   164→   164→
   165→   165→  const now = new Date();

   168→   168→    data: {
   169→   169→      title: selected.title,
   170→   170→      description: selected.description,
   171→   171→      type: selected.type,
   172→   172→      effects: JSON.stringify(selected.effects),
   173→   173→      active: true,
   174→   174→      startsAt: now,
   175→   175→      endsAt: endAt,

   186→   186→export async function updateMarketPrices(): Promise<void> {
   187→   187→  const allProducts = Object.values(PRODUCTS).flat();

   190→   190→  // Get active events for price modifiers

   192→   192→  const categoryPriceMod: Record<string, number> = {};
   193→   193→  let allPriceMod = 0;
   194→   194→  for (const event of activeEvents) {
   195→   195→    const effects = JSON.parse(event.effects) as Record<string, number>;
   196→   196→    for (const [key, value] of Object.entries(effects)) {
   197→   197→      if (key === 'all_price' || key === 'all_imported') allPriceMod += value;
   198→   198→      if (key.endsWith('_price')) {
   199→   199→        const cat = key.replace('_price', '');
   200→   200→        categoryPriceMod[cat] = (categoryPriceMod[cat] || 0) + value;
   201→   201→      }
   202→   202→    }
   203→   203→  }
   204→   204→
   205→   205→  const updates: { productName: string; city: string; priceMultiplier: number; demandMultiplier: number }[] = [];
   206→   206→
   207→   207→  for (const city of cities) {
   208→   208→    for (const product of allProducts) {
   209→   209→      // Get current market price (for retention)
   210→   210→      const existing = await db.marketPrice.findUnique({
   211→   211→        where: { productName_city: { productName: product.name, city } },
   212→   212→      });
   213→   213→      const previousPriceMult = existing?.priceMultiplier ?? 1;
   214→   214→
   215→   215→      // Calculate event price modifier for this product
   216→   216→      const eventPriceMod = allPriceMod + (categoryPriceMod[product.category] || 0);
   217→   217→
   218→   218→      // Use Phase 1 retention-based update
   219→   219→      const { priceMultiplier, demandMultiplier } = calculateMarketPriceUpdate(
   220→   220→        previousPriceMult,
   221→   221→        eventPriceMod
   222→   222→      );
   223→   223→
   224→   224→      updates.push({ productName: product.name, city, priceMultiplier, demandMultiplier });
   225→   225→    }
   226→   226→  }
   227→   227→
   228→   228→  // Batch update in a transaction
   229→   229→  await db.$transaction(async (tx) => {
   230→   230→    for (const upd of updates) {
   231→   231→      await tx.marketPrice.upsert({
   232→   232→        where: { productName_city: { productName: upd.productName, city: upd.city } },
   233→   233→        create: upd,
   234→   234→        update: {
   235→   235→          priceMultiplier: upd.priceMultiplier,
   236→   236→          demandMultiplier: upd.demandMultiplier,
   237→   237→          updatedAt: new Date(),
   238→   238→        },
   239→   239→      });
   240→   240→    }
   241→   241→  });

   276→   276→export async function simulateBusinessTick(businessId: string): Promise<void> {
   277→   277→  const business = await db.business.findUnique({
   278→   278→    where: { id: businessId },
   279→   279→    include: {
   280→   280→      inventories: true,
   281→   281→      employees: true,
   282→   282→      player: { select: { id: true, cash: true, netWorth: true } },
   283→   283→    },
   284→   284→  });
   285→   285→
   286→   286→  if (!business) return;
   287→   287→
   288→   288→  const businessType = getBusinessType(business.type);
   289→   289→  const city = getCity(business.city);
   290→   290→  if (!businessType || !city) return;
   291→   291→
   292→   292→  const economyConfig = getBusinessEconomyConfig(business.type);
   293→   293→
   294→   294→  // Get active event effects (combined)
   295→   295→  const activeEvents = await db.gameEvent.findMany({
   296→   296→    where: { active: true },
   297→   297→    select: { effects: true },
   298→   298→  });
   299→   299→  const combinedEffects: Record<string, number> = {};
   300→   300→  for (const event of activeEvents) {
   301→   301→    const effects = JSON.parse(event.effects) as Record<string, number>;
   302→   302→    for (const [key, value] of Object.entries(effects)) {
   303→   303→      combinedEffects[key] = (combinedEffects[key] || 0) + value;
   304→   304→    }
   305→   305→  }
   306→   306→
   307→   307→  // Get market prices for this city
   308→   308→  const marketPrices = await db.marketPrice.findMany({
   309→   309→    where: { city: business.city },
   310→   310→  });
   311→   311→  const marketPriceMap: Record<string, { priceMultiplier: number; demandMultiplier: number }> = {};
   312→   312→  for (const mp of marketPrices) {
   313→   313→    marketPriceMap[mp.productName] = {
   314→   314→      priceMultiplier: mp.priceMultiplier,
   315→   315→      demandMultiplier: mp.demandMultiplier,
   316→   316→    };
   317→   317→  }
   318→   318→
   319→   319→  // ---- Step 1: Calculate potential customers ----
   320→   320→  const totalStock = business.inventories.reduce((sum, inv) => sum + inv.quantity, 0);
   321→   321→  // Max stock capacity: sum of product maxStock values for this business type
   322→   322→  const productDefs = PRODUCTS[business.type] || [];
   323→   323→  const maxStockCapacity = productDefs.reduce((sum, p) => sum + p.maxStock, 0);
   324→   324→  const avgEmployeeSkill = business.employees.length > 0
   325→   325→    ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   326→   326→    : 0;
   327→   327→
   328→   328→  const potentialCustomers = calculatePotentialCustomers({
   329→   329→    baseCustomers: businessType.baseCustomers,
   330→   330→    cityMultiplier: city.customerMultiplier,
   331→   331→    level: business.level,
   332→   332→    reputation: business.reputation,
   333→   333→    totalStock,
   334→   334→    maxStockCapacity,
   335→   335→    employeeCount: business.employees.length,
   336→   336→    avgEmployeeSkill,
   337→   337→    eventCustomerEffect: combinedEffects['all_customers'] || 0,
   338→   338→    businessDemandEffect: combinedEffects[`${businessType.id}_demand`] || 0,
   339→   339→    businessTypeId: business.type,
   340→   340→  });
   341→   341→
   342→   342→  // ---- Step 2: Product-level sales simulation ----
   343→   343→  let totalRevenue = 0;
   344→   344→  let totalCOGS = 0;
   345→   345→  const inventoryUpdates: { id: string; quantityDecrement: number }[] = [];
   346→   346→  const salesResults: ProductSalesResult[] = [];
   347→   347→
   348→   348→  for (const inventory of business.inventories) {
   349→   349→    if (inventory.quantity <= 0) continue;
   350→   350→
   351→   351→    const productDef = productDefs.find(p => p.name === inventory.productName);
   352→   352→    if (!productDef) continue;
   353→   353→
   354→   354→    // Product demand config
   355→   355→    const prodDemandConfig = getProductDemandConfig(inventory.productName);
   356→   356→
   357→   357→    // Market reference price = typical retail price (basePrice × (1 + suggestedMarkup) × priceMultiplier)
   358→   358→    // This represents what the MARKET typically charges, not the wholesale cost.
   359→   359→    // Players compare their price to market retail price, not to their own cost.
   360→   360→    const mp = marketPriceMap[inventory.productName];
   361→   361→    const typicalRetailPrice = productDef.basePrice * (1 + productDef.suggestedMarkup);
   362→   362→    const marketReferencePrice = Math.round(
   363→   363→      typicalRetailPrice * (mp?.priceMultiplier || 1)
   364→   364→    );
   365→   365→
   366→   366→    // Market demand multiplier from events
   367→   367→    const demandMult = mp?.demandMultiplier || 1;
   368→   368→
   369→   369→    // Event demand effect for this product's category
   370→   370→    const eventDemandEffect = combinedEffects[`${productDef.category}_demand`] || 0;
   371→   371→
   372→   372→    // Run the complete sales pipeline
   373→   373→    const salesResult = simulateProductSales({
   374→   374→      inventoryId: inventory.id,
   375→   375→      productName: inventory.productName,
   376→   376→      quantity: inventory.quantity,
   377→   377→      purchasePrice: inventory.purchasePrice,
   378→   378→      sellPrice: inventory.sellPrice,
   379→   379→      marketReferencePrice,
   380→   380→      baseDemand: prodDemandConfig.baseDemand,
   381→   381→      demandMultiplier: demandMult,
   382→   382→      eventDemandEffect,
   383→   383→      productPriceSensitivity: prodDemandConfig.priceSensitivity,
   384→   384→      productVolatility: prodDemandConfig.volatility,
   385→   385→      businessPriceSensitivity: economyConfig.priceSensitivity,
   386→   386→      potentialCustomers,
   387→   387→    });
   388→   388→
   389→   389→    if (salesResult.itemsSold > 0) {
   390→   390→      totalRevenue += salesResult.revenue;
   391→   391→      totalCOGS += salesResult.costOfGoodsSold;
   392→   392→      inventoryUpdates.push({
   393→   393→        id: inventory.id,
   394→   394→        quantityDecrement: salesResult.itemsSold,
   395→   395→      });
   396→   396→    }
   397→   397→
   398→   398→    salesResults.push(salesResult);
   399→   399→  }
   400→   400→
   401→   401→  // ---- Step 3: Calculate expenses ----
   402→   402→  const totalMonthlySalaries = business.employees.reduce((sum, emp) => sum + emp.salary, 0);
   403→   403→  const grossProfit = totalRevenue - totalCOGS;
   404→   404→
   405→   405→  const expenses = calculateBusinessExpenses({
   406→   406→    baseRent: businessType.rent,
   407→   407→    level: business.level,
   408→   408→    cityRentMultiplier: city.rentMultiplier,
   409→   409→    totalMonthlySalaries,
   410→   410→    businessLevel: business.level,
   411→   411→    businessTypeId: business.type,
   412→   412→    revenue: totalRevenue,
   413→   413→    grossProfit,
   414→   414→  });
   415→   415→
   416→   416→  // ---- Step 4: Calculate net profit ----
   417→   417→  const dailyProfit = calculateNetProfit(totalRevenue, totalCOGS, expenses.totalExpense);
   418→   418→
   419→   419→  // ---- Step 5: Calculate reputation change ----
   420→   420→  const outOfStockRatio = business.inventories.length > 0
   421→   421→    ? business.inventories.filter(inv => inv.quantity <= 0).length / business.inventories.length
   422→   422→    : 0;
   423→   423→  const managers = business.employees.filter(e => e.role === 'MANAGER');
   424→   424→  const cleaners = business.employees.filter(e => e.role === 'CLEANER');
   425→   425→
   426→   426→  const reputationChange = calculateReputationChange({
   427→   427→    dailyProfit,
   428→   428→    outOfStockRatio,
   429→   429→    managers: managers.map(m => ({ skill: m.skill })),
   430→   430→    cleaners: cleaners.map(c => ({ skill: c.skill })),
   431→   431→  });
   432→   432→
   433→   433→  const newReputation = Math.max(
   434→   434→    ECONOMY_CONFIG.minReputation,
   435→   435→    Math.min(ECONOMY_CONFIG.maxReputation, business.reputation + reputationChange)
   436→   436→  );
   437→   437→
   438→   438→  // ---- Step 6: Calculate business health score ----
   439→   439→  const healthResult = calculateBusinessHealth({
   440→   440→    dailyProfit,
   441→   441→    dailyRevenue: totalRevenue,
   442→   442→    businessCash: business.cash + dailyProfit,
   443→   443→    totalStock,
   444→   444→    maxStockCapacity,
   445→   445→    reputation: newReputation,
   446→   446→    businessTypeId: business.type,
   447→   447→  });
   448→   448→
   449→   449→  // ---- Step 7: Get game day for metrics ----
   450→   450→  const gameDayState = await db.gameState.findUnique({ where: { key: 'gameDay' } });
   451→   451→  const gameDay = parseInt(gameDayState?.value || '1', 10);
   452→   452→
   453→   453→  // ---- Step 8: Execute all updates in a single transaction ----
   454→   454→  await db.$transaction(async (tx) => {
   455→   455→    // Update inventory quantities
   456→   456→    for (const upd of inventoryUpdates) {
   457→   457→      await tx.inventory.update({
   458→   458→        where: { id: upd.id },
   459→   459→        data: { quantity: { decrement: upd.quantityDecrement } },
   460→   460→      });
   461→   461→    }
   462→   462→
   463→   463→    // Phase 1 FIX: Distribute business profit to player cash
   464→   464→    // Previously, profit accumulated in business.cash but was never accessible.
   465→   465→    // Now: business profit is transferred to the player each tick.
   466→   466→    const profitToDistribute = dailyProfit;

   533→   533→async function processLoanPayments(): Promise<void> {
   534→   534→  const activeLoans = await db.loan.findMany({
   535→   535→    where: { status: 'ACTIVE' },
   536→   536→  });
   537→   537→
   538→   538→  if (activeLoans.length === 0) return;
   539→   539→
   540→   540→  for (const loan of activeLoans) {
   541→   541→    const payment = loan.dailyPayment;
   542→   542→    const isPaidOff = loan.daysRemaining - 1 <= 0;
   543→   543→    const actualPayment = isPaidOff ? loan.remainingDebt : payment;
   544→   544→
   545→   545→    await db.$transaction(async (tx) => {
   546→   546→      const player = await tx.player.findUnique({
   547→   547→        where: { id: loan.playerId },
   548→   548→        select: { cash: true },
   549→   549→      });
   550→   550→      if (!player) {
   551→   551→        console.error(`[GameEngine] processLoanPayments: Player ${loan.playerId} not found for loan ${loan.id}`);
   552→   552→        return;
   553→   553→      }
   554→   554→
   555→   555→      const deductAmount = Math.min(actualPayment, Math.max(0, player.cash));
   556→   556→      const newRemainingDebt = Math.max(loan.remainingDebt - deductAmount, 0);
   557→   557→
   558→   558→      await tx.loan.update({
   559→   559→        where: { id: loan.id },
   560→   560→        data: {
   561→   561→          remainingDebt: newRemainingDebt,
   562→   562→          daysRemaining: Math.max(loan.daysRemaining - 1, 0),
   563→   563→          status: newRemainingDebt <= 0 ? 'PAID_OFF' : 'ACTIVE',
   564→   564→        },
   565→   565→      });
   566→   566→
   567→   567→      await tx.player.update({
   568→   568→        where: { id: loan.playerId },
   569→   569→        data: { cash: { decrement: deductAmount } },
   570→   570→      });
   571→   571→
   572→   572→      await tx.gameLog.create({
   573→   573→        data: {
   574→   574→          playerId: loan.playerId,
   575→   575→          type: newRemainingDebt <= 0 ? 'LOAN_PAID' : 'LOAN_PAYMENT',
   576→   576→          message: newRemainingDebt <= 0
   577→   577→            ? `Loan of ৳${loan.amount.toLocaleString()} fully repaid! Final deduction: ৳${Math.round(deductAmount).toLocaleString()}`
   578→   578→            : `Loan payment: ৳${Math.round(deductAmount).toLocaleString()} deducted. Remaining debt: ৳${Math.round(newRemainingDebt).toLocaleString()} (${loan.daysRemaining - 1} days left)`,
   579→   579→          amount: -deductAmount,
   580→   580→        },
   581→   581→      });
   582→   582→    });
   583→   583→  }
   584→   584→}

   645→   645→    } catch (err) {

   649→   649→  }

   666→   666→  }
   667→   667→

   672→   672→    await simulateAIPlayersTick(currentGameDay);
   673→   673→  } catch (err) {

   675→   675→  }
   676→   676→

   681→   681→    } catch (err) {
   682→   682→      console.error('[GameEngine] Failed to generate news:', err);
   683→   683→    }
   684→   684→  }
   685→   685→
   686→   686→  // Log tick completion
   687→   687→  await db.gameLog.create({
   688→   688→    data: {
   689→   689→      type: 'GAME_TICK_COMPLETED',
   690→   690→      message: `Game tick completed at ${new Date().toISOString()}`,
   691→   691→    },
   692→   692→  });
