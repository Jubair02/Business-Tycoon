    66→const STALE_TICK_TIMEOUT_MS = 2 * 60 * 1000; // 2 minutes

    68→// ---- Tick Concurrency Lock ----
    69→
    70→/**
    71→ * Atomically attempt to acquire the tick lock using the GameState key-value store.
    72→ * Returns true if the lock was acquired, false if it was already held.
    73→ *
    74→ * Stale lock recovery: if the lock has been held for longer than
    75→ * STALE_TICK_TIMEOUT_MS, the lock is stolen and a warning is logged.
    76→ */
    77→export async function acquireTickLock(): Promise<boolean> {
    78→  const now = new Date();
    79→
    80→  return db.$transaction(async (tx) => {
    81→    // Read current lock state
    82→    const lockState = await tx.gameState.findUnique({ where: { key: 'tickInProgress' } });
    83→    const startedAtState = await tx.gameState.findUnique({ where: { key: 'tickStartedAt' } });
    84→
    85→    const isLocked = lockState?.value === 'true';
    86→    const startedAt = startedAtState?.value ? new Date(startedAtState.value) : null;
    87→
    88→    if (isLocked && startedAt) {
    89→      const elapsed = now.getTime() - startedAt.getTime();
    90→      if (elapsed < STALE_TICK_TIMEOUT_MS) {
    91→        return false; // Lock is legitimately held
    92→      }
    93→      console.warn(`[GameEngine] Stale tick lock detected (held for ${Math.round(elapsed / 1000)}s). Stealing lock.`);
    94→    } else if (isLocked) {
    95→      console.warn('[GameEngine] Tick lock held with no timestamp. Stealing lock.');
    96→    }
    97→
    98→    // Acquire the lock
    99→    const nowISO = now.toISOString();
   100→    await tx.gameState.upsert({
   101→      where: { key: 'tickInProgress' },
   102→      create: { key: 'tickInProgress', value: 'true' },
   103→      update: { value: 'true' },
   104→    });
   105→    await tx.gameState.upsert({
   106→      where: { key: 'tickStartedAt' },
   107→      create: { key: 'tickStartedAt', value: nowISO },
   108→      update: { value: nowISO },
   109→    });
   110→
   111→    // Increment tick version
   112→    const versionState = await tx.gameState.findUnique({ where: { key: 'tickVersion' } });
   113→    const currentVersion = parseInt(versionState?.value || '0', 10);
   114→    await tx.gameState.upsert({
   115→      where: { key: 'tickVersion' },
   116→      create: { key: 'tickVersion', value: '1' },
   117→      update: { value: String(currentVersion + 1) },
   118→    });
   119→
   120→    return true;
   121→  });
   122→}

   132→// ---- Net Worth Recalculation ----
   133→async function recalculateNetWorth(tx: PrismaTx, playerId: string): Promise<void> {
   134→  const player = await tx.player.findUnique({
   135→    where: { id: playerId },
   136→    select: { id: true, cash: true },
   137→  });
   138→  if (!player) {
   139→    console.error(`[GameEngine] recalculateNetWorth: Player ${playerId} not found`);
   140→    return;
   141→  }
   142→
   143→  const allBusinesses = await tx.business.findMany({
   144→    where: { playerId },
   145→    select: { cash: true },
   146→  });
   147→  const totalBusinessCash = allBusinesses.reduce((sum, b) => sum + b.cash, 0);
   148→
   149→  const allInventory = await tx.inventory.findMany({
   150→    where: { business: { playerId } },
   151→    select: { quantity: true, purchasePrice: true },
   152→  });
   153→  const totalInventoryValue = allInventory.reduce((sum, inv) => sum + inv.quantity * inv.purchasePrice, 0);
   154→
   155→  const activeLoans = await tx.loan.findMany({
   156→    where: { playerId, status: 'ACTIVE' },
   157→    select: { remainingDebt: true },
   158→  });
   159→  const outstandingDebt = activeLoans.reduce((sum, loan) => sum + loan.remainingDebt, 0);
   160→
   161→  const netWorth = player.cash + totalBusinessCash + totalInventoryValue - outstandingDebt;
   162→  await tx.player.update({ where: { id: playerId }, data: { netWorth } });
   163→}

   165→// ---- Event Generation ----
   166→export async function generateEvent(): Promise<void> {
   167→  const totalWeight = EVENT_TEMPLATES.reduce((sum, e) => sum + e.weight, 0);
   168→  let random = Math.random() * totalWeight;
   169→  let selected: EventTemplate | null = null;
   170→  for (const event of EVENT_TEMPLATES) {
   171→    random -= event.weight;
   172→    if (random <= 0) {
   173→      selected = event;
   174→      break;
   175→    }
   176→  }
   177→  if (!selected) selected = EVENT_TEMPLATES[0];
   178→
   179→  const now = new Date();
   180→  const endAt = new Date(now.getTime() + selected.durationDays * 60 * 1000);
   181→  await db.gameEvent.create({
   182→    data: {
   183→      title: selected.title,
   184→      description: selected.description,
   185→      type: selected.type,
   186→      effects: JSON.stringify(selected.effects),
   187→      active: true,
   188→      startsAt: now,
   189→      endsAt: endAt,
   190→    },
   191→  });
   192→}

   200→export async function updateMarketPrices(): Promise<void> {
   201→  const allProducts = Object.values(PRODUCTS).flat();
   202→  const cities = CITIES.map(c => c.id);
   203→
   204→  // Get active events for price modifiers
   205→  const activeEvents = await db.gameEvent.findMany({ where: { active: true } });
   206→  const categoryPriceMod: Record<string, number> = {};
   207→  let allPriceMod = 0;
   208→  for (const event of activeEvents) {
   209→    const effects = JSON.parse(event.effects) as Record<string, number>;
   210→    for (const [key, value] of Object.entries(effects)) {
   211→      if (key === 'all_price' || key === 'all_imported') allPriceMod += value;
   212→      if (key.endsWith('_price')) {
   213→        const cat = key.replace('_price', '');
   214→        categoryPriceMod[cat] = (categoryPriceMod[cat] || 0) + value;
   215→      }
   216→    }
   217→  }
   218→
   219→  const updates: { productName: string; city: string; priceMultiplier: number; demandMultiplier: number }[] = [];
   220→
   221→  for (const city of cities) {
   222→    for (const product of allProducts) {
   223→      // Get current market price (for retention)
   224→      const existing = await db.marketPrice.findUnique({
   225→        where: { productName_city: { productName: product.name, city } },
   226→      });
   227→      const previousPriceMult = existing?.priceMultiplier ?? 1;
   228→
   229→      // Calculate event price modifier for this product
   230→      const eventPriceMod = allPriceMod + (categoryPriceMod[product.category] || 0);
   231→
   232→      // Use Phase 1 retention-based update
   233→      const { priceMultiplier, demandMultiplier } = calculateMarketPriceUpdate(
   234→        previousPriceMult,
   235→        eventPriceMod
   236→      );
   237→
   238→      updates.push({ productName: product.name, city, priceMultiplier, demandMultiplier });
   239→    }
   240→  }
   241→
   242→  // Batch update in a transaction
   243→  await db.$transaction(async (tx) => {
   244→    for (const upd of updates) {
   245→      await tx.marketPrice.upsert({
   246→        where: { productName_city: { productName: upd.productName, city: upd.city } },
   247→        create: upd,
   248→        update: {
   249→          priceMultiplier: upd.priceMultiplier,
   250→          demandMultiplier: upd.demandMultiplier,
   251→          updatedAt: new Date(),
   252→        },
   253→      });
   254→    }
   255→  });
   256→}

   290→export async function simulateBusinessTick(businessId: string): Promise<void> {
   291→  const business = await db.business.findUnique({
   292→    where: { id: businessId },
   293→    include: {
   294→      inventories: true,
   295→      employees: true,
   296→      player: { select: { id: true, cash: true, netWorth: true } },
   297→    },
   298→  });
   299→
   300→  if (!business) return;
   301→
   302→  const businessType = getBusinessType(business.type);
   303→  const city = getCity(business.city);
   304→  if (!businessType || !city) return;
   305→
   306→  const economyConfig = getBusinessEconomyConfig(business.type);
   307→
   308→  // Get active event effects (combined)
   309→  const activeEvents = await db.gameEvent.findMany({
   310→    where: { active: true },
   311→    select: { effects: true },
   312→  });
   313→  const combinedEffects: Record<string, number> = {};
   314→  for (const event of activeEvents) {
   315→    const effects = JSON.parse(event.effects) as Record<string, number>;
   316→    for (const [key, value] of Object.entries(effects)) {
   317→      combinedEffects[key] = (combinedEffects[key] || 0) + value;
   318→    }
   319→  }
   320→
   321→  // Get market prices for this city
   322→  const marketPrices = await db.marketPrice.findMany({
   323→    where: { city: business.city },
   324→  });
   325→  const marketPriceMap: Record<string, { priceMultiplier: number; demandMultiplier: number }> = {};
   326→  for (const mp of marketPrices) {
   327→    marketPriceMap[mp.productName] = {
   328→      priceMultiplier: mp.priceMultiplier,
   329→      demandMultiplier: mp.demandMultiplier,
   330→    };
   331→  }
   332→
   333→  // ---- Step 1: Calculate potential customers ----
   334→  const totalStock = business.inventories.reduce((sum, inv) => sum + inv.quantity, 0);
   335→  // Max stock capacity: sum of product maxStock values for this business type
   336→  const productDefs = PRODUCTS[business.type] || [];
   337→  const maxStockCapacity = productDefs.reduce((sum, p) => sum + p.maxStock, 0);
   338→  const avgEmployeeSkill = business.employees.length > 0
   339→    ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   340→    : 0;
   341→
   342→  const potentialCustomers = calculatePotentialCustomers({
   343→    baseCustomers: businessType.baseCustomers,
   344→    cityMultiplier: city.customerMultiplier,
   345→    level: business.level,
   346→    reputation: business.reputation,
   347→    totalStock,
   348→    maxStockCapacity,
   349→    employeeCount: business.employees.length,
   350→    avgEmployeeSkill,
   351→    eventCustomerEffect: combinedEffects['all_customers'] || 0,
   352→    businessDemandEffect: combinedEffects[`${businessType.id}_demand`] || 0,
   353→    businessTypeId: business.type,
   354→  });
   355→
   356→  // ---- Step 2: Product-level sales simulation ----
   357→  let totalRevenue = 0;
   358→  let totalCOGS = 0;
   359→  const inventoryUpdates: { id: string; quantityDecrement: number }[] = [];
   360→  const salesResults: ProductSalesResult[] = [];
   361→
   362→  for (const inventory of business.inventories) {
   363→    if (inventory.quantity <= 0) continue;
   364→
   365→    const productDef = productDefs.find(p => p.name === inventory.productName);
   366→    if (!productDef) continue;
   367→
   368→    // Product demand config
   369→    const prodDemandConfig = getProductDemandConfig(inventory.productName);
   370→
   371→    // Market reference price = typical retail price (basePrice × (1 + suggestedMarkup) × priceMultiplier)
   372→    // This represents what the MARKET typically charges, not the wholesale cost.
   373→    // Players compare their price to market retail price, not to their own cost.
   374→    const mp = marketPriceMap[inventory.productName];
   375→    const typicalRetailPrice = productDef.basePrice * (1 + productDef.suggestedMarkup);
   376→    const marketReferencePrice = Math.round(
   377→      typicalRetailPrice * (mp?.priceMultiplier || 1)
   378→    );
   379→
   380→    // Market demand multiplier from events
   381→    const demandMult = mp?.demandMultiplier || 1;
   382→
   383→    // Event demand effect for this product's category
   384→    const eventDemandEffect = combinedEffects[`${productDef.category}_demand`] || 0;
   385→
   386→    // Run the complete sales pipeline
   387→    const salesResult = simulateProductSales({
   388→      inventoryId: inventory.id,
   389→      productName: inventory.productName,
   390→      quantity: inventory.quantity,
   391→      purchasePrice: inventory.purchasePrice,
   392→      sellPrice: inventory.sellPrice,
   393→      marketReferencePrice,
   394→      baseDemand: prodDemandConfig.baseDemand,
   395→      demandMultiplier: demandMult,
   396→      eventDemandEffect,
   397→      productPriceSensitivity: prodDemandConfig.priceSensitivity,
   398→      productVolatility: prodDemandConfig.volatility,
   399→      businessPriceSensitivity: economyConfig.priceSensitivity,
   400→      potentialCustomers,
   401→    });
   402→
   403→    if (salesResult.itemsSold > 0) {
   404→      totalRevenue += salesResult.revenue;
   405→      totalCOGS += salesResult.costOfGoodsSold;
   406→      inventoryUpdates.push({
   407→        id: inventory.id,
   408→        quantityDecrement: salesResult.itemsSold,
   409→      });
   410→    }
   411→
   412→    salesResults.push(salesResult);
   413→  }
   414→
   415→  // ---- Step 3: Calculate expenses ----
   416→  const totalMonthlySalaries = business.employees.reduce((sum, emp) => sum + emp.salary, 0);
   417→  const grossProfit = totalRevenue - totalCOGS;
   418→
   419→  const expenses = calculateBusinessExpenses({
   420→    baseRent: businessType.rent,
   421→    level: business.level,
   422→    cityRentMultiplier: city.rentMultiplier,
   423→    totalMonthlySalaries,
   424→    businessLevel: business.level,
   425→    businessTypeId: business.type,
   426→    revenue: totalRevenue,
   427→    grossProfit,
   428→  });
   429→
   430→  // ---- Step 4: Calculate net profit ----
   431→  const dailyProfit = calculateNetProfit(totalRevenue, totalCOGS, expenses.totalExpense);
   432→
   433→  // ---- Step 5: Calculate reputation change ----
   434→  const outOfStockRatio = business.inventories.length > 0
   435→    ? business.inventories.filter(inv => inv.quantity <= 0).length / business.inventories.length
   436→    : 0;
   437→  const managers = business.employees.filter(e => e.role === 'MANAGER');
   438→  const cleaners = business.employees.filter(e => e.role === 'CLEANER');
   439→
   440→  const reputationChange = calculateReputationChange({
   441→    dailyProfit,
   442→    outOfStockRatio,
   443→    managers: managers.map(m => ({ skill: m.skill })),
   444→    cleaners: cleaners.map(c => ({ skill: c.skill })),
   445→  });
   446→
   447→  const newReputation = Math.max(
   448→    ECONOMY_CONFIG.minReputation,
   449→    Math.min(ECONOMY_CONFIG.maxReputation, business.reputation + reputationChange)
   450→  );
   451→
   452→  // ---- Step 6: Calculate business health score ----
   453→  const healthResult = calculateBusinessHealth({
   454→    dailyProfit,
   455→    dailyRevenue: totalRevenue,
   456→    businessCash: business.cash + dailyProfit,
   457→    totalStock,
   458→    maxStockCapacity,
   459→    reputation: newReputation,
   460→    businessTypeId: business.type,
   461→  });
   462→
   463→  // ---- Step 6.5: Customer Experience (Phase 3) ----
   464→  // Calculate price competitiveness across all products
   465→  const cxPriceCompetitiveness = calculatePriceCompetitiveness(
   466→    business.inventories.map(inv => ({ productName: inv.productName, sellPrice: inv.sellPrice })),
   467→    productDefs.map(p => ({ name: p.name, basePrice: p.basePrice, suggestedMarkup: p.suggestedMarkup })),
   468→    marketPriceMap,
   469→  );
   470→
   471→  // Calculate service quality from employees
   472→  const idealEmployeeCount = 2 + business.level; // Base 2 + 1 per level
   473→  const cxServiceQuality = calculateServiceQuality(
   474→    business.employees.length,
   475→    avgEmployeeSkill,
   476→    idealEmployeeCount,
   477→  );
   478→
   479→  // Calculate product quality from margin (higher margin = better products)
   480→  const cxProductQuality = totalRevenue > 0
   481→    ? Math.min(1, Math.max(0, (totalRevenue - totalCOGS) / totalRevenue / 0.3))
   482→    : 0.5;
   483→
   484→  // Calculate satisfaction
   485→  const cxSatisfaction = calculateSatisfaction({
   486→    priceCompetitiveness: cxPriceCompetitiveness,
   487→    stockAvailability: maxStockCapacity > 0 ? totalStock / maxStockCapacity : 0,
   488→    serviceQuality: cxServiceQuality,
   489→    productQuality: cxProductQuality,
   490→    atmosphere: newReputation / 100,
   491→    currentSatisfaction: business.satisfactionScore,
   492→  });
   493→
   494→  // Determine if this was a positive experience
   495→  const wasPositiveExperience = cxSatisfaction.overall >= 50 && dailyProfit >= 0;
   496→
   497→  // Calculate loyalty
   498→  const cxLoyalty = calculateLoyalty({
   499→    currentLoyaltyScore: business.loyaltyScore,
   500→    currentRepeatRate: business.repeatCustomerRate,
   501→    wasPositiveExperience,
   502→    totalCustomers: potentialCustomers,
   503→    numberOfReviews: business.totalReviews,
   504→    averageRating: business.avgReviewRating,
   505→    loyaltyMultiplier: getLoyaltyTier(business.loyaltyScore).multiplier,
   506→  });
   507→
   508→  // Calculate CX demand modifier
   509→  const cxDemandModifier = calculateCXDemandModifier(
   510→    cxSatisfaction.overall,
   511→    cxLoyalty.repeatCustomerRate,
   512→    cxLoyalty.tierMultiplier,
   513→  );
   514→
   515→  // ---- Step 7: Get game day for metrics ----
   516→  const gameDayState = await db.gameState.findUnique({ where: { key: 'gameDay' } });
   517→  const gameDay = parseInt(gameDayState?.value || '1', 10);
   518→
   519→  // ---- Step 8: Execute all updates in a single transaction ----
   520→  await db.$transaction(async (tx) => {
   521→    // Update inventory quantities
   522→    for (const upd of inventoryUpdates) {
   523→      await tx.inventory.update({
   524→        where: { id: upd.id },
   525→        data: { quantity: { decrement: upd.quantityDecrement } },
   526→      });
   527→    }
   528→
   529→    // Phase 1 FIX: Distribute business profit to player cash
   530→    // Previously, profit accumulated in business.cash but was never accessible.
   531→    // Now: business profit is transferred to the player each tick.
   532→    const profitToDistribute = dailyProfit;
   533→
   534→    // Update business (including Phase 3 CX fields)
   535→    await tx.business.update({
   536→      where: { id: business.id },
   537→      data: {
   538→        dailyRevenue: roundTaka(totalRevenue),
   539→        dailyExpense: roundTaka(expenses.totalExpense),
   540→        dailyProfit: roundTaka(dailyProfit),
   541→        dailyCOGS: roundTaka(totalCOGS),
   542→        dailyCustomers: potentialCustomers,
   543→        cash: business.cash + roundTaka(dailyProfit),
   544→        totalRevenue: business.totalRevenue + roundTaka(totalRevenue),
   545→        totalProfit: business.totalProfit + roundTaka(dailyProfit),
   546→        reputation: newReputation,
   547→        healthScore: healthResult.score,
   548→        // Phase 3: CX fields
   549→        satisfactionScore: cxSatisfaction.overall,
   550→        loyaltyScore: cxLoyalty.loyaltyScore,
   551→        repeatCustomerRate: cxLoyalty.repeatCustomerRate,
   552→      },
   553→    });
   554→
   555→    // Distribute profit to player cash
   556→    await tx.player.update({
   557→      where: { id: business.playerId },
   558→      data: { cash: { increment: roundTaka(profitToDistribute) } },
   559→    });
   560→

   665→// ---- Loan Payment Processing ----
   666→
   667→async function processLoanPayments(): Promise<void> {
   668→  const activeLoans = await db.loan.findMany({
   669→    where: { status: 'ACTIVE' },
   670→  });
   671→
   672→  if (activeLoans.length === 0) return;
   673→
   674→  for (const loan of activeLoans) {
   675→    const payment = loan.dailyPayment;
   676→    const isPaidOff = loan.daysRemaining - 1 <= 0;
   677→    const actualPayment = isPaidOff ? loan.remainingDebt : payment;
   678→
   679→    await db.$transaction(async (tx) => {
   680→      const player = await tx.player.findUnique({
   681→        where: { id: loan.playerId },
   682→        select: { cash: true },
   683→      });
   684→      if (!player) {
   685→        console.error(`[GameEngine] processLoanPayments: Player ${loan.playerId} not found for loan ${loan.id}`);
   686→        return;
   687→      }
   688→
   689→      const deductAmount = Math.min(actualPayment, Math.max(0, player.cash));
   690→      const newRemainingDebt = Math.max(loan.remainingDebt - deductAmount, 0);
   691→
   692→      await tx.loan.update({
   693→        where: { id: loan.id },
   694→        data: {
   695→          remainingDebt: newRemainingDebt,
   696→          daysRemaining: Math.max(loan.daysRemaining - 1, 0),
   697→          status: newRemainingDebt <= 0 ? 'PAID_OFF' : 'ACTIVE',
   698→        },
   699→      });
   700→
   701→      await tx.player.update({
   702→        where: { id: loan.playerId },
   703→        data: { cash: { decrement: deductAmount } },
   704→      });
   705→
   706→      await tx.gameLog.create({
   707→        data: {
   708→          playerId: loan.playerId,
   709→          type: newRemainingDebt <= 0 ? 'LOAN_PAID' : 'LOAN_PAYMENT',
   710→          message: newRemainingDebt <= 0
   711→            ? `Loan of ৳${loan.amount.toLocaleString()} fully repaid! Final deduction: ৳${Math.round(deductAmount).toLocaleString()}`
   712→            : `Loan payment: ৳${Math.round(deductAmount).toLocaleString()} deducted. Remaining debt: ৳${Math.round(newRemainingDebt).toLocaleString()} (${loan.daysRemaining - 1} days left)`,
   713→          amount: -deductAmount,
   714→        },
   715→      });
   716→    });
   717→  }
   718→}

   722→export async function gameTick(): Promise<void> {
   723→  // NOTE: The tick lock is managed by the API route handler.
   724→  // gameTick() is the pure simulation function and does NOT acquire/release the lock.
   725→
   726→  // Log tick start
   727→  await db.gameLog.create({
   728→    data: {
   729→      type: 'GAME_TICK_STARTED',
   730→      message: `Game tick started at ${new Date().toISOString()}`,
   731→    },
   732→  });
   733→
   734→  // 0. Game state housekeeping
   735→  const tickState = await db.gameState.findUnique({ where: { key: 'tickCount' } });
   736→  const tickNum = parseInt(tickState?.value || '0');
   737→
   738→  await db.$transaction(async (tx) => {
   739→    // Expire old events
   740→    await tx.gameEvent.updateMany({
   741→      where: { active: true, endsAt: { lt: new Date() } },
   742→      data: { active: false },
   743→    });
   744→
   745→    // Update game state
   746→    const now = new Date().toISOString();
   747→    await tx.gameState.upsert({
   748→      where: { key: 'lastTick' },
   749→      create: { key: 'lastTick', value: now },
   750→      update: { value: now },
   751→    });
   752→    await tx.gameState.upsert({
   753→      where: { key: 'tickCount' },
   754→      create: { key: 'tickCount', value: '1' },
   755→      update: { value: String(tickNum + 1) },
   756→    });
   757→    const dayState = await tx.gameState.findUnique({ where: { key: 'gameDay' } });
   758→    await tx.gameState.upsert({
   759→      where: { key: 'gameDay' },
   760→      create: { key: 'gameDay', value: '1' },
   761→      update: { value: String(parseInt(dayState?.value || '0') + 1) },
   762→    });
   763→  });
   764→
   765→  // Generate new events (20% chance)
   766→  if (Math.random() < 0.2) {
   767→    try {
   768→      await generateEvent();
   769→    } catch (err) {
   770→      console.error('[GameEngine] Failed to generate event:', err);
   771→    }
   772→  }
   773→
   774→  // Update market prices (every 3 ticks)
   775→  // Phase 1: This now uses retention-based updates with event effects integrated
   776→  if (tickNum % 3 === 0) {
   777→    try {
   778→      await updateMarketPrices();
   779→    } catch (err) {
   780→      console.error('[GameEngine] Failed to update market prices:', err);
   781→    }
   782→    // applyEventEffects() is now a no-op — events handled in updateMarketPrices()
   783→  }
   784→
   785→  // 1. Simulate all businesses
   786→  const businesses = await db.business.findMany({ select: { id: true } });
   787→  for (const business of businesses) {
   788→    try {
   789→      await simulateBusinessTick(business.id);
   790→    } catch (err) {
   791→      console.error(`[GameEngine] Failed to simulate business ${business.id}:`, err);
   792→    }
   793→  }
   794→
   795→  // 2. Process loan payments
   796→  try {
   797→    await processLoanPayments();
   798→  } catch (err) {
   799→    console.error('[GameEngine] Failed to process loan payments:', err);
   800→  }
   801→
   802→  // 3. Simulate AI player activity (Phase 2: Real AI decisions)
   803→  try {
   804→    const gameDayState2 = await db.gameState.findUnique({ where: { key: 'gameDay' } });
   805→    const currentGameDay = parseInt(gameDayState2?.value || '1', 10);
   806→    await simulateAIPlayersTick(currentGameDay);
   807→  } catch (err) {
   808→    console.error('[GameEngine] Failed to simulate AI tick:', err);
   809→  }
   810→
   811→  // 4. Generate news (50% chance)
   812→  if (Math.random() < 0.5) {
   813→    try {
   814→      await generateNews();
   815→    } catch (err) {
   816→      console.error('[GameEngine] Failed to generate news:', err);
   817→    }
   818→  }
   819→
   820→  // Log tick completion
   821→  await db.gameLog.create({
   822→    data: {
   823→      type: 'GAME_TICK_COMPLETED',
   824→      message: `Game tick completed at ${new Date().toISOString()}`,
   825→    },
   826→  });
   827→}
