    78→    78→// ---- Tick Concurrency Lock Constants ----
    79→    79→const STALE_TICK_TIMEOUT_MS = 2 * 60 * 1000; // 2 minutes
    80→    80→
    81→    81→// ---- Tick Concurrency Lock ----
    82→    82→
    83→    83→/**
    84→    84→ * Atomically attempt to acquire the tick lock using the GameState key-value store.
    85→    85→ * Returns true if the lock was acquired, false if it was already held.
    86→    86→ *
    87→    87→ * Stale lock recovery: if the lock has been held for longer than
    88→    88→ * STALE_TICK_TIMEOUT_MS, the lock is stolen and a warning is logged.
    89→    89→ */
    90→    90→export async function acquireTickLock(): Promise<boolean> {
    91→    91→  const now = new Date();
    92→    92→
    93→    93→  return db.$transaction(async (tx) => {
    94→    94→    // Read current lock state
    95→    95→    const lockState = await tx.gameState.findUnique({ where: { key: 'tickInProgress' } });
    96→    96→    const startedAtState = await tx.gameState.findUnique({ where: { key: 'tickStartedAt' } });
    97→    97→
    98→    98→    const isLocked = lockState?.value === 'true';
    99→    99→    const startedAt = startedAtState?.value ? new Date(startedAtState.value) : null;
   100→   100→
   101→   101→    if (isLocked && startedAt) {
   102→   102→      const elapsed = now.getTime() - startedAt.getTime();
   103→   103→      if (elapsed < STALE_TICK_TIMEOUT_MS) {
   104→   104→        return false; // Lock is legitimately held
   105→   105→      }
   106→   106→      console.warn(`[GameEngine] Stale tick lock detected (held for ${Math.round(elapsed / 1000)}s). Stealing lock.`);
   107→   107→    } else if (isLocked) {
   108→   108→      console.warn('[GameEngine] Tick lock held with no timestamp. Stealing lock.');
   109→   109→    }
   110→   110→
   111→   111→    // Acquire the lock
   112→   112→    const nowISO = now.toISOString();
   113→   113→    await tx.gameState.upsert({
   114→   114→      where: { key: 'tickInProgress' },
   115→   115→      create: { key: 'tickInProgress', value: 'true' },
   116→   116→      update: { value: 'true' },
   117→   117→    });
   118→   118→    await tx.gameState.upsert({
   119→   119→      where: { key: 'tickStartedAt' },
   120→   120→      create: { key: 'tickStartedAt', value: nowISO },
   121→   121→      update: { value: nowISO },
   122→   122→    });
   123→   123→
   124→   124→    // Increment tick version
   125→   125→    const versionState = await tx.gameState.findUnique({ where: { key: 'tickVersion' } });
   126→   126→    const currentVersion = parseInt(versionState?.value || '0', 10);
   127→   127→    await tx.gameState.upsert({
   128→   128→      where: { key: 'tickVersion' },
   129→   129→      create: { key: 'tickVersion', value: '1' },
   130→   130→      update: { value: String(currentVersion + 1) },
   131→   131→    });
   132→   132→
   133→   133→    return true;
   134→   134→  });
   135→   135→}

   146→   146→async function recalculateNetWorth(tx: PrismaTx, playerId: string): Promise<void> {
   147→   147→  const player = await tx.player.findUnique({
   148→   148→    where: { id: playerId },
   149→   149→    select: { id: true, cash: true },
   150→   150→  });
   151→   151→  if (!player) {
   152→   152→    console.error(`[GameEngine] recalculateNetWorth: Player ${playerId} not found`);
   153→   153→    return;
   154→   154→  }
   155→   155→
   156→   156→  const allBusinesses = await tx.business.findMany({
   157→   157→    where: { playerId },
   158→   158→    select: { cash: true },
   159→   159→  });
   160→   160→  const totalBusinessCash = allBusinesses.reduce((sum, b) => sum + b.cash, 0);
   161→   161→
   162→   162→  const allInventory = await tx.inventory.findMany({
   163→   163→    where: { business: { playerId } },
   164→   164→    select: { quantity: true, purchasePrice: true },
   165→   165→  });
   166→   166→  const totalInventoryValue = allInventory.reduce((sum, inv) => sum + inv.quantity * inv.purchasePrice, 0);
   167→   167→
   168→   168→  const activeLoans = await tx.loan.findMany({
   169→   169→    where: { playerId, status: 'ACTIVE' },
   170→   170→    select: { remainingDebt: true },
   171→   171→  });
   172→   172→  const outstandingDebt = activeLoans.reduce((sum, loan) => sum + loan.remainingDebt, 0);
   173→   173→
   174→   174→  const netWorth = player.cash + totalBusinessCash + totalInventoryValue - outstandingDebt;
   175→   175→  await tx.player.update({ where: { id: playerId }, data: { netWorth } });
   176→   176→}

   179→   179→export async function generateEvent(): Promise<void> {
   180→   180→  const totalWeight = EVENT_TEMPLATES.reduce((sum, e) => sum + e.weight, 0);
   181→   181→  let random = Math.random() * totalWeight;
   182→   182→  let selected: EventTemplate | null = null;
   183→   183→  for (const event of EVENT_TEMPLATES) {
   184→   184→    random -= event.weight;
   185→   185→    if (random <= 0) {
   186→   186→      selected = event;
   187→   187→      break;
   188→   188→    }
   189→   189→  }
   190→   190→  if (!selected) selected = EVENT_TEMPLATES[0];
   191→   191→
   192→   192→  const now = new Date();
   193→   193→  const endAt = new Date(now.getTime() + selected.durationDays * 60 * 1000);
   194→   194→  await db.gameEvent.create({
   195→   195→    data: {
   196→   196→      title: selected.title,
   197→   197→      description: selected.description,
   198→   198→      type: selected.type,
   199→   199→      effects: JSON.stringify(selected.effects),
   200→   200→      active: true,
   201→   201→      startsAt: now,
   202→   202→      endsAt: endAt,
   203→   203→    },
   204→   204→  });
   205→   205→}

   213→   213→export async function updateMarketPrices(): Promise<void> {
   214→   214→  const allProducts = Object.values(PRODUCTS).flat();
   215→   215→  const cities = CITIES.map(c => c.id);
   216→   216→
   217→   217→  // Get active events for price modifiers
   218→   218→  const activeEvents = await db.gameEvent.findMany({ where: { active: true } });
   219→   219→  const categoryPriceMod: Record<string, number> = {};
   220→   220→  let allPriceMod = 0;
   221→   221→  for (const event of activeEvents) {
   222→   222→    const effects = JSON.parse(event.effects) as Record<string, number>;
   223→   223→    for (const [key, value] of Object.entries(effects)) {
   224→   224→      if (key === 'all_price' || key === 'all_imported') allPriceMod += value;
   225→   225→      if (key.endsWith('_price')) {
   226→   226→        const cat = key.replace('_price', '');
   227→   227→        categoryPriceMod[cat] = (categoryPriceMod[cat] || 0) + value;
   228→   228→      }
   229→   229→    }
   230→   230→  }
   231→   231→
   232→   232→  const updates: { productName: string; city: string; priceMultiplier: number; demandMultiplier: number }[] = [];
   233→   233→
   234→   234→  for (const city of cities) {
   235→   235→    for (const product of allProducts) {
   236→   236→      // Get current market price (for retention)
   237→   237→      const existing = await db.marketPrice.findUnique({
   238→   238→        where: { productName_city: { productName: product.name, city } },
   239→   239→      });
   240→   240→      const previousPriceMult = existing?.priceMultiplier ?? 1;
   241→   241→
   242→   242→      // Calculate event price modifier for this product
   243→   243→      const eventPriceMod = allPriceMod + (categoryPriceMod[product.category] || 0);
   244→   244→
   245→   245→      // Use Phase 1 retention-based update
   246→   246→      const { priceMultiplier, demandMultiplier } = calculateMarketPriceUpdate(
   247→   247→        previousPriceMult,
   248→   248→        eventPriceMod
   249→   249→      );
   250→   250→
   251→   251→      updates.push({ productName: product.name, city, priceMultiplier, demandMultiplier });
   252→   252→    }
   253→   253→  }
   254→   254→
   255→   255→  // Batch update in a transaction
   256→   256→  await db.$transaction(async (tx) => {
   257→   257→    for (const upd of updates) {
   258→   258→      await tx.marketPrice.upsert({
   259→   259→        where: { productName_city: { productName: upd.productName, city: upd.city } },
   260→   260→        create: upd,
   261→   261→        update: {
   262→   262→          priceMultiplier: upd.priceMultiplier,
   263→   263→          demandMultiplier: upd.demandMultiplier,
   264→   264→          updatedAt: new Date(),
   265→   265→        },
   266→   266→      });
   267→   267→    }
   268→   268→  });
   269→   269→}

   447→   447→export async function simulateBusinessTick(businessId: string): Promise<void> {

   449→   449→    where: { id: businessId },
   450→   450→    include: {
   451→   451→      inventories: true,
   452→   452→      employees: true,
   453→   453→      player: { select: { id: true, cash: true, netWorth: true } },
   454→   454→    },
   455→   455→  });
   456→   456→
   457→   457→  if (!business) return;
   458→   458→
   459→   459→  const businessType = getBusinessType(business.type);
   460→   460→  const city = getCity(business.city);
   461→   461→  if (!businessType || !city) return;
   462→   462→
   463→   463→  const economyConfig = getBusinessEconomyConfig(business.type);
   464→   464→
   465→   465→  // Get active event effects (combined)
   466→   466→  const activeEvents = await db.gameEvent.findMany({
   467→   467→    where: { active: true },
   468→   468→    select: { effects: true },
   469→   469→  });
   470→   470→  const combinedEffects: Record<string, number> = {};
   471→   471→  for (const event of activeEvents) {
   472→   472→    const effects = JSON.parse(event.effects) as Record<string, number>;
   473→   473→    for (const [key, value] of Object.entries(effects)) {
   474→   474→      combinedEffects[key] = (combinedEffects[key] || 0) + value;
   475→   475→    }
   476→   476→  }
   477→   477→
   478→   478→  // Get market prices for this city
   479→   479→  const marketPrices = await db.marketPrice.findMany({
   480→   480→    where: { city: business.city },
   481→   481→  });
   482→   482→  const marketPriceMap: Record<string, { priceMultiplier: number; demandMultiplier: number }> = {};
   483→   483→  for (const mp of marketPrices) {
   484→   484→    marketPriceMap[mp.productName] = {
   485→   485→      priceMultiplier: mp.priceMultiplier,
   486→   486→      demandMultiplier: mp.demandMultiplier,
   487→   487→    };
   488→   488→  }
   489→   489→
   490→   490→  // ---- Step 0.5: CX Demand Modifier (from previous tick) ----
   491→   491→  // Use the business's existing satisfaction/loyalty/reputation to calculate
   492→   492→  // how CX affects THIS tick's customer count. This avoids circular dependency
   493→   493→  // because we use previous-tick CX values to influence current-tick demand.
   494→   494→  const previousCxDemandModifier = calculateCXDemandModifier(
   495→   495→    business.satisfactionScore,
   496→   496→    business.repeatCustomerRate,
   497→   497→    getLoyaltyTier(business.loyaltyScore).multiplier,
   498→   498→  );
   499→   499→
   500→   500→  // Calculate segment demand modifier from previous-tick data
   501→   501→  // Segments react differently to quality, service, reputation
   502→   502→  // NOTE: priceCompetitiveness is NOT passed to segment demands because price is already
   503→   503→  // handled by per-product priceDemandMultiplier AND satisfaction's priceCompetitiveness.
   504→   504→  // Passing it here would triple-count price effects.
   505→   505→  // Similarly, reputation is already in calculatePotentialCustomers Layer 3 AND satisfaction's
   506→   506→  // atmosphere factor, so we use a moderate value to avoid quadruple-counting.
   507→   507→  const previousCxServiceQuality = calculateServiceQuality(
   508→   508→    business.employees.length,
   509→   509→    business.employees.length > 0
   510→   510→      ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   511→   511→      : 0,
   512→   512→    2 + business.level,
   513→   513→  );
   514→   514→  const segmentDemands = calculateSegmentDemands({
   515→   515→    priceCompetitiveness: 1.0,  // Price already handled by priceDemandMultiplier + satisfaction
   516→   516→    productQuality: Math.max(0, Math.min(1, business.healthScore / 100)), // Use health as quality proxy from previous tick
   517→   517→    serviceQuality: previousCxServiceQuality,
   518→   518→    reputation: Math.min(100, business.reputation * 0.5 + 50),  // Dampened: avg 50 ± 25 from reputation
   519→   519→  });
   520→   520→  // Aggregate segment demand: sum of demand multipliers (NOT weighted by share again)
   521→   521→  // demandMultiplier already includes baseShare, so we just sum them
   522→   522→  const segmentDemandModifier = segmentDemands.reduce(
   523→   523→    (sum, seg) => sum + seg.demandMultiplier, 0
   524→   524→  );
   525→   525→
   526→   526→  // ---- Step 0.6: Marketing Demand Modifier (Phase 4) ----
   527→   527→  // Fetch gameDay early (needed for marketing metrics and later for business metrics)
   528→   528→  const gameDayState = await db.gameState.findUnique({ where: { key: 'gameDay' } });
   529→   529→  const gameDay = parseInt(gameDayState?.value || '1', 10);
   530→   530→
   531→   531→  const marketingEffect = await processMarketingTick(
   532→   532→    business.id,
   533→   533→    business.level,
   534→   534→    business.satisfactionScore,
   535→   535→    business.loyaltyScore,
   536→   536→    business.repeatCustomerRate,
   537→   537→    business.brandAwareness,
   538→   538→    segmentDemands,
   539→   539→    gameDay,
   540→   540→  );
   541→   541→
   542→   542→  // ---- Step 1: Calculate potential customers ----
   543→   543→  const totalStock = business.inventories.reduce((sum, inv) => sum + inv.quantity, 0);
   544→   544→  // Max stock capacity: sum of product maxStock values for this business type
   545→   545→  const productDefs = PRODUCTS[business.type] || [];
   546→   546→  const maxStockCapacity = productDefs.reduce((sum, p) => sum + p.maxStock, 0);
   547→   547→  const avgEmployeeSkill = business.employees.length > 0
   548→   548→    ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   549→   549→    : 0;
   550→   550→
   551→   551→  const basePotentialCustomers = calculatePotentialCustomers({
   552→   552→    baseCustomers: businessType.baseCustomers,
   553→   553→    cityMultiplier: city.customerMultiplier,
   554→   554→    level: business.level,
   555→   555→    reputation: business.reputation,
   556→   556→    totalStock,
   557→   557→    maxStockCapacity,
   558→   558→    employeeCount: business.employees.length,
   559→   559→    avgEmployeeSkill,
   560→   560→    eventCustomerEffect: combinedEffects['all_customers'] || 0,
   561→   561→    businessDemandEffect: combinedEffects[`${businessType.id}_demand`] || 0,
   562→   562→    businessTypeId: business.type,
   563→   563→  });
   564→   564→
   565→   565→  // Apply CX demand modifier, segment demand modifier, and marketing modifiers
   566→   566→  // CX modifier: satisfaction/loyalty affect how many customers visit (0.2-2.0 range)
   567→   567→  // Segment modifier: different customer segments find the business more/less attractive
   568→   568→  // Phase 4: Marketing demand modifier and brand awareness bonus
   569→   569→  const marketingDemandModifier = marketingEffect.combinedDemandModifier;
   570→   570→  const brandAwarenessBonus = calculateBrandAwarenessDemandBonus(marketingEffect.newBrandAwareness);
   571→   571→
   572→   572→  const potentialCustomers = Math.max(0, Math.floor(
   573→   573→    basePotentialCustomers * previousCxDemandModifier * segmentDemandModifier * marketingDemandModifier * brandAwarenessBonus
   574→   574→  ));
   575→   575→
   576→   576→  // ---- Step 2: Product-level sales simulation ----
   577→   577→  let totalRevenue = 0;
   578→   578→  let totalCOGS = 0;
   579→   579→  const inventoryUpdates: { id: string; quantityDecrement: number }[] = [];
   580→   580→  const salesResults: ProductSalesResult[] = [];
   581→   581→
   582→   582→  for (const inventory of business.inventories) {
   583→   583→    if (inventory.quantity <= 0) continue;
   584→   584→
   585→   585→    const productDef = productDefs.find(p => p.name === inventory.productName);
   586→   586→    if (!productDef) continue;
   587→   587→
   588→   588→    // Product demand config
   589→   589→    const prodDemandConfig = getProductDemandConfig(inventory.productName);
   590→   590→
   591→   591→    // Market reference price = typical retail price (basePrice × (1 + suggestedMarkup) × priceMultiplier)
   592→   592→    // This represents what the MARKET typically charges, not the wholesale cost.
   593→   593→    // Players compare their price to market retail price, not to their own cost.
   594→   594→    const mp = marketPriceMap[inventory.productName];
   595→   595→    const typicalRetailPrice = productDef.basePrice * (1 + productDef.suggestedMarkup);
   596→   596→    const marketReferencePrice = Math.round(
   597→   597→      typicalRetailPrice * (mp?.priceMultiplier || 1)
   598→   598→    );
   599→   599→
   600→   600→    // Market demand multiplier from events
   601→   601→    const demandMult = mp?.demandMultiplier || 1;
   602→   602→
   603→   603→    // Event demand effect for this product's category
   604→   604→    const eventDemandEffect = combinedEffects[`${productDef.category}_demand`] || 0;
   605→   605→
   606→   606→    // Run the complete sales pipeline
   607→   607→    const salesResult = simulateProductSales({
   608→   608→      inventoryId: inventory.id,
   609→   609→      productName: inventory.productName,
   610→   610→      quantity: inventory.quantity,
   611→   611→      purchasePrice: inventory.purchasePrice,
   612→   612→      sellPrice: inventory.sellPrice,
   613→   613→      marketReferencePrice,
   614→   614→      baseDemand: prodDemandConfig.baseDemand,
   615→   615→      demandMultiplier: demandMult,
   616→   616→      eventDemandEffect,
   617→   617→      productPriceSensitivity: prodDemandConfig.priceSensitivity,
   618→   618→      productVolatility: prodDemandConfig.volatility,
   619→   619→      businessPriceSensitivity: economyConfig.priceSensitivity,
   620→   620→      potentialCustomers,
   621→   621→    });
   622→   622→
   623→   623→    if (salesResult.itemsSold > 0) {
   624→   624→      totalRevenue += salesResult.revenue;
   625→   625→      totalCOGS += salesResult.costOfGoodsSold;
   626→   626→      inventoryUpdates.push({
   627→   627→        id: inventory.id,
   628→   628→        quantityDecrement: salesResult.itemsSold,
   629→   629→      });
   630→   630→    }
   631→   631→
   632→   632→    salesResults.push(salesResult);
   633→   633→  }
   634→   634→
   635→   635→  // ---- Step 3: Calculate expenses ----
   636→   636→  const totalMonthlySalaries = business.employees.reduce((sum, emp) => sum + emp.salary, 0);
   637→   637→  const grossProfit = totalRevenue - totalCOGS;
   638→   638→
   639→   639→  const expenses = calculateBusinessExpenses({
   640→   640→    baseRent: businessType.rent,
   641→   641→    level: business.level,
   642→   642→    cityRentMultiplier: city.rentMultiplier,
   643→   643→    totalMonthlySalaries,
   644→   644→    businessLevel: business.level,
   645→   645→    businessTypeId: business.type,
   646→   646→    revenue: totalRevenue,
   647→   647→    grossProfit,
   648→   648→  });
   649→   649→
   650→   650→  // ---- Step 4: Calculate net profit ----
   651→   651→  // Phase 4: Include marketing spend in total expenses
   652→   652→  const totalExpenseWithMarketing = expenses.totalExpense + marketingEffect.totalMarketingSpend;
   653→   653→  const dailyProfit = calculateNetProfit(totalRevenue, totalCOGS, totalExpenseWithMarketing);
   654→   654→
   655→   655→  // ---- Step 5: Calculate reputation change ----
   656→   656→  const outOfStockRatio = business.inventories.length > 0
   657→   657→    ? business.inventories.filter(inv => inv.quantity <= 0).length / business.inventories.length
   658→   658→    : 0;
   659→   659→  const managers = business.employees.filter(e => e.role === 'MANAGER');
   660→   660→  const cleaners = business.employees.filter(e => e.role === 'CLEANER');
   661→   661→
   662→   662→  const reputationChange = calculateReputationChange({
   663→   663→    dailyProfit,
   664→   664→    outOfStockRatio,
   665→   665→    managers: managers.map(m => ({ skill: m.skill })),
   666→   666→    cleaners: cleaners.map(c => ({ skill: c.skill })),
   667→   667→  });
   668→   668→
   669→   669→  const newReputation = Math.max(
   670→   670→    ECONOMY_CONFIG.minReputation,
   671→   671→    Math.min(ECONOMY_CONFIG.maxReputation, business.reputation + reputationChange)
   672→   672→  );
   673→   673→
   674→   674→  // ---- Step 6: Calculate business health score ----
   675→   675→  const healthResult = calculateBusinessHealth({
   676→   676→    dailyProfit,
   677→   677→    dailyRevenue: totalRevenue,
   678→   678→    businessCash: business.cash + dailyProfit,
   679→   679→    totalStock,
   680→   680→    maxStockCapacity,
   681→   681→    reputation: newReputation,
   682→   682→    businessTypeId: business.type,
   683→   683→  });
   684→   684→
   685→   685→  // ---- Step 6.5: Customer Experience (Phase 3) ----
   686→   686→  // Calculate price competitiveness across all products
   687→   687→  const cxPriceCompetitiveness = calculatePriceCompetitiveness(
   688→   688→    business.inventories.map(inv => ({ productName: inv.productName, sellPrice: inv.sellPrice })),
   689→   689→    productDefs.map(p => ({ name: p.name, basePrice: p.basePrice, suggestedMarkup: p.suggestedMarkup })),
   690→   690→    marketPriceMap,
   691→   691→  );
   692→   692→
   693→   693→  // Calculate service quality from employees
   694→   694→  const idealEmployeeCount = 2 + business.level; // Base 2 + 1 per level
   695→   695→  const cxServiceQuality = calculateServiceQuality(
   696→   696→    business.employees.length,
   697→   697→    avgEmployeeSkill,
   698→   698→    idealEmployeeCount,
   699→   699→  );
   700→   700→
   701→   701→  // Calculate product quality from margin (higher margin = better products)
   702→   702→  const cxProductQuality = totalRevenue > 0
   703→   703→    ? Math.min(1, Math.max(0, (totalRevenue - totalCOGS) / totalRevenue / 0.3))
   704→   704→    : 0.5;
   705→   705→
   706→   706→  // Calculate satisfaction
   707→   707→  const cxSatisfaction = calculateSatisfaction({
   708→   708→    priceCompetitiveness: cxPriceCompetitiveness,
   709→   709→    stockAvailability: maxStockCapacity > 0 ? totalStock / maxStockCapacity : 0,
   710→   710→    serviceQuality: cxServiceQuality,
   711→   711→    productQuality: cxProductQuality,
   712→   712→    atmosphere: newReputation / 100,
   713→   713→    currentSatisfaction: business.satisfactionScore,
   714→   714→  });
   715→   715→
   716→   716→  // Determine if this was a positive experience
   717→   717→  const wasPositiveExperience = cxSatisfaction.overall >= 50 && dailyProfit >= 0;
   718→   718→
   719→   719→  // Calculate loyalty
   720→   720→  const cxLoyalty = calculateLoyalty({
   721→   721→    currentLoyaltyScore: business.loyaltyScore,
   722→   722→    currentRepeatRate: business.repeatCustomerRate,
   723→   723→    wasPositiveExperience,
   724→   724→    totalCustomers: potentialCustomers,
   725→   725→    numberOfReviews: business.totalReviews,
   726→   726→    averageRating: business.avgReviewRating,
   727→   727→    loyaltyMultiplier: getLoyaltyTier(business.loyaltyScore).multiplier,
   728→   728→  });
   729→   729→
   730→   730→  // ---- Step 7: Get game day for metrics ----
   731→   731→  // (gameDay already fetched at Step 0.6 for marketing processing)
   732→   732→
   733→   733→  // ---- Step 8: Execute all updates in a single transaction ----
   734→   734→  await db.$transaction(async (tx) => {
   735→   735→    // Update inventory quantities
   736→   736→    for (const upd of inventoryUpdates) {
   737→   737→      await tx.inventory.update({
   738→   738→        where: { id: upd.id },
   739→   739→        data: { quantity: { decrement: upd.quantityDecrement } },
   740→   740→      });
   741→   741→    }
   742→   742→
   743→   743→    // Phase 1 FIX: Distribute business profit to player cash
   744→   744→    // Previously, profit accumulated in business.cash but was never accessible.
   745→   745→    // Now: business profit is transferred to the player each tick.
   746→   746→    const profitToDistribute = dailyProfit;
   747→   747→
   748→   748→    // Update business (including Phase 3 CX fields and Phase 4 Marketing fields)
   749→   749→    await tx.business.update({
   750→   750→      where: { id: business.id },
   751→   751→      data: {
   752→   752→        dailyRevenue: roundTaka(totalRevenue),
   753→   753→        dailyExpense: roundTaka(totalExpenseWithMarketing),
   754→   754→        dailyProfit: roundTaka(dailyProfit),
   755→   755→        dailyCOGS: roundTaka(totalCOGS),
   756→   756→        dailyCustomers: potentialCustomers,
   757→   757→        cash: business.cash + roundTaka(dailyProfit),
   758→   758→        totalRevenue: business.totalRevenue + roundTaka(totalRevenue),
   759→   759→        totalProfit: business.totalProfit + roundTaka(dailyProfit),
   760→   760→        reputation: newReputation,
   761→   761→        healthScore: healthResult.score,
   762→   762→        // Phase 3: CX fields
   763→   763→        satisfactionScore: cxSatisfaction.overall,
   764→   764→        loyaltyScore: cxLoyalty.loyaltyScore,
   765→   765→        repeatCustomerRate: cxLoyalty.repeatCustomerRate,
   766→   766→        // Phase 4: Marketing fields
   767→   767→        brandAwareness: marketingEffect.newBrandAwareness,
   768→   768→      },
   769→   769→    });
   770→   770→
   771→   771→    // Distribute profit to player cash
   772→   772→    await tx.player.update({
   773→   773→      where: { id: business.playerId },
   774→   774→      data: { cash: { increment: roundTaka(profitToDistribute) } },
   775→   775→    });
   776→   776→
   777→   777→    // Phase 4: Deduct marketing spend from player cash
   778→   778→    // Marketing spend is separate from business expenses (it's a direct player cost)
   779→   779→    if (marketingEffect.totalMarketingSpend > 0) {
   780→   780→      await tx.player.update({
   781→   781→        where: { id: business.playerId },
   782→   782→        data: { cash: { decrement: roundTaka(marketingEffect.totalMarketingSpend) } },
   783→   783→      });
   784→   784→
   785→   785→      // Phase 4: Attribute revenue to active campaigns (proportional to their spend)
   786→   786→      // This gives campaigns ROI data for analytics
   787→   787→      if (totalRevenue > 0 && marketingEffect.totalConversions > 0) {
   788→   788→        const activeCampaigns = await tx.marketingCampaign.findMany({
   789→   789→          where: { businessId: business.id, status: 'ACTIVE' },
   790→   790→          select: { id: true, totalSpend: true },
   791→   791→        });
   792→   792→        const totalSpend = activeCampaigns.reduce((sum, c) => sum + c.totalSpend, 0);
   793→   793→        if (totalSpend > 0) {
   794→   794→          for (const campaign of activeCampaigns) {
   795→   795→            // Attribute revenue proportional to this campaign's spend share
   796→   796→            const spendShare = campaign.totalSpend / totalSpend;
   797→   797→            // Only attribute the "extra" revenue from marketing (not base revenue)
   798→   798→            const marketingRevenueBoost = totalRevenue * (marketingDemandModifier - 1) * (brandAwarenessBonus - 1 + 1);
   799→   799→            const attributedRevenue = Math.max(0, marketingRevenueBoost * spendShare);
   800→   800→            if (attributedRevenue > 0) {
   801→   801→              await tx.marketingCampaign.update({
   802→   802→                where: { id: campaign.id },
   803→   803→                data: { revenueInfluenced: { increment: roundTaka(attributedRevenue) } },
   804→   804→              });
   805→   805→            }
   806→   806→          }
   807→   807→        }
   808→   808→      }
   809→   809→    }
   810→   810→
   811→   811→    // Save daily metrics for analytics (including Phase 3 CX metrics)
   812→   812→    await tx.businessMetric.upsert({
   813→   813→      where: { businessId_gameDay: { businessId: business.id, gameDay } },
   814→   814→      create: {
   815→   815→        businessId: business.id,
   816→   816→        gameDay,
   817→   817→        revenue: roundTaka(totalRevenue),
   818→   818→        expenses: roundTaka(totalExpenseWithMarketing),
   819→   819→        profit: roundTaka(dailyProfit),
   820→   820→        cogs: roundTaka(totalCOGS),
   821→   821→        customers: potentialCustomers,
   822→   822→        reputation: newReputation,
   823→   823→        // Phase 3: CX metrics
   824→   824→        satisfaction: cxSatisfaction.overall,
   825→   825→        loyalty: cxLoyalty.loyaltyScore,
   826→   826→        nps: business.npsScore,
   827→   827→      },
   828→   828→      update: {
   829→   829→        revenue: roundTaka(totalRevenue),
   830→   830→        expenses: roundTaka(totalExpenseWithMarketing),
   831→   831→        profit: roundTaka(dailyProfit),
   832→   832→        cogs: roundTaka(totalCOGS),
   833→   833→        customers: potentialCustomers,
   834→   834→        reputation: newReputation,
   835→   835→        // Phase 3: CX metrics
   836→   836→        satisfaction: cxSatisfaction.overall,
   837→   837→        loyalty: cxLoyalty.loyaltyScore,
   838→   838→        nps: business.npsScore,
   839→   839→      },
   840→   840→    });
   841→   841→
   842→   842→    // Log (with Phase 3 CX info)
   843→   843→    const cogsStr = totalCOGS > 0 ? ` | COGS ৳${roundTaka(totalCOGS).toLocaleString()}` : '';
   844→   844→    await tx.gameLog.create({
   845→   845→      data: {
   846→   846→        playerId: business.playerId,
   847→   847→        businessId: business.id,
   848→   848→        type: dailyProfit >= 0 ? 'PROFIT' : 'LOSS',
   849→   849→        message: `${business.name}: Rev ৳${roundTaka(totalRevenue).toLocaleString()}${cogsStr} | Exp ৳${roundTaka(totalExpenseWithMarketing).toLocaleString()} | ${dailyProfit >= 0 ? 'Profit' : 'Loss'} ৳${Math.abs(roundTaka(dailyProfit)).toLocaleString()} | ${potentialCustomers} customers | Health ${healthResult.score} | Sat ${Math.round(cxSatisfaction.overall)} | Loy ${cxLoyalty.tier} | Mkt ৳${roundTaka(marketingEffect.totalMarketingSpend).toLocaleString()}`,
   850→   850→        amount: dailyProfit,
   851→   851→      },
   852→   852→    });
   853→   853→
   854→   854→    // Phase 3: Generate and persist customer reviews
   855→   855→    const cxReviews = generateReviews(business.id, gameDay, cxSatisfaction, REVIEW_CONFIG.maxPerTick);
   856→   856→    for (const review of cxReviews) {
   857→   857→      await tx.customerReview.create({
   858→   858→        data: {
   859→   859→          businessId: review.businessId,
   860→   860→          gameDay: review.gameDay,
   861→   861→          rating: review.rating,
   862→   862→          sentiment: review.sentiment,
   863→   863→          category: review.category,
   864→   864→          comment: review.comment,
   865→   865→          segment: review.segment,
   866→   866→        },
   867→   867→      });
   868→   868→    }
   869→   869→
   870→   870→    // Phase 3: Update NPS and review stats
   871→   871→    if (cxReviews.length > 0) {
   872→   872→      const allReviews = await tx.customerReview.findMany({
   873→   873→        where: { businessId: business.id },
   874→   874→        select: { rating: true },
   875→   875→        orderBy: { createdAt: 'desc' },
   876→   876→        take: REVIEW_CONFIG.keepLast,
   877→   877→      });
   878→   878→      const npsResult = calculateNPS(allReviews);
   879→   879→      const totalReviews = allReviews.length;
   880→   880→      const avgRating = totalReviews > 0
   881→   881→        ? allReviews.reduce((sum, r) => sum + r.rating, 0) / totalReviews
   882→   882→        : 0;
   883→   883→
   884→   884→      await tx.business.update({
   885→   885→        where: { id: business.id },
   886→   886→        data: {
   887→   887→          npsScore: npsResult.nps,
   888→   888→          totalReviews,
   889→   889→          avgReviewRating: Math.round(avgRating * 100) / 100,
   890→   890→        },
   891→   891→      });
   892→   892→
   893→   893→      // Prune old reviews (keep only last N)
   894→   894→      const reviewCount = await tx.customerReview.count({
   895→   895→        where: { businessId: business.id },
   896→   896→      });
   897→   897→      if (reviewCount > REVIEW_CONFIG.keepLast) {
   898→   898→        const oldReviews = await tx.customerReview.findMany({
   899→   899→          where: { businessId: business.id },
   900→   900→          orderBy: { createdAt: 'asc' },
   901→   901→          take: reviewCount - REVIEW_CONFIG.keepLast,
   902→   902→          select: { id: true },
   903→   903→        });
   904→   904→        for (const old of oldReviews) {
   905→   905→          await tx.customerReview.delete({ where: { id: old.id } });
   906→   906→        }
   907→   907→      }
   908→   908→    }
   909→   909→
   910→   910→    // Recalculate net worth within the same transaction
   911→   911→    await recalculateNetWorth(tx, business.playerId);
   912→   912→  });
   913→   913→}

   915→   915→// ---- Loan Payment Processing ----
   916→   916→
   917→   917→async function processLoanPayments(): Promise<void> {
   918→   918→  const activeLoans = await db.loan.findMany({
   919→   919→    where: { status: 'ACTIVE' },
   920→   920→  });
   921→   921→
   922→   922→  if (activeLoans.length === 0) return;
   923→   923→
   924→   924→  for (const loan of activeLoans) {
