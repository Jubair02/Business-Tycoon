    68→// ---- Tick Concurrency Lock ----
    69→
    70→/**
    71→ * Atomically attempt to acquire the tick lock using the GameState key-value store.
    72→ * Returns true if the lock was acquired, false if it was already held.
    73→ *
    74→ * Stale lock recovery: if the lock has been held for longer than
    75→ * STALE_TICK_TIMEOUT_MS, the lock is stolen and a warning is logged.
    76→ */
    77→export async function acquireTickLock(): Promise<boolean> {
    78→  const now = new Date();
    79→
    80→  return db.$transaction(async (tx) => {
    81→    // Read current lock state
    82→    const lockState = await tx.gameState.findUnique({ where: { key: 'tickInProgress' } });
    83→    const startedAtState = await tx.gameState.findUnique({ where: { key: 'tickStartedAt' } });
    84→
    85→    const isLocked = lockState?.value === 'true';
    86→    const startedAt = startedAtState?.value ? new Date(startedAtState.value) : null;
    87→
    88→    if (isLocked && startedAt) {
    89→      const elapsed = now.getTime() - startedAt.getTime();
    90→      if (elapsed < STALE_TICK_TIMEOUT_MS) {
    91→        return false; // Lock is legitimately held
    92→      }
    93→      console.warn(`[GameEngine] Stale tick lock detected (held for ${Math.round(elapsed / 1000)}s). Stealing lock.`);
    94→    } else if (isLocked) {
    95→      console.warn('[GameEngine] Tick lock held with no timestamp. Stealing lock.');
    96→    }
    97→
    98→    // Acquire the lock
    99→    const nowISO = now.toISOString();
   100→    await tx.gameState.upsert({
   101→      where: { key: 'tickInProgress' },
   102→      create: { key: 'tickInProgress', value: 'true' },
   103→      update: { value: 'true' },
   104→    });
   105→    await tx.gameState.upsert({
   106→      where: { key: 'tickStartedAt' },
   107→      create: { key: 'tickStartedAt', value: nowISO },
   108→      update: { value: nowISO },
   109→    });
   110→
   111→    // Increment tick version
   112→    const versionState = await tx.gameState.findUnique({ where: { key: 'tickVersion' } });
   113→    const currentVersion = parseInt(versionState?.value || '0', 10);
   114→    await tx.gameState.upsert({
   115→      where: { key: 'tickVersion' },
   116→      create: { key: 'tickVersion', value: '1' },
   117→      update: { value: String(currentVersion + 1) },
   118→    });
   119→
   120→    return true;
   121→  });
   122→}

   132→// ---- Net Worth Recalculation ----
   133→async function recalculateNetWorth(tx: PrismaTx, playerId: string): Promise<void> {
   134→  const player = await tx.player.findUnique({
   135→    where: { id: playerId },
   136→    select: { id: true, cash: true },
   137→  });
   138→  if (!player) {
   139→    console.error(`[GameEngine] recalculateNetWorth: Player ${playerId} not found`);
   140→    return;
   141→  }
   142→
   143→  const allBusinesses = await tx.business.findMany({
   144→    where: { playerId },
   145→    select: { cash: true },
   146→  });
   147→  const totalBusinessCash = allBusinesses.reduce((sum, b) => sum + b.cash, 0);
   148→
   149→  const allInventory = await tx.inventory.findMany({
   150→    where: { business: { playerId } },
   151→    select: { quantity: true, purchasePrice: true },
   152→  });
   153→  const totalInventoryValue = allInventory.reduce((sum, inv) => sum + inv.quantity * inv.purchasePrice, 0);
   154→
   155→  const activeLoans = await tx.loan.findMany({
   156→    where: { playerId, status: 'ACTIVE' },
   157→    select: { remainingDebt: true },
   158→  });
   159→  const outstandingDebt = activeLoans.reduce((sum, loan) => sum + loan.remainingDebt, 0);
   160→
   161→  const netWorth = player.cash + totalBusinessCash + totalInventoryValue - outstandingDebt;
   162→  await tx.player.update({ where: { id: playerId }, data: { netWorth } });
   163→}

   165→// ---- Event Generation ----
   166→export async function generateEvent(): Promise<void> {
   167→  const totalWeight = EVENT_TEMPLATES.reduce((sum, e) => sum + e.weight, 0);
   168→  let random = Math.random() * totalWeight;
   169→  let selected: EventTemplate | null = null;
   170→  for (const event of EVENT_TEMPLATES) {
   171→    random -= event.weight;
   172→    if (random <= 0) {
   173→      selected = event;
   174→      break;
   175→    }
   176→  }
   177→  if (!selected) selected = EVENT_TEMPLATES[0];
   178→
   179→  const now = new Date();
   180→  const endAt = new Date(now.getTime() + selected.durationDays * 60 * 1000);
   181→  await db.gameEvent.create({
   182→    data: {
   183→      title: selected.title,
   184→      description: selected.description,
   185→      type: selected.type,
   186→      effects: JSON.stringify(selected.effects),
   187→      active: true,
   188→      startsAt: now,
   189→      endsAt: endAt,
   190→    },
   191→  });
   192→}

   200→export async function updateMarketPrices(): Promise<void> {
   201→  const allProducts = Object.values(PRODUCTS).flat();
   202→  const cities = CITIES.map(c => c.id);
   203→
   204→  // Get active events for price modifiers
   205→  const activeEvents = await db.gameEvent.findMany({ where: { active: true } });
   206→  const categoryPriceMod: Record<string, number> = {};
   207→  let allPriceMod = 0;
   208→  for (const event of activeEvents) {
   209→    const effects = JSON.parse(event.effects) as Record<string, number>;
   210→    for (const [key, value] of Object.entries(effects)) {
   211→      if (key === 'all_price' || key === 'all_imported') allPriceMod += value;
   212→      if (key.endsWith('_price')) {
   213→        const cat = key.replace('_price', '');
   214→        categoryPriceMod[cat] = (categoryPriceMod[cat] || 0) + value;
   215→      }
   216→    }
   217→  }
   218→
   219→  const updates: { productName: string; city: string; priceMultiplier: number; demandMultiplier: number }[] = [];
   220→
   221→  for (const city of cities) {
   222→    for (const product of allProducts) {
   223→      // Get current market price (for retention)
   224→      const existing = await db.marketPrice.findUnique({
   225→        where: { productName_city: { productName: product.name, city } },
   226→      });
   227→      const previousPriceMult = existing?.priceMultiplier ?? 1;
   228→
   229→      // Calculate event price modifier for this product
   230→      const eventPriceMod = allPriceMod + (categoryPriceMod[product.category] || 0);
   231→
   232→      // Use Phase 1 retention-based update
   233→      const { priceMultiplier, demandMultiplier } = calculateMarketPriceUpdate(
   234→        previousPriceMult,
   235→        eventPriceMod
   236→      );
   237→
   238→      updates.push({ productName: product.name, city, priceMultiplier, demandMultiplier });
   239→    }
   240→  }
   241→
   242→  // Batch update in a transaction
   243→  await db.$transaction(async (tx) => {
   244→    for (const upd of updates) {
   245→      await tx.marketPrice.upsert({
   246→        where: { productName_city: { productName: upd.productName, city: upd.city } },
   247→        create: upd,
   248→        update: {
   249→          priceMultiplier: upd.priceMultiplier,
   250→          demandMultiplier: upd.demandMultiplier,
   251→          updatedAt: new Date(),
   252→        },
   253→      });
   254→    }
   255→  });
   256→}

   290→export async function simulateBusinessTick(businessId: string): Promise<void> {
   291→  const business = await db.business.findUnique({
   292→    where: { id: businessId },
   293→    include: {
   294→      inventories: true,
   295→      employees: true,
   296→      player: { select: { id: true, cash: true, netWorth: true } },
   297→    },
   298→  });
   299→
   300→  if (!business) return;
   301→
   302→  const businessType = getBusinessType(business.type);
   303→  const city = getCity(business.city);
   304→  if (!businessType || !city) return;
   305→
   306→  const economyConfig = getBusinessEconomyConfig(business.type);
   307→
   308→  // Get active event effects (combined)
   309→  const activeEvents = await db.gameEvent.findMany({
   310→    where: { active: true },
   311→    select: { effects: true },
   312→  });
   313→  const combinedEffects: Record<string, number> = {};
   314→  for (const event of activeEvents) {
   315→    const effects = JSON.parse(event.effects) as Record<string, number>;
   316→    for (const [key, value] of Object.entries(effects)) {
   317→      combinedEffects[key] = (combinedEffects[key] || 0) + value;
   318→    }
   319→  }
   320→
   321→  // Get market prices for this city
   322→  const marketPrices = await db.marketPrice.findMany({
   323→    where: { city: business.city },
   324→  });
   325→  const marketPriceMap: Record<string, { priceMultiplier: number; demandMultiplier: number }> = {};
   326→  for (const mp of marketPrices) {
   327→    marketPriceMap[mp.productName] = {
   328→      priceMultiplier: mp.priceMultiplier,
   329→      demandMultiplier: mp.demandMultiplier,
   330→    };
   331→  }
   332→
   333→  // ---- Step 0.5: CX Demand Modifier (from previous tick) ----
   334→  // Use the business's existing satisfaction/loyalty/reputation to calculate
   335→  // how CX affects THIS tick's customer count. This avoids circular dependency
   336→  // because we use previous-tick CX values to influence current-tick demand.
   337→  const previousCxDemandModifier = calculateCXDemandModifier(
   338→    business.satisfactionScore,
   339→    business.repeatCustomerRate,
   340→    getLoyaltyTier(business.loyaltyScore).multiplier,
   341→  );
   342→
   343→  // Calculate segment demand modifier from previous-tick data
   344→  // Segments react differently to quality, service, reputation
   345→  // NOTE: priceCompetitiveness is NOT passed to segment demands because price is already
   346→  // handled by per-product priceDemandMultiplier AND satisfaction's priceCompetitiveness.
   347→  // Passing it here would triple-count price effects.
   348→  // Similarly, reputation is already in calculatePotentialCustomers Layer 3 AND satisfaction's
   349→  // atmosphere factor, so we use a moderate value to avoid quadruple-counting.
   350→  const previousCxServiceQuality = calculateServiceQuality(
   351→    business.employees.length,
   352→    business.employees.length > 0
   353→      ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   354→      : 0,
   355→    2 + business.level,
   356→  );
   357→  const segmentDemands = calculateSegmentDemands({
   358→    priceCompetitiveness: 1.0,  // Price already handled by priceDemandMultiplier + satisfaction
   359→    productQuality: Math.max(0, Math.min(1, business.healthScore / 100)), // Use health as quality proxy from previous tick
   360→    serviceQuality: previousCxServiceQuality,
   361→    reputation: Math.min(100, business.reputation * 0.5 + 50),  // Dampened: avg 50 ± 25 from reputation
   362→  });
   363→  // Aggregate segment demand: sum of demand multipliers (NOT weighted by share again)
   364→  // demandMultiplier already includes baseShare, so we just sum them
   365→  const segmentDemandModifier = segmentDemands.reduce(
   366→    (sum, seg) => sum + seg.demandMultiplier, 0
   367→  );
   368→
   369→  // ---- Step 1: Calculate potential customers ----
   370→  const totalStock = business.inventories.reduce((sum, inv) => sum + inv.quantity, 0);
   371→  // Max stock capacity: sum of product maxStock values for this business type
   372→  const productDefs = PRODUCTS[business.type] || [];
   373→  const maxStockCapacity = productDefs.reduce((sum, p) => sum + p.maxStock, 0);
   374→  const avgEmployeeSkill = business.employees.length > 0
   375→    ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   376→    : 0;
   377→
   378→  const basePotentialCustomers = calculatePotentialCustomers({
   379→    baseCustomers: businessType.baseCustomers,
   380→    cityMultiplier: city.customerMultiplier,
   381→    level: business.level,
   382→    reputation: business.reputation,
   383→    totalStock,
   384→    maxStockCapacity,
   385→    employeeCount: business.employees.length,
   386→    avgEmployeeSkill,
   387→    eventCustomerEffect: combinedEffects['all_customers'] || 0,
   388→    businessDemandEffect: combinedEffects[`${businessType.id}_demand`] || 0,
   389→    businessTypeId: business.type,
   390→  });
   391→
   392→  // Apply CX demand modifier and segment demand modifier
   393→  // CX modifier: satisfaction/loyalty affect how many customers visit (0.2-2.0 range)
   394→  // Segment modifier: different customer segments find the business more/less attractive
   395→  const potentialCustomers = Math.max(0, Math.floor(
   396→    basePotentialCustomers * previousCxDemandModifier * segmentDemandModifier
   397→  ));
   398→
   399→  // ---- Step 2: Product-level sales simulation ----
   400→  let totalRevenue = 0;
   401→  let totalCOGS = 0;
   402→  const inventoryUpdates: { id: string; quantityDecrement: number }[] = [];
   403→  const salesResults: ProductSalesResult[] = [];
   404→
   405→  for (const inventory of business.inventories) {
   406→    if (inventory.quantity <= 0) continue;
   407→
   408→    const productDef = productDefs.find(p => p.name === inventory.productName);
   409→    if (!productDef) continue;
   410→
   411→    // Product demand config
   412→    const prodDemandConfig = getProductDemandConfig(inventory.productName);
   413→
   414→    // Market reference price = typical retail price (basePrice × (1 + suggestedMarkup) × priceMultiplier)
   415→    // This represents what the MARKET typically charges, not the wholesale cost.
   416→    // Players compare their price to market retail price, not to their own cost.
   417→    const mp = marketPriceMap[inventory.productName];
   418→    const typicalRetailPrice = productDef.basePrice * (1 + productDef.suggestedMarkup);
   419→    const marketReferencePrice = Math.round(
   420→      typicalRetailPrice * (mp?.priceMultiplier || 1)
   421→    );
   422→
   423→    // Market demand multiplier from events
   424→    const demandMult = mp?.demandMultiplier || 1;
   425→
   426→    // Event demand effect for this product's category
   427→    const eventDemandEffect = combinedEffects[`${productDef.category}_demand`] || 0;
   428→
   429→    // Run the complete sales pipeline
   430→    const salesResult = simulateProductSales({
   431→      inventoryId: inventory.id,
   432→      productName: inventory.productName,
   433→      quantity: inventory.quantity,
   434→      purchasePrice: inventory.purchasePrice,
   435→      sellPrice: inventory.sellPrice,
   436→      marketReferencePrice,
   437→      baseDemand: prodDemandConfig.baseDemand,
   438→      demandMultiplier: demandMult,
   439→      eventDemandEffect,
   440→      productPriceSensitivity: prodDemandConfig.priceSensitivity,
   441→      productVolatility: prodDemandConfig.volatility,
   442→      businessPriceSensitivity: economyConfig.priceSensitivity,
   443→      potentialCustomers,
   444→    });
   445→
   446→    if (salesResult.itemsSold > 0) {
   447→      totalRevenue += salesResult.revenue;
   448→      totalCOGS += salesResult.costOfGoodsSold;
   449→      inventoryUpdates.push({
   450→        id: inventory.id,
   451→        quantityDecrement: salesResult.itemsSold,
   452→      });
   453→    }
   454→
   455→    salesResults.push(salesResult);
   456→  }
   457→
   458→  // ---- Step 3: Calculate expenses ----
   459→  const totalMonthlySalaries = business.employees.reduce((sum, emp) => sum + emp.salary, 0);
   460→  const grossProfit = totalRevenue - totalCOGS;
   461→
   462→  const expenses = calculateBusinessExpenses({
   463→    baseRent: businessType.rent,
   464→    level: business.level,
   465→    cityRentMultiplier: city.rentMultiplier,
   466→    totalMonthlySalaries,
   467→    businessLevel: business.level,
   468→    businessTypeId: business.type,
   469→    revenue: totalRevenue,
   470→    grossProfit,
   471→  });
   472→
   473→  // ---- Step 4: Calculate net profit ----
   474→  const dailyProfit = calculateNetProfit(totalRevenue, totalCOGS, expenses.totalExpense);
   475→
   476→  // ---- Step 5: Calculate reputation change ----
   477→  const outOfStockRatio = business.inventories.length > 0
   478→    ? business.inventories.filter(inv => inv.quantity <= 0).length / business.inventories.length
   479→    : 0;
   480→  const managers = business.employees.filter(e => e.role === 'MANAGER');
   481→  const cleaners = business.employees.filter(e => e.role === 'CLEANER');
   482→
   483→  const reputationChange = calculateReputationChange({
   484→    dailyProfit,
   485→    outOfStockRatio,
   486→    managers: managers.map(m => ({ skill: m.skill })),
   487→    cleaners: cleaners.map(c => ({ skill: c.skill })),
   488→  });
   489→
   490→  const newReputation = Math.max(
   491→    ECONOMY_CONFIG.minReputation,
   492→    Math.min(ECONOMY_CONFIG.maxReputation, business.reputation + reputationChange)
   493→  );
   494→
   495→  // ---- Step 6: Calculate business health score ----
   496→  const healthResult = calculateBusinessHealth({
   497→    dailyProfit,
   498→    dailyRevenue: totalRevenue,
   499→    businessCash: business.cash + dailyProfit,
   500→    totalStock,
   501→    maxStockCapacity,
   502→    reputation: newReputation,
   503→    businessTypeId: business.type,
   504→  });
   505→
   506→  // ---- Step 6.5: Customer Experience (Phase 3) ----
   507→  // Calculate price competitiveness across all products
   508→  const cxPriceCompetitiveness = calculatePriceCompetitiveness(
   509→    business.inventories.map(inv => ({ productName: inv.productName, sellPrice: inv.sellPrice })),
   510→    productDefs.map(p => ({ name: p.name, basePrice: p.basePrice, suggestedMarkup: p.suggestedMarkup })),
   511→    marketPriceMap,
   512→  );
   513→
   514→  // Calculate service quality from employees
   515→  const idealEmployeeCount = 2 + business.level; // Base 2 + 1 per level
   516→  const cxServiceQuality = calculateServiceQuality(
   517→    business.employees.length,
   518→    avgEmployeeSkill,
   519→    idealEmployeeCount,
   520→  );
   521→
   522→  // Calculate product quality from margin (higher margin = better products)
   523→  const cxProductQuality = totalRevenue > 0
   524→    ? Math.min(1, Math.max(0, (totalRevenue - totalCOGS) / totalRevenue / 0.3))
   525→    : 0.5;
   526→
   527→  // Calculate satisfaction
   528→  const cxSatisfaction = calculateSatisfaction({
   529→    priceCompetitiveness: cxPriceCompetitiveness,
   530→    stockAvailability: maxStockCapacity > 0 ? totalStock / maxStockCapacity : 0,
   531→    serviceQuality: cxServiceQuality,
   532→    productQuality: cxProductQuality,
   533→    atmosphere: newReputation / 100,
   534→    currentSatisfaction: business.satisfactionScore,
   535→  });
   536→
   537→  // Determine if this was a positive experience
   538→  const wasPositiveExperience = cxSatisfaction.overall >= 50 && dailyProfit >= 0;
   539→
   540→  // Calculate loyalty
   541→  const cxLoyalty = calculateLoyalty({
   542→    currentLoyaltyScore: business.loyaltyScore,
   543→    currentRepeatRate: business.repeatCustomerRate,
   544→    wasPositiveExperience,
   545→    totalCustomers: potentialCustomers,
   546→    numberOfReviews: business.totalReviews,
   547→    averageRating: business.avgReviewRating,
   548→    loyaltyMultiplier: getLoyaltyTier(business.loyaltyScore).multiplier,
   549→  });
   550→
   551→  // ---- Step 7: Get game day for metrics ----
   552→  const gameDayState = await db.gameState.findUnique({ where: { key: 'gameDay' } });

   609→        // Phase 3: CX metrics
   610→        satisfaction: cxSatisfaction.overall,
   611→        loyalty: cxLoyalty.loyaltyScore,
   612→        nps: business.npsScore,
   613→      },
   614→      update: {
   615→        revenue: roundTaka(totalRevenue),
   616→        expenses: roundTaka(expenses.totalExpense),
   617→        profit: roundTaka(dailyProfit),
   618→        cogs: roundTaka(totalCOGS),
   619→        customers: potentialCustomers,
   620→        reputation: newReputation,
   621→        // Phase 3: CX metrics
   622→        satisfaction: cxSatisfaction.overall,
   623→        loyalty: cxLoyalty.loyaltyScore,
   624→        nps: business.npsScore,
   625→      },
   626→    });
   627→

   629→    const cogsStr = totalCOGS > 0 ? ` | COGS ৳${roundTaka(totalCOGS).toLocaleString()}` : '';
   630→    await tx.gameLog.create({

'Profit' : 'Loss'} ৳${Math.abs(roundTaka(dailyProfit)).toLocaleString()} | ${potentialCustomers} customers | Health ${healthResult.score} | Sat ${Math.round(cxSatisfaction.overall)} | Loy ${cxLoyalty.tier}`,
   636→        amount: dailyProfit,

   650→          comment: review.comment,

   656→    // Phase 3: Update NPS and review stats
   657→    if (cxReviews.length > 0) {
   658→      const allReviews = await tx.customerReview.findMany({
   659→        where: { businessId: business.id },
   660→        select: { rating: true },
   661→        orderBy: { createdAt: 'desc' },
   662→        take: REVIEW_CONFIG.keepLast,
   663→      });
   664→      const npsResult = calculateNPS(allReviews);
   665→      const totalReviews = allReviews.length;
   666→      const avgRating = totalReviews > 0
   667→        ? allReviews.reduce((sum, r) => sum + r.rating, 0) / totalReviews
   668→        : 0;
   669→
   670→      await tx.business.update({
   671→        where: { id: business.id },
   672→        data: {
   673→          npsScore: npsResult.nps,
   674→          totalReviews,
   675→          avgReviewRating: Math.round(avgRating * 100) / 100,
   676→        },
   677→      });
   678→
   679→      // Prune old reviews (keep only last N)
   680→      const reviewCount = await tx.customerReview.count({
   681→        where: { businessId: business.id },
   682→      });
   683→      if (reviewCount > REVIEW_CONFIG.keepLast) {
   684→        const oldReviews = await tx.customerReview.findMany({
   685→          where: { businessId: business.id },
   686→          orderBy: { createdAt: 'asc' },
   687→          take: reviewCount - REVIEW_CONFIG.keepLast,
   688→          select: { id: true },
   689→        });
   690→        for (const old of oldReviews) {
   691→          await tx.customerReview.delete({ where: { id: old.id } });
   692→        }
   693→      }
   694→    }
   695→
   696→    // Recalculate net worth within the same transaction
   697→    await recalculateNetWorth(tx, business.playerId);
   698→  });
   699→}

   701→// ---- Loan Payment Processing ----
   702→
   703→async function processLoanPayments(): Promise<void> {
   704→  const activeLoans = await db.loan.findMany({
   705→    where: { status: 'ACTIVE' },
   706→  });
   707→
   708→  if (activeLoans.length === 0) return;
   709→
   710→  for (const loan of activeLoans) {
   711→    const payment = loan.dailyPayment;
   712→    const isPaidOff = loan.daysRemaining - 1 <= 0;
   713→    const actualPayment = isPaidOff ? loan.remainingDebt : payment;
   714→
   715→    await db.$transaction(async (tx) => {
   716→      const player = await tx.player.findUnique({
   717→        where: { id: loan.playerId },
   718→        select: { cash: true },
   719→      });
   720→      if (!player) {
   721→        console.error(`[GameEngine] processLoanPayments: Player ${loan.playerId} not found for loan ${loan.id}`);
   722→        return;
   723→      }
   724→
   725→      const deductAmount = Math.min(actualPayment, Math.max(0, player.cash));
   726→      const newRemainingDebt = Math.max(loan.remainingDebt - deductAmount, 0);
   727→
   728→      await tx.loan.update({
   729→        where: { id: loan.id },
   730→        data: {
   731→          remainingDebt: newRemainingDebt,
   732→          daysRemaining: Math.max(loan.daysRemaining - 1, 0),
   733→          status: newRemainingDebt <= 0 ? 'PAID_OFF' : 'ACTIVE',
   734→        },
   735→      });
   736→
   737→      await tx.player.update({
   738→        where: { id: loan.playerId },
   739→        data: { cash: { decrement: deductAmount } },
   740→      });
   741→
   742→      await tx.gameLog.create({
   743→        data: {
   744→          playerId: loan.playerId,
   745→          type: newRemainingDebt <= 0 ? 'LOAN_PAID' : 'LOAN_PAYMENT',
   746→          message: newRemainingDebt <= 0
   747→            ? `Loan of ৳${loan.amount.toLocaleString()} fully repaid! Final deduction: ৳${Math.round(deductAmount).toLocaleString()}`
   748→            : `Loan payment: ৳${Math.round(deductAmount).toLocaleString()} deducted. Remaining debt: ৳${Math.round(newRemainingDebt).toLocaleString()} (${loan.daysRemaining - 1} days left)`,
   749→          amount: -deductAmount,
   750→        },
   751→      });
   752→    });
   753→  }
   754→}

   758→export async function gameTick(): Promise<void> {
   759→  // NOTE: The tick lock is managed by the API route handler.
   760→  // gameTick() is the pure simulation function and does NOT acquire/release the lock.

   770→  // 0. Game state housekeeping

   777→      where: { active: true, endsAt: { lt: new Date() } },
   778→      data: { active: false },

   783→    await tx.gameState.upsert({
   784→      where: { key: 'lastTick' },
   785→      create: { key: 'lastTick', value: now },

   789→      where: { key: 'tickCount' },
   790→      create: { key: 'tickCount', value: '1' },
   791→      update: { value: String(tickNum + 1) },
   792→    });
   793→    const dayState = await tx.gameState.findUnique({ where: { key: 'gameDay' } });
   794→    await tx.gameState.upsert({
   795→      where: { key: 'gameDay' },
   796→      create: { key: 'gameDay', value: '1' },
   797→      update: { value: String(parseInt(dayState?.value || '0') + 1) },
   798→    });
   799→  });
   800→
   801→  // Generate new events (20% chance)
   802→  if (Math.random() < 0.2) {
   803→    try {
   804→      await generateEvent();
   805→    } catch (err) {
   806→      console.error('[GameEngine] Failed to generate event:', err);
   807→    }
   808→  }
   809→
   810→  // Update market prices (every 3 ticks)
   811→  // Phase 1: This now uses retention-based updates with event effects integrated
   812→  if (tickNum % 3 === 0) {
   813→    try {
   814→      await updateMarketPrices();
   815→    } catch (err) {
   816→      console.error('[GameEngine] Failed to update market prices:', err);
   817→    }
   818→    // applyEventEffects() is now a no-op — events handled in updateMarketPrices()
   819→  }
   820→
   821→  // 1. Simulate all businesses
   822→  const businesses = await db.business.findMany({ select: { id: true } });
   823→  for (const business of businesses) {
   824→    try {
   825→      await simulateBusinessTick(business.id);
   826→    } catch (err) {
   827→      console.error(`[GameEngine] Failed to simulate business ${business.id}:`, err);
   828→    }
   829→  }
   830→
   831→  // 2. Process loan payments
   832→  try {
   833→    await processLoanPayments();
   834→  } catch (err) {
   835→    console.error('[GameEngine] Failed to process loan payments:', err);
   836→  }
   837→
   838→  // 3. Simulate AI player activity (Phase 2: Real AI decisions)
   839→  try {
   840→    const gameDayState2 = await db.gameState.findUnique({ where: { key: 'gameDay' } });
   841→    const currentGameDay = parseInt(gameDayState2?.value || '1', 10);
   842→    await simulateAIPlayersTick(currentGameDay);
   843→  } catch (err) {
   844→    console.error('[GameEngine] Failed to simulate AI tick:', err);
   845→  }
   846→
   847→  // 4. Generate news (50% chance)
   848→  if (Math.random() < 0.5) {
   849→    try {
   850→      await generateNews();
   851→    } catch (err) {
   852→      console.error('[GameEngine] Failed to generate news:', err);
   853→    }
   854→  }
   855→
   856→  // Log tick completion
   857→  await db.gameLog.create({
   858→    data: {
   859→      type: 'GAME_TICK_COMPLETED',
   860→      message: `Game tick completed at ${new Date().toISOString()}`,
   861→    },
   862→  });
   863→}
