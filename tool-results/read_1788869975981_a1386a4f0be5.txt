    65→    65→// ---- Tick Concurrency Lock Constants ----
    66→    66→const STALE_TICK_TIMEOUT_MS = 2 * 60 * 1000; // 2 minutes
    67→    67→
    68→    68→// ---- Tick Concurrency Lock ----
    69→    69→
    70→    70→/**
    71→    71→ * Atomically attempt to acquire the tick lock using the GameState key-value store.
    72→    72→ * Returns true if the lock was acquired, false if it was already held.
    73→    73→ *
    74→    74→ * Stale lock recovery: if the lock has been held for longer than
    75→    75→ * STALE_TICK_TIMEOUT_MS, the lock is stolen and a warning is logged.
    76→    76→ */
    77→    77→export async function acquireTickLock(): Promise<boolean> {
    78→    78→  const now = new Date();
    79→    79→
    80→    80→  return db.$transaction(async (tx) => {
    81→    81→    // Read current lock state
    82→    82→    const lockState = await tx.gameState.findUnique({ where: { key: 'tickInProgress' } });
    83→    83→    const startedAtState = await tx.gameState.findUnique({ where: { key: 'tickStartedAt' } });
    84→    84→
    85→    85→    const isLocked = lockState?.value === 'true';
    86→    86→    const startedAt = startedAtState?.value ? new Date(startedAtState.value) : null;
    87→    87→
    88→    88→    if (isLocked && startedAt) {
    89→    89→      const elapsed = now.getTime() - startedAt.getTime();
    90→    90→      if (elapsed < STALE_TICK_TIMEOUT_MS) {
    91→    91→        return false; // Lock is legitimately held
    92→    92→      }
    93→    93→      console.warn(`[GameEngine] Stale tick lock detected (held for ${Math.round(elapsed / 1000)}s). Stealing lock.`);
    94→    94→    } else if (isLocked) {
    95→    95→      console.warn('[GameEngine] Tick lock held with no timestamp. Stealing lock.');
    96→    96→    }
    97→    97→
    98→    98→    // Acquire the lock
    99→    99→    const nowISO = now.toISOString();
   100→   100→    await tx.gameState.upsert({
   101→   101→      where: { key: 'tickInProgress' },
   102→   102→      create: { key: 'tickInProgress', value: 'true' },
   103→   103→      update: { value: 'true' },
   104→   104→    });
   105→   105→    await tx.gameState.upsert({
   106→   106→      where: { key: 'tickStartedAt' },
   107→   107→      create: { key: 'tickStartedAt', value: nowISO },
   108→   108→      update: { value: nowISO },
   109→   109→    });
   110→   110→
   111→   111→    // Increment tick version
   112→   112→    const versionState = await tx.gameState.findUnique({ where: { key: 'tickVersion' } });
   113→   113→    const currentVersion = parseInt(versionState?.value || '0', 10);
   114→   114→    await tx.gameState.upsert({
   115→   115→      where: { key: 'tickVersion' },
   116→   116→      create: { key: 'tickVersion', value: '1' },
   117→   117→      update: { value: String(currentVersion + 1) },
   118→   118→    });
   119→   119→
   120→   120→    return true;
   121→   121→  });
   122→   122→}

   133→   133→async function recalculateNetWorth(tx: PrismaTx, playerId: string): Promise<void> {
   134→   134→  const player = await tx.player.findUnique({
   135→   135→    where: { id: playerId },
   136→   136→    select: { id: true, cash: true },
   137→   137→  });
   138→   138→  if (!player) {
   139→   139→    console.error(`[GameEngine] recalculateNetWorth: Player ${playerId} not found`);
   140→   140→    return;
   141→   141→  }
   142→   142→
   143→   143→  const allBusinesses = await tx.business.findMany({
   144→   144→    where: { playerId },
   145→   145→    select: { cash: true },
   146→   146→  });
   147→   147→  const totalBusinessCash = allBusinesses.reduce((sum, b) => sum + b.cash, 0);
   148→   148→
   149→   149→  const allInventory = await tx.inventory.findMany({
   150→   150→    where: { business: { playerId } },
   151→   151→    select: { quantity: true, purchasePrice: true },
   152→   152→  });
   153→   153→  const totalInventoryValue = allInventory.reduce((sum, inv) => sum + inv.quantity * inv.purchasePrice, 0);
   154→   154→
   155→   155→  const activeLoans = await tx.loan.findMany({
   156→   156→    where: { playerId, status: 'ACTIVE' },
   157→   157→    select: { remainingDebt: true },
   158→   158→  });
   159→   159→  const outstandingDebt = activeLoans.reduce((sum, loan) => sum + loan.remainingDebt, 0);
   160→   160→
   161→   161→  const netWorth = player.cash + totalBusinessCash + totalInventoryValue - outstandingDebt;
   162→   162→  await tx.player.update({ where: { id: playerId }, data: { netWorth } });
   163→   163→}

   166→   166→export async function generateEvent(): Promise<void> {
   167→   167→  const totalWeight = EVENT_TEMPLATES.reduce((sum, e) => sum + e.weight, 0);
   168→   168→  let random = Math.random() * totalWeight;
   169→   169→  let selected: EventTemplate | null = null;
   170→   170→  for (const event of EVENT_TEMPLATES) {
   171→   171→    random -= event.weight;
   172→   172→    if (random <= 0) {
   173→   173→      selected = event;
   174→   174→      break;
   175→   175→    }
   176→   176→  }
   177→   177→  if (!selected) selected = EVENT_TEMPLATES[0];
   178→   178→
   179→   179→  const now = new Date();
   180→   180→  const endAt = new Date(now.getTime() + selected.durationDays * 60 * 1000);
   181→   181→  await db.gameEvent.create({
   182→   182→    data: {
   183→   183→      title: selected.title,
   184→   184→      description: selected.description,
   185→   185→      type: selected.type,
   186→   186→      effects: JSON.stringify(selected.effects),
   187→   187→      active: true,
   188→   188→      startsAt: now,
   189→   189→      endsAt: endAt,
   190→   190→    },
   191→   191→  });
   192→   192→}

   200→   200→export async function updateMarketPrices(): Promise<void> {
   201→   201→  const allProducts = Object.values(PRODUCTS).flat();
   202→   202→  const cities = CITIES.map(c => c.id);
   203→   203→
   204→   204→  // Get active events for price modifiers
   205→   205→  const activeEvents = await db.gameEvent.findMany({ where: { active: true } });
   206→   206→  const categoryPriceMod: Record<string, number> = {};
   207→   207→  let allPriceMod = 0;
   208→   208→  for (const event of activeEvents) {
   209→   209→    const effects = JSON.parse(event.effects) as Record<string, number>;
   210→   210→    for (const [key, value] of Object.entries(effects)) {
   211→   211→      if (key === 'all_price' || key === 'all_imported') allPriceMod += value;
   212→   212→      if (key.endsWith('_price')) {
   213→   213→        const cat = key.replace('_price', '');
   214→   214→        categoryPriceMod[cat] = (categoryPriceMod[cat] || 0) + value;
   215→   215→      }
   216→   216→    }
   217→   217→  }
   218→   218→
   219→   219→  const updates: { productName: string; city: string; priceMultiplier: number; demandMultiplier: number }[] = [];
   220→   220→
   221→   221→  for (const city of cities) {
   222→   222→    for (const product of allProducts) {
   223→   223→      // Get current market price (for retention)
   224→   224→      const existing = await db.marketPrice.findUnique({
   225→   225→        where: { productName_city: { productName: product.name, city } },
   226→   226→      });
   227→   227→      const previousPriceMult = existing?.priceMultiplier ?? 1;
   228→   228→
   229→   229→      // Calculate event price modifier for this product
   230→   230→      const eventPriceMod = allPriceMod + (categoryPriceMod[product.category] || 0);
   231→   231→
   232→   232→      // Use Phase 1 retention-based update
   233→   233→      const { priceMultiplier, demandMultiplier } = calculateMarketPriceUpdate(
   234→   234→        previousPriceMult,
   235→   235→        eventPriceMod
   236→   236→      );
   237→   237→
   238→   238→      updates.push({ productName: product.name, city, priceMultiplier, demandMultiplier });
   239→   239→    }
   240→   240→  }
   241→   241→
   242→   242→  // Batch update in a transaction
   243→   243→  await db.$transaction(async (tx) => {
   244→   244→    for (const upd of updates) {
   245→   245→      await tx.marketPrice.upsert({
   246→   246→        where: { productName_city: { productName: upd.productName, city: upd.city } },
   247→   247→        create: upd,
   248→   248→        update: {
   249→   249→          priceMultiplier: upd.priceMultiplier,
   250→   250→          demandMultiplier: upd.demandMultiplier,
   251→   251→          updatedAt: new Date(),
   252→   252→        },
   253→   253→      });
   254→   254→    }
   255→   255→  });
   256→   256→}

   290→   290→export async function simulateBusinessTick(businessId: string): Promise<void> {
   291→   291→  const business = await db.business.findUnique({
   292→   292→    where: { id: businessId },
   293→   293→    include: {
   294→   294→      inventories: true,
   295→   295→      employees: true,
   296→   296→      player: { select: { id: true, cash: true, netWorth: true } },
   297→   297→    },
   298→   298→  });
   299→   299→
   300→   300→  if (!business) return;
   301→   301→
   302→   302→  const businessType = getBusinessType(business.type);
   303→   303→  const city = getCity(business.city);
   304→   304→  if (!businessType || !city) return;
   305→   305→
   306→   306→  const economyConfig = getBusinessEconomyConfig(business.type);
   307→   307→
   308→   308→  // Get active event effects (combined)
   309→   309→  const activeEvents = await db.gameEvent.findMany({
   310→   310→    where: { active: true },
   311→   311→    select: { effects: true },
   312→   312→  });
   313→   313→  const combinedEffects: Record<string, number> = {};
   314→   314→  for (const event of activeEvents) {
   315→   315→    const effects = JSON.parse(event.effects) as Record<string, number>;
   316→   316→    for (const [key, value] of Object.entries(effects)) {
   317→   317→      combinedEffects[key] = (combinedEffects[key] || 0) + value;
   318→   318→    }
   319→   319→  }
   320→   320→
   321→   321→  // Get market prices for this city
   322→   322→  const marketPrices = await db.marketPrice.findMany({
   323→   323→    where: { city: business.city },
   324→   324→  });
   325→   325→  const marketPriceMap: Record<string, { priceMultiplier: number; demandMultiplier: number }> = {};
   326→   326→  for (const mp of marketPrices) {
   327→   327→    marketPriceMap[mp.productName] = {
   328→   328→      priceMultiplier: mp.priceMultiplier,
   329→   329→      demandMultiplier: mp.demandMultiplier,
   330→   330→    };
   331→   331→  }
   332→   332→
   333→   333→  // ---- Step 0.5: CX Demand Modifier (from previous tick) ----
   334→   334→  // Use the business's existing satisfaction/loyalty/reputation to calculate
   335→   335→  // how CX affects THIS tick's customer count. This avoids circular dependency
   336→   336→  // because we use previous-tick CX values to influence current-tick demand.
   337→   337→  const previousCxDemandModifier = calculateCXDemandModifier(
   338→   338→    business.satisfactionScore,
   339→   339→    business.repeatCustomerRate,
   340→   340→    getLoyaltyTier(business.loyaltyScore).multiplier,
   341→   341→  );
   342→   342→
   343→   343→  // Calculate segment demand modifier from previous-tick data
   344→   344→  // Segments react differently to quality, service, reputation
   345→   345→  // NOTE: priceCompetitiveness is NOT passed to segment demands because price is already
   346→   346→  // handled by per-product priceDemandMultiplier AND satisfaction's priceCompetitiveness.
   347→   347→  // Passing it here would triple-count price effects.
   348→   348→  // Similarly, reputation is already in calculatePotentialCustomers Layer 3 AND satisfaction's
   349→   349→  // atmosphere factor, so we use a moderate value to avoid quadruple-counting.
   350→   350→  const previousCxServiceQuality = calculateServiceQuality(
   351→   351→    business.employees.length,
   352→   352→    business.employees.length > 0
   353→   353→      ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   354→   354→      : 0,
   355→   355→    2 + business.level,
   356→   356→  );
   357→   357→  const segmentDemands = calculateSegmentDemands({
   358→   358→    priceCompetitiveness: 1.0,  // Price already handled by priceDemandMultiplier + satisfaction
   359→   359→    productQuality: Math.max(0, Math.min(1, business.healthScore / 100)), // Use health as quality proxy from previous tick
   360→   360→    serviceQuality: previousCxServiceQuality,
   361→   361→    reputation: Math.min(100, business.reputation * 0.5 + 50),  // Dampened: avg 50 ± 25 from reputation
   362→   362→  });
   363→   363→  // Aggregate segment demand: sum of demand multipliers (NOT weighted by share again)
   364→   364→  // demandMultiplier already includes baseShare, so we just sum them
   365→   365→  const segmentDemandModifier = segmentDemands.reduce(
   366→   366→    (sum, seg) => sum + seg.demandMultiplier, 0
   367→   367→  );
   368→   368→
   369→   369→  // ---- Step 1: Calculate potential customers ----
   370→   370→  const totalStock = business.inventories.reduce((sum, inv) => sum + inv.quantity, 0);
   371→   371→  // Max stock capacity: sum of product maxStock values for this business type
   372→   372→  const productDefs = PRODUCTS[business.type] || [];
   373→   373→  const maxStockCapacity = productDefs.reduce((sum, p) => sum + p.maxStock, 0);
   374→   374→  const avgEmployeeSkill = business.employees.length > 0
   375→   375→    ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   376→   376→    : 0;
   377→   377→
   378→   378→  const basePotentialCustomers = calculatePotentialCustomers({
   379→   379→    baseCustomers: businessType.baseCustomers,
   380→   380→    cityMultiplier: city.customerMultiplier,
   381→   381→    level: business.level,
   382→   382→    reputation: business.reputation,
   383→   383→    totalStock,
   384→   384→    maxStockCapacity,
   385→   385→    employeeCount: business.employees.length,
   386→   386→    avgEmployeeSkill,
   387→   387→    eventCustomerEffect: combinedEffects['all_customers'] || 0,
   388→   388→    businessDemandEffect: combinedEffects[`${businessType.id}_demand`] || 0,
   389→   389→    businessTypeId: business.type,
   390→   390→  });
   391→   391→
   392→   392→  // Apply CX demand modifier and segment demand modifier
   393→   393→  // CX modifier: satisfaction/loyalty affect how many customers visit (0.2-2.0 range)
   394→   394→  // Segment modifier: different customer segments find the business more/less attractive
   395→   395→  const potentialCustomers = Math.max(0, Math.floor(
   396→   396→    basePotentialCustomers * previousCxDemandModifier * segmentDemandModifier
   397→   397→  ));
   398→   398→
   399→   399→  // ---- Step 2: Product-level sales simulation ----
   400→   400→  let totalRevenue = 0;
   401→   401→  let totalCOGS = 0;
   402→   402→  const inventoryUpdates: { id: string; quantityDecrement: number }[] = [];
   403→   403→  const salesResults: ProductSalesResult[] = [];
   404→   404→
   405→   405→  for (const inventory of business.inventories) {
   406→   406→    if (inventory.quantity <= 0) continue;
   407→   407→
   408→   408→    const productDef = productDefs.find(p => p.name === inventory.productName);
   409→   409→    if (!productDef) continue;
   410→   410→
   411→   411→    // Product demand config
   412→   412→    const prodDemandConfig = getProductDemandConfig(inventory.productName);
   413→   413→
   414→   414→    // Market reference price = typical retail price (basePrice × (1 + suggestedMarkup) × priceMultiplier)
   415→   415→    // This represents what the MARKET typically charges, not the wholesale cost.
   416→   416→    // Players compare their price to market retail price, not to their own cost.
   417→   417→    const mp = marketPriceMap[inventory.productName];
   418→   418→    const typicalRetailPrice = productDef.basePrice * (1 + productDef.suggestedMarkup);
   419→   419→    const marketReferencePrice = Math.round(
   420→   420→      typicalRetailPrice * (mp?.priceMultiplier || 1)
   421→   421→    );
   422→   422→
   423→   423→    // Market demand multiplier from events
   424→   424→    const demandMult = mp?.demandMultiplier || 1;
   425→   425→
   426→   426→    // Event demand effect for this product's category
   427→   427→    const eventDemandEffect = combinedEffects[`${productDef.category}_demand`] || 0;
   428→   428→
   429→   429→    // Run the complete sales pipeline
   430→   430→    const salesResult = simulateProductSales({
   431→   431→      inventoryId: inventory.id,
   432→   432→      productName: inventory.productName,
   433→   433→      quantity: inventory.quantity,
   434→   434→      purchasePrice: inventory.purchasePrice,
   435→   435→      sellPrice: inventory.sellPrice,
   436→   436→      marketReferencePrice,
   437→   437→      baseDemand: prodDemandConfig.baseDemand,
   438→   438→      demandMultiplier: demandMult,
   439→   439→      eventDemandEffect,
   440→   440→      productPriceSensitivity: prodDemandConfig.priceSensitivity,
   441→   441→      productVolatility: prodDemandConfig.volatility,
   442→   442→      businessPriceSensitivity: economyConfig.priceSensitivity,
   443→   443→      potentialCustomers,
   444→   444→    });
   445→   445→
   446→   446→    if (salesResult.itemsSold > 0) {
   447→   447→      totalRevenue += salesResult.revenue;
   448→   448→      totalCOGS += salesResult.costOfGoodsSold;
   449→   449→      inventoryUpdates.push({
   450→   450→        id: inventory.id,
   451→   451→        quantityDecrement: salesResult.itemsSold,
   452→   452→      });
   453→   453→    }
   454→   454→
   455→   455→    salesResults.push(salesResult);
   456→   456→  }
   457→   457→
   458→   458→  // ---- Step 3: Calculate expenses ----
   459→   459→  const totalMonthlySalaries = business.employees.reduce((sum, emp) => sum + emp.salary, 0);

   696→   696→    // Recalculate net worth within the same transaction

   703→   703→async function processLoanPayments(): Promise<void> {
   704→   704→  const activeLoans = await db.loan.findMany({
   705→   705→    where: { status: 'ACTIVE' },
   706→   706→  });
   707→   707→
   708→   708→  if (activeLoans.length === 0) return;
   709→   709→
   710→   710→  for (const loan of activeLoans) {
   711→   711→    const payment = loan.dailyPayment;
   712→   712→    const isPaidOff = loan.daysRemaining - 1 <= 0;
   713→   713→    const actualPayment = isPaidOff ? loan.remainingDebt : payment;
   714→   714→
   715→   715→    await db.$transaction(async (tx) => {
   716→   716→      const player = await tx.player.findUnique({
   717→   717→        where: { id: loan.playerId },
   718→   718→        select: { cash: true },
   719→   719→      });
   720→   720→      if (!player) {
   721→   721→        console.error(`[GameEngine] processLoanPayments: Player ${loan.playerId} not found for loan ${loan.id}`);
   722→   722→        return;
   723→   723→      }
   724→   724→
   725→   725→      const deductAmount = Math.min(actualPayment, Math.max(0, player.cash));
   726→   726→      const newRemainingDebt = Math.max(loan.remainingDebt - deductAmount, 0);
   727→   727→
   728→   728→      await tx.loan.update({
   729→   729→        where: { id: loan.id },
   730→   730→        data: {
   731→   731→          remainingDebt: newRemainingDebt,
   732→   732→          daysRemaining: Math.max(loan.daysRemaining - 1, 0),
   733→   733→          status: newRemainingDebt <= 0 ? 'PAID_OFF' : 'ACTIVE',
   734→   734→        },
   735→   735→      });
   736→   736→
   737→   737→      await tx.player.update({
   738→   738→        where: { id: loan.playerId },
   739→   739→        data: { cash: { decrement: deductAmount } },
   740→   740→      });
   741→   741→
   742→   742→      await tx.gameLog.create({
   743→   743→        data: {
   744→   744→          playerId: loan.playerId,
   745→   745→          type: newRemainingDebt <= 0 ? 'LOAN_PAID' : 'LOAN_PAYMENT',
   746→   746→          message: newRemainingDebt <= 0
   747→   747→            ? `Loan of ৳${loan.amount.toLocaleString()} fully repaid! Final deduction: ৳${Math.round(deductAmount).toLocaleString()}`
   748→   748→            : `Loan payment: ৳${Math.round(deductAmount).toLocaleString()} deducted. Remaining debt: ৳${Math.round(newRemainingDebt).toLocaleString()} (${loan.daysRemaining - 1} days left)`,
   749→   749→          amount: -deductAmount,
   750→   750→        },
   751→   751→      });
   752→   752→    });
   753→   753→  }
   754→   754→}
