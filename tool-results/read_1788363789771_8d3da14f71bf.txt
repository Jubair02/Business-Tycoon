   146→          productId: buyProduct.id || buyProduct.productId,

   165→    }

   172→      const res = await fetch(`/api/businesses/${currentBusiness.id}/employees/hire`, {
   173→        method: 'POST',
   174→        headers: { 'Content-Type': 'application/json' },

   187→      toast.error('Hiring failed');

   193→  const handleFire = async (empId: string) => {

   205→      toast.error('Failed to fire employee');
   206→    } finally {
   207→      setFiringId(null);
   208→    }
   209→  };
   210→
   211→  const handleUpgrade = async () => {

   213→      toast.error('Not enough cash!');

   218→      const res = await fetch(`/api/businesses/${currentBusiness.id}/upgrade`, { method: 'POST' });

   243→  };
   244→
   245→  return (
   246→    <div className="pb-24 md:pb-4">
   247→      <Tabs defaultValue="overview" className="w-full">

   249→      <div className="sticky top-14 z-40 bg-white border-b">
   250→        <div className={`px-3 md:px-4 py-3 md:py-4 game-hero-banner ${bannerClass}`}>

   252→            <Button variant="ghost" size="icon" className="shrink-0 hover:bg-white/50" onClick={() => setView('businesses')}>
   253→              <ArrowLeft className="h-4 w-4" />
   254→            </Button>
   255→            <div className="flex-1 min-w-0">

   257→                <span className="text-2xl game-float">{bt?.icon}</span>
   258→                <div className="min-w-0">
   259→                  <h2 className="font-bold text-base truncate">{currentBusiness.name}</h2>
   260→                  <div className="flex items-center gap-2 mt-0.5">
   261→                    <Badge variant="outline" className={`text-[10px] shrink-0 ${bannerClass}`}>{bt?.name}</Badge>
   262→                    <span className="text-[10px] opacity-80 flex items-center gap-1">📍 {city?.name}</span>
   263→                  </div>
   264→                </div>
   265→              </div>
   266→            </div>
   267→            <Badge className="text-[10px] shrink-0 text-white font-bold" style={{ background: 'linear-gradient(135deg, #006a4e, #00a86b)' }}>Lv.{currentBusiness.level}</Badge>
   268→          </div>
   269→        </div>
   270→        {/* Tabs inside the sticky block */}
   271→        <div className="px-3 md:px-4 pt-2">
   272→          <TabsList className="w-full grid grid-cols-5">
   273→            <TabsTrigger value="overview" className="text-xs">Overview</TabsTrigger>
   274→            <TabsTrigger value="inventory" className="text-xs">Inventory</TabsTrigger>
   275→            <TabsTrigger value="employees" className="text-xs">Staff</TabsTrigger>
   276→            <TabsTrigger value="logs" className="text-xs">Log</TabsTrigger>

   280→      </div>
   281→
   282→      <div className="p-3 md:p-4">
   283→
   284→          <TabsContent value="overview">
   285→            <div className="grid grid-cols-2 gap-3 mb-4">
   286→              <Card className="border-l-4 border-l-green-500 game-stat-card game-shine">
   287→                <CardContent className="p-3">
   288→                  <div className="text-[10px] text-muted-foreground uppercase" style={{ letterSpacing: '0.08em' }}>Daily Revenue</div>
   289→                  <div className="text-lg font-bold text-green-600 mt-0.5">{formatTakaShort(revenue)}</div>
   290→                </CardContent>
   291→              </Card>
   292→              <Card className={`${profit >= 0 ? 'border-l-4 border-l-green-500' : 'border-l-4 border-l-red-500'} game-stat-card game-shine`}>
   293→                <CardContent className="p-3">
   294→                  <div className="text-[10px] text-muted-foreground uppercase" style={{ letterSpacing: '0.08em' }}>Daily Profit</div>
   295→                  <div className={`text-lg font-bold mt-0.5 ${profit >= 0 ? 'text-green-600' : 'text-red-500'}`}>
   296→                    {profit >= 0 ? '+' : ''}{formatTakaShort(profit)}
   297→                  </div>
   298→                </CardContent>
   299→              </Card>
   300→              <Card className="border-l-4 border-l-amber-500 game-stat-card game-shine">
   301→                <CardContent className="p-3">
   302→                  <div className="text-[10px] text-muted-foreground uppercase" style={{ letterSpacing: '0.08em' }}>Daily Expenses</div>
   303→                  <div className="text-lg font-bold text-amber-600 mt-0.5">{formatTakaShort(expenses)}</div>
   304→                </CardContent>
   305→              </Card>
   306→              <Card className="border-l-4 border-l-purple-500 game-stat-card game-shine">
   307→                <CardContent className="p-3">
   308→                  <div className="text-[10px] text-muted-foreground uppercase" style={{ letterSpacing: '0.08em' }}>Reputation</div>
   309→                  <div className="text-lg font-bold mt-0.5">{currentBusiness.reputation || 0}%</div>
   310→                </CardContent>
   311→              </Card>
   312→            </div>
   313→
   314→            {/* Profit Breakdown */}
   315→            <Card className="mb-4 game-gradient-card game-shine">

   318→              </CardHeader>
   319→              <CardContent className="px-4 pb-4 space-y-2">
   320→                <div className="flex items-center justify-between text-sm">
   321→                  <div className="flex items-center gap-2">
   322→                    <TrendingUp className="h-4 w-4 text-green-600" />
   323→                    <span className="text-muted-foreground">Revenue</span>
   324→                  </div>
   325→                  <span className="font-medium text-green-600">+{formatTakaShort(revenue)}</span>
   326→                </div>
   327→                <div className="flex items-center justify-between text-sm">
   328→                  <div className="flex items-center gap-2">
   329→                    <TrendingDown className="h-4 w-4 text-red-500" />
   330→                    <span className="text-muted-foreground">Expenses</span>
   331→                  </div>
   332→                  <span className="font-medium text-red-500">-{formatTakaShort(expenses)}</span>
   333→                </div>
   334→                <Separator />
   335→                <div className="flex items-center justify-between text-sm font-bold">
   336→                  <span>Net Profit</span>
   337→                  <span className={profit >= 0 ? 'text-green-600' : 'text-red-500'}>
   338→                    {profit >= 0 ? '+' : ''}{formatTakaShort(profit)}
   339→                  </span>
   340→                </div>
   341→              </CardContent>
   342→            </Card>
   343→
   344→            <Card className="mb-4 game-shine">
   345→              <CardHeader className="pb-2 pt-4 px-4">
   346→                <CardTitle className="text-sm font-semibold game-gradient-text">Reputation</CardTitle>
   347→              </CardHeader>
   348→              <CardContent className="px-4 pb-4">
   349→                <Progress value={currentBusiness.reputation || 0} className="h-3" />
   350→                <div className="flex justify-between mt-1.5 text-[10px] text-muted-foreground">
   351→                  <span>Poor</span><span>Average</span><span>Excellent</span>
   352→                </div>
   353→              </CardContent>
   354→            </Card>
   355→
   356→            {/* Business Performance mini chart (placeholder) */}
   357→            <Card className="game-shine">
   358→              <CardHeader className="pb-2 pt-4 px-4">
   359→                <CardTitle className="text-sm font-semibold game-gradient-text">Business Performance</CardTitle>
   360→              </CardHeader>
   361→              <CardContent className="px-4 pb-4">
   362→                <div className="h-24 flex items-end gap-1">
   363→                  {[...Array(14)].map((_, i) => {
   364→                    const h = 20 + Math.sin((i + 3) * 0.8) * 30 + Math.random() * 15;
   365→                    const isPositive = h > 35;
   366→                    return (
   367→                      <motion.div
   368→                        key={i}
   369→                        className="flex-1 rounded-t"
   370→                        style={{
   371→                          height: `${h}%`,
   372→                          background: isPositive ? 'rgba(0, 106, 78, 0.6)' : 'rgba(244, 42, 65, 0.5)',
   373→                        }}
   374→                        initial={{ height: 0 }}
   375→                        animate={{ height: `${h}%` }}
   376→                        transition={{ delay: i * 0.03, duration: 0.3 }}
   377→                      />
   378→                    );
   379→                  })}
   380→                </div>
   381→                <div className="text-[10px] text-muted-foreground text-center mt-2">Last 14 days</div>
   382→              </CardContent>
   383→            </Card>
   384→          </TabsContent>
   385→
   386→          <TabsContent value="inventory">
   387→            <div className="space-y-3">
   388→              <div className="flex items-center justify-between">

   390→                <Dialog open={showBuyDialog} onOpenChange={setShowBuyDialog}>
   391→                  <Button size="sm" className="gap-1 text-white text-xs" style={{ background: '#006a4e' }} onClick={() => { if (marketProducts.length === 0) { fetchMarketProducts(); } }} asChild>
   392→                    <DialogTrigger asChild>
   393→                      <button><ShoppingCart className="h-3.5 w-3.5" /> Buy Stock</button>
   394→                    </DialogTrigger>
   395→                  </Button>
   396→                  <DialogContent className="max-w-md">
   397→                    <DialogHeader>
   398→                      <DialogTitle>Buy Inventory</DialogTitle>
   399→                      <DialogDescription>Choose a product and quantity to buy</DialogDescription>
   400→                    </DialogHeader>
   401→                    {!buyProduct ? (
   402→                      <ScrollArea className="max-h-64">
   403→                        <div className="space-y-2">

   405→                            <button key={p.id || p.name} onClick={() => setBuyProduct(p)} className="w-full text-left p-3 rounded-lg border hover:border-green-400 transition-all hover:shadow-md hover:shadow-green-100 hover:-translate-y-0.5">
   406→                              <div className="flex items-center gap-2">
   407→                                <span className="text-lg">{p.icon}</span>
   408→                                <div className="flex-1 min-w-0">
   409→                                  <div className="text-sm font-medium">{p.name}</div>
   410→                                  <div className="text-xs text-muted-foreground">Market: ৳{p.currentPrice?.toLocaleString() || p.basePrice?.toLocaleString()}</div>
   411→                                </div>
   412→                                <Badge className={`text-[10px] border ${getDemandColor(p.currentDemand || 1)}`} variant="outline">
   413→                                  {getDemandLabel(p.currentDemand || 1)}
   414→                                </Badge>
   415→                              </div>
   416→                            </button>
   417→                          ))}
   418→                        </div>
   419→                      </ScrollArea>
   420→                    ) : (
   421→                      <div className="space-y-4">
   422→                        <div className="flex items-center gap-2 p-3 rounded-lg bg-green-50/60 border border-green-100">

   424→                          <div>
   425→                            <div className="font-medium">{buyProduct.name}</div>
   426→                            <div className="text-sm text-muted-foreground">Unit price: ৳{(buyProduct.currentPrice || buyProduct.basePrice || 0).toLocaleString()}</div>

   432→                          <div className="text-sm text-muted-foreground">

   434→                          </div>
   435→                        </div>
   436→                      </div>

   438→                    <DialogFooter className="gap-2">
   439→                      {buyProduct && (

   442→                      <Button onClick={buyProduct ? handleBuy : () => setShowBuyDialog(false)} disabled={buying || (!buyProduct)} className="text-white game-shine" style={{ background: 'linear-gradient(135deg, #006a4e 0%, #00895e 60%, #00a86b 100%)' }}>
   443→                        {buying ? 'Buying...' : buyProduct ? `Buy for ${formatTaka((buyProduct.currentPrice || buyProduct.basePrice || 0) * buyQuantity)}` : 'Close'}
   444→                      </Button>

   447→                </Dialog>
   448→              </div>
   449→
   450→              {inventories.length === 0 ? (
   451→                <Card className="border-dashed">

   456→                </Card>
   457→              ) : (
   458→                <div className="space-y-1.5 rounded-lg overflow-hidden border border-green-100/60">
   459→                  {inventories.map((inv: any) => {
   460→                    const mp = marketProducts.find((p: any) => p.id === inv.productId || p.name === inv.productName);
   461→                    return (
   462→                      <div key={inv.id} className="inventory-row flex items-center gap-3 p-3">
   463→                        <span className="text-xl">{mp?.icon || inv.productName?.charAt(0)}</span>

   466→                          <div className="flex items-center gap-2 mt-0.5">
