    94→  if (!currentBusiness) {

   102→

   146→          productId: buyProduct.id || buyProduct.productId,
   147→          productName: buyProduct.name,

   165→    }

   172→      const res = await fetch(`/api/businesses/${currentBusiness.id}/employees/hire`, {
   173→        method: 'POST',
   174→        headers: { 'Content-Type': 'application/json' },
   175→        body: JSON.stringify({ role: hireRole }),

   189→      setHiring(false);

   205→      toast.error('Failed to fire employee');
   206→    } finally {
   207→      setFiringId(null);
   208→    }
   209→  };
   210→
   211→  const handleUpgrade = async () => {

   213→      toast.error('Not enough cash!');

   218→      const res = await fetch(`/api/businesses/${currentBusiness.id}/upgrade`, { method: 'POST' });

   232→

   243→  };
   244→
   245→  return (
   246→    <div className="pb-24 md:pb-4">
   247→      {/* Enhanced hero banner header */}
   248→      <div className={`sticky top-14 z-40 border-b px-3 md:px-4 py-4 md:py-5 game-hero-banner ${bannerClass}`}>

   251→            <ArrowLeft className="h-4 w-4" />
   252→          </Button>
   253→          <div className="flex-1 min-w-0">
   254→            <div className="flex items-center gap-2.5">
   255→              <span className="text-2xl game-float">{bt?.icon}</span>
   256→              <div className="min-w-0">
   257→                <h2 className="font-bold text-base truncate">{currentBusiness.name}</h2>
   258→                <div className="flex items-center gap-2 mt-0.5">
   259→                  <Badge variant="outline" className={`text-[10px] shrink-0 ${bannerClass}`}>{bt?.name}</Badge>
   260→                  <span className="text-[10px] opacity-80 flex items-center gap-1">📍 {city?.name}</span>
   261→                </div>
   262→              </div>
   263→            </div>
   264→          </div>
   265→          <Badge className="text-[10px] shrink-0 text-white font-bold" style={{ background: 'linear-gradient(135deg, #006a4e, #00a86b)' }}>Lv.{currentBusiness.level}</Badge>
   266→        </div>
   267→      </div>
   268→
   269→      <div className="p-3 md:p-4">
   270→        <Tabs defaultValue="overview" className="w-full">
   271→          <TabsList className="w-full grid grid-cols-5 mb-4">
   272→            <TabsTrigger value="overview" className="text-xs">Overview</TabsTrigger>
   273→            <TabsTrigger value="inventory" className="text-xs">Inventory</TabsTrigger>
   274→            <TabsTrigger value="employees" className="text-xs">Staff</TabsTrigger>

   279→          <TabsContent value="overview">
   280→            <div className="grid grid-cols-2 gap-3 mb-4">
   281→              <Card className="border-l-4 border-l-green-500 game-stat-card game-shine">

   283→                  <div className="text-[10px] text-muted-foreground uppercase" style={{ letterSpacing: '0.08em' }}>Daily Revenue</div>
   284→                  <div className="text-lg font-bold text-green-600 mt-0.5">{formatTakaShort(revenue)}</div>

   286→              </Card>
   287→              <Card className={`${profit >= 0 ? 'border-l-4 border-l-green-500' : 'border-l-4 border-l-red-500'} game-stat-card game-shine`}>
   288→                <CardContent className="p-3">
   289→                  <div className="text-[10px] text-muted-foreground uppercase" style={{ letterSpacing: '0.08em' }}>Daily Profit</div>
   290→                  <div className={`text-lg font-bold mt-0.5 ${profit >= 0 ? 'text-green-600' : 'text-red-500'}`}>
   291→                    {profit >= 0 ? '+' : ''}{formatTakaShort(profit)}
   292→                  </div>
   293→                </CardContent>
   294→              </Card>
   295→              <Card className="border-l-4 border-l-amber-500 game-stat-card game-shine">

   297→                  <div className="text-[10px] text-muted-foreground uppercase" style={{ letterSpacing: '0.08em' }}>Daily Expenses</div>
   298→                  <div className="text-lg font-bold text-amber-600 mt-0.5">{formatTakaShort(expenses)}</div>
   299→                </CardContent>
   300→              </Card>
   301→              <Card className="border-l-4 border-l-purple-500 game-stat-card game-shine">
   302→                <CardContent className="p-3">
   303→                  <div className="text-[10px] text-muted-foreground uppercase" style={{ letterSpacing: '0.08em' }}>Reputation</div>

   305→                </CardContent>
   306→              </Card>
   307→            </div>
   308→
   309→            {/* Profit Breakdown */}
   310→            <Card className="mb-4 game-gradient-card game-shine">

   312→                <CardTitle className="text-sm font-semibold game-gradient-text">Profit Breakdown</CardTitle>
   313→              </CardHeader>
   314→              <CardContent className="px-4 pb-4 space-y-2">
   315→                <div className="flex items-center justify-between text-sm">
   316→                  <div className="flex items-center gap-2">
   317→                    <TrendingUp className="h-4 w-4 text-green-600" />
   318→                    <span className="text-muted-foreground">Revenue</span>
   319→                  </div>
   320→                  <span className="font-medium text-green-600">+{formatTakaShort(revenue)}</span>
   321→                </div>
   322→                <div className="flex items-center justify-between text-sm">
   323→                  <div className="flex items-center gap-2">
   324→                    <TrendingDown className="h-4 w-4 text-red-500" />
   325→                    <span className="text-muted-foreground">Expenses</span>
   326→                  </div>
   327→                  <span className="font-medium text-red-500">-{formatTakaShort(expenses)}</span>
   328→                </div>
   329→                <Separator />
   330→                <div className="flex items-center justify-between text-sm font-bold">
   331→                  <span>Net Profit</span>
   332→                  <span className={profit >= 0 ? 'text-green-600' : 'text-red-500'}>
   333→                    {profit >= 0 ? '+' : ''}{formatTakaShort(profit)}
   334→                  </span>
   335→                </div>
   336→              </CardContent>
   337→            </Card>
   338→
   339→            <Card className="mb-4 game-shine">
   340→              <CardHeader className="pb-2 pt-4 px-4">
   341→                <CardTitle className="text-sm font-semibold game-gradient-text">Reputation</CardTitle>
   342→              </CardHeader>
   343→              <CardContent className="px-4 pb-4">
   344→                <Progress value={currentBusiness.reputation || 0} className="h-3" />
   345→                <div className="flex justify-between mt-1.5 text-[10px] text-muted-foreground">
   346→                  <span>Poor</span><span>Average</span><span>Excellent</span>
   347→                </div>
   348→              </CardContent>
   349→            </Card>
   350→
   351→            {/* Business Performance mini chart (placeholder) */}
   352→            <Card className="game-shine">
   353→              <CardHeader className="pb-2 pt-4 px-4">
   354→                <CardTitle className="text-sm font-semibold game-gradient-text">Business Performance</CardTitle>
   355→              </CardHeader>
   356→              <CardContent className="px-4 pb-4">
   357→                <div className="h-24 flex items-end gap-1">
   358→                  {[...Array(14)].map((_, i) => {
   359→                    const h = 20 + Math.sin((i + 3) * 0.8) * 30 + Math.random() * 15;
   360→                    const isPositive = h > 35;
   361→                    return (
   362→                      <motion.div
   363→                        key={i}
   364→                        className="flex-1 rounded-t"
   365→                        style={{
   366→                          height: `${h}%`,
   367→                          background: isPositive ? 'rgba(0, 106, 78, 0.6)' : 'rgba(244, 42, 65, 0.5)',
   368→                        }}
   369→                        initial={{ height: 0 }}
   370→                        animate={{ height: `${h}%` }}
   371→                        transition={{ delay: i * 0.03, duration: 0.3 }}
   372→                      />
   373→                    );
   374→                  })}
   375→                </div>
   376→                <div className="text-[10px] text-muted-foreground text-center mt-2">Last 14 days</div>
   377→              </CardContent>
   378→            </Card>
   379→          </TabsContent>
   380→
   381→          <TabsContent value="inventory">
   382→            <div className="space-y-3">
   383→              <div className="flex items-center justify-between">
   384→                <h3 className="text-sm font-semibold">Inventory ({inventories.length})</h3>
   385→                <Dialog open={showBuyDialog} onOpenChange={setShowBuyDialog}>
   386→                  <Button size="sm" className="gap-1 text-white text-xs" style={{ background: '#006a4e' }} onClick={() => { if (marketProducts.length === 0) { fetchMarketProducts(); } }} asChild>
   387→                    <DialogTrigger asChild>
   388→                      <button><ShoppingCart className="h-3.5 w-3.5" /> Buy Stock</button>
   389→                    </DialogTrigger>
   390→                  </Button>
   391→                  <DialogContent className="max-w-md">
   392→                    <DialogHeader>
   393→                      <DialogTitle>Buy Inventory</DialogTitle>
   394→                      <DialogDescription>Choose a product and quantity to buy</DialogDescription>
   395→                    </DialogHeader>
   396→                    {!buyProduct ? (
   397→                      <ScrollArea className="max-h-64">
   398→                        <div className="space-y-2">
   399→                          {marketProducts.map((p: any) => (
   400→                            <button key={p.id || p.name} onClick={() => setBuyProduct(p)} className="w-full text-left p-3 rounded-lg border hover:border-green-400 transition-all hover:shadow-md hover:shadow-green-100 hover:-translate-y-0.5">
   401→                              <div className="flex items-center gap-2">
   402→                                <span className="text-lg">{p.icon}</span>
   403→                                <div className="flex-1 min-w-0">
   404→                                  <div className="text-sm font-medium">{p.name}</div>

   406→                                </div>
   407→                                <Badge className={`text-[10px] border ${getDemandColor(p.currentDemand || 1)}`} variant="outline">
   408→                                  {getDemandLabel(p.currentDemand || 1)}
   409→                                </Badge>
   410→                              </div>
   411→                            </button>

   414→                      </ScrollArea>

   416→                      <div className="space-y-4">
   417→                        <div className="flex items-center gap-2 p-3 rounded-lg bg-green-50/60 border border-green-100">

   419→                          <div>
   420→                            <div className="font-medium">{buyProduct.name}</div>

   422→                          </div>
   423→                        </div>

   427→                          <div className="text-sm text-muted-foreground">

   433→                    <DialogFooter className="gap-2">

   437→                      <Button onClick={buyProduct ? handleBuy : () => setShowBuyDialog(false)} disabled={buying || (!buyProduct)} className="text-white game-shine" style={{ background: 'linear-gradient(135deg, #006a4e 0%, #00895e 60%, #00a86b 100%)' }}>
   438→                        {buying ? 'Buying...' : buyProduct ? `Buy for ${formatTaka((buyProduct.currentPrice || buyProduct.basePrice || 0) * buyQuantity)}` : 'Close'}

   441→                  </DialogContent>
   442→                </Dialog>

   444→
   445→              {inventories.length === 0 ? (
   446→                <Card className="border-dashed">
   447→                  <CardContent className="py-8 text-center">

   454→                  {inventories.map((inv: any) => {

   468→                        <div className="text-right shrink-0">
