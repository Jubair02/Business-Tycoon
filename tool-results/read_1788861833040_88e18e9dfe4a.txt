   163→describe('AI Strategy - calculateAIPrice', () => {
   164→  const marketPrice = 100;
   165→
   166→  it('returns different values for different strategies', () => {
   167→    // Run multiple times since there's randomness
   168→    const results = new Set<number>();
   169→    const strategies: AIPricingStrategy[] = ['LOW_PRICE', 'MARKET_PRICE', 'PREMIUM', 'DYNAMIC'];
   170→    for (const strategy of strategies) {
   171→      // Run a few times to account for randomness
   172→      for (let i = 0; i < 5; i++) {
   173→        results.add(calculateAIPrice(marketPrice, 'BALANCED', strategy, 70, 0.5));
   174→      }
   175→    }
   176→    // With 4 different strategies and randomness, we should get multiple distinct values
   177→    expect(results.size).toBeGreaterThan(1);
   178→  });
   179→
   180→  it('LOW_PRICE strategy gives prices below market', () => {
   181→    // LOW_PRICE multiplier is 0.85 + random*0.13 → range 0.85-0.98
   182→    // With CONSERVATIVE personality modifier (*0.98), effective range: ~0.83-0.96
   183→    for (let i = 0; i < 20; i++) {
   184→      const price = calculateAIPrice(marketPrice, 'CONSERVATIVE', 'LOW_PRICE', 70, 0.5);
   185→      // Should be at or below market price (with small margin for rounding)
   186→      expect(price).toBeLessThanOrEqual(marketPrice);
   187→    }
   188→  });
   189→
   190→  it('PREMIUM strategy gives prices above market', () => {
   191→    // PREMIUM multiplier is 1.05 + random*0.15 → range 1.05-1.20
   192→    // With AGGRESSIVE personality modifier (*1.02), effective range: ~1.07-1.22
   193→    for (let i = 0; i < 20; i++) {
   194→      const price = calculateAIPrice(marketPrice, 'AGGRESSIVE', 'PREMIUM', 70, 0.5);
   195→      expect(price).toBeGreaterThanOrEqual(marketPrice);
   196→    }
   197→  });
   198→
   199→  it('MARKET_PRICE strategy gives prices near market', () => {
   200→    // MARKET_PRICE multiplier is 0.98 + random*0.07 → range 0.98-1.05
   201→    for (let i = 0; i < 20; i++) {
   202→      const price = calculateAIPrice(marketPrice, 'BALANCED', 'MARKET_PRICE', 70, 0.5);
   203→      // Should be within ~7% of market price
   204→      expect(price).toBeGreaterThanOrEqual(marketPrice * 0.95);
   205→      expect(price).toBeLessThanOrEqual(marketPrice * 1.1);
   206→    }
   207→  });
   208→
   209→  it('DYNAMIC strategy reacts to low health by cutting prices', () => {
   210→    let lowHealthTotal = 0;
   211→    let normalHealthTotal = 0;
   212→    const iterations = 30;
   213→    for (let i = 0; i < iterations; i++) {
   214→      lowHealthTotal += calculateAIPrice(marketPrice, 'BALANCED', 'DYNAMIC', 15, 0.5);
   215→      normalHealthTotal += calculateAIPrice(marketPrice, 'BALANCED', 'DYNAMIC', 80, 0.5);
   216→    }
   217→    // Low health should produce lower average prices than normal health
   218→    expect(lowHealthTotal / iterations).toBeLessThan(normalHealthTotal / iterations);
   219→  });
   220→
   221→  it('DYNAMIC strategy reacts to overstock by discounting', () => {
   222→    let overstockTotal = 0;
   223→    let normalStockTotal = 0;
   224→    const iterations = 30;
   225→    for (let i = 0; i < iterations; i++) {
   226→      overstockTotal += calculateAIPrice(marketPrice, 'BALANCED', 'DYNAMIC', 70, 0.9); // stockRatio > 0.8
   227→      normalStockTotal += calculateAIPrice(marketPrice, 'BALANCED', 'DYNAMIC', 70, 0.5);
   228→    }
   229→    // Overstocked should produce slightly lower average prices
   230→    expect(overstockTotal / iterations).toBeLessThanOrEqual(normalStockTotal / iterations);
   231→  });

   234→    let lowStockTotal = 0;
   235→    let normalStockTotal = 0;
   236→    const iterations = 30;
   237→    for (let i = 0; i < iterations; i++) {
   238→      lowStockTotal += calculateAIPrice(marketPrice, 'BALANCED', 'DYNAMIC', 70, 0.1); // stockRatio < 0.2
   239→      normalStockTotal += calculateAIPrice(marketPrice, 'BALANCED', 'DYNAMIC', 70, 0.5);

   618→// (since we can't call the async DB-dependent function)

   620→interface MarketShareInput {
   621→  reputation: number;      // 0-100
   622→  level: number;           // 1-10
   623→  healthScore: number;     // 0-100
   624→  totalStock: number;
   625→  maxStockCapacity: number;
   626→  avgSellPrice: number;
   627→  dailyRevenue: number;
   628→}
   629→
   630→function calculateAttractiveness(
   631→  biz: MarketShareInput,
   632→  avgMarketPrice: number,
   633→  avgRevenue: number,
   634→): number {
   635→  const repFactor = 0.4 + (biz.reputation / 100) * 1.2;
   636→  const levelFactor = 1 + (biz.level - 1) * 0.12;
   637→  const healthFactor = 0.5 + (biz.healthScore / 100);
   638→  const stockRatio = biz.maxStockCapacity > 0 ? biz.totalStock / biz.maxStockCapacity : 0;
   639→  const inventoryFactor = 0.3 + stockRatio * 0.7;
   640→  const pricingFactor = Math.max(0.5, Math.min(1.5, avgMarketPrice / Math.max(biz.avgSellPrice, 1)));
   641→  const revenueFactor = 0.5 + Math.min(biz.dailyRevenue / Math.max(avgRevenue, 1), 2.0) * 0.25;
   642→  return repFactor * levelFactor * healthFactor * inventoryFactor * pricingFactor * revenueFactor;
   643→}
   644→
   645→function computeMarketShares(businesses: MarketShareInput[]): number[] {
   646→  if (businesses.length === 0) return [];
   647→
   648→  const avgMarketPrice = businesses.reduce((s, b) => s + b.avgSellPrice, 0) / businesses.length;
   649→  const avgRevenue = businesses.reduce((s, b) => s + b.dailyRevenue, 0) / businesses.length;
   650→
   651→  const scores = businesses.map(b => calculateAttractiveness(b, avgMarketPrice, avgRevenue));
   652→  const totalScore = scores.reduce((s, sc) => s + sc, 0);
   653→
   654→  return totalScore > 0 ? scores.map(sc => sc / totalScore) : scores.map(() => 0);
   655→}
