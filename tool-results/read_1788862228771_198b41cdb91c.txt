    52→const STALE_TICK_TIMEOUT_MS = 2 * 60 * 1000; // 2 minutes
    53→
    54→// ---- Tick Concurrency Lock ----
    55→
    56→/**
    57→ * Atomically attempt to acquire the tick lock using the GameState key-value store.
    58→ * Returns true if the lock was acquired, false if it was already held.
    59→ *
    60→ * Stale lock recovery: if the lock has been held for longer than
    61→ * STALE_TICK_TIMEOUT_MS, the lock is stolen and a warning is logged.
    62→ */
    63→export async function acquireTickLock(): Promise<boolean> {
    64→  const now = new Date();
    65→
    66→  return db.$transaction(async (tx) => {
    67→    // Read current lock state
    68→    const lockState = await tx.gameState.findUnique({ where: { key: 'tickInProgress' } });
    69→    const startedAtState = await tx.gameState.findUnique({ where: { key: 'tickStartedAt' } });
    70→
    71→    const isLocked = lockState?.value === 'true';
    72→    const startedAt = startedAtState?.value ? new Date(startedAtState.value) : null;
    73→
    74→    if (isLocked && startedAt) {
    75→      const elapsed = now.getTime() - startedAt.getTime();
    76→      if (elapsed < STALE_TICK_TIMEOUT_MS) {
    77→        return false; // Lock is legitimately held
    78→      }
    79→      console.warn(`[GameEngine] Stale tick lock detected (held for ${Math.round(elapsed / 1000)}s). Stealing lock.`);
    80→    } else if (isLocked) {
    81→      console.warn('[GameEngine] Tick lock held with no timestamp. Stealing lock.');
    82→    }
    83→
    84→    // Acquire the lock
    85→    const nowISO = now.toISOString();
    86→    await tx.gameState.upsert({
    87→      where: { key: 'tickInProgress' },
    88→      create: { key: 'tickInProgress', value: 'true' },
    89→      update: { value: 'true' },
    90→    });
    91→    await tx.gameState.upsert({
    92→      where: { key: 'tickStartedAt' },
    93→      create: { key: 'tickStartedAt', value: nowISO },
    94→      update: { value: nowISO },
    95→    });
    96→
    97→    // Increment tick version
    98→    const versionState = await tx.gameState.findUnique({ where: { key: 'tickVersion' } });
    99→    const currentVersion = parseInt(versionState?.value || '0', 10);
   100→    await tx.gameState.upsert({
   101→      where: { key: 'tickVersion' },
   102→      create: { key: 'tickVersion', value: '1' },
   103→      update: { value: String(currentVersion + 1) },
   104→    });
   105→
   106→    return true;
   107→  });
   108→}

   118→// ---- Net Worth Recalculation ----
   119→async function recalculateNetWorth(tx: PrismaTx, playerId: string): Promise<void> {
   120→  const player = await tx.player.findUnique({
   121→    where: { id: playerId },
   122→    select: { id: true, cash: true },
   123→  });
   124→  if (!player) {
   125→    console.error(`[GameEngine] recalculateNetWorth: Player ${playerId} not found`);
   126→    return;
   127→  }
   128→
   129→  const allBusinesses = await tx.business.findMany({
   130→    where: { playerId },
   131→    select: { cash: true },
   132→  });
   133→  const totalBusinessCash = allBusinesses.reduce((sum, b) => sum + b.cash, 0);
   134→
   135→  const allInventory = await tx.inventory.findMany({
   136→    where: { business: { playerId } },
   137→    select: { quantity: true, purchasePrice: true },
   138→  });
   139→  const totalInventoryValue = allInventory.reduce((sum, inv) => sum + inv.quantity * inv.purchasePrice, 0);
   140→
   141→  const activeLoans = await tx.loan.findMany({
   142→    where: { playerId, status: 'ACTIVE' },
   143→    select: { remainingDebt: true },
   144→  });
   145→  const outstandingDebt = activeLoans.reduce((sum, loan) => sum + loan.remainingDebt, 0);
   146→
   147→  const netWorth = player.cash + totalBusinessCash + totalInventoryValue - outstandingDebt;
   148→  await tx.player.update({ where: { id: playerId }, data: { netWorth } });
   149→}

   152→export async function generateEvent(): Promise<void> {
   153→  const totalWeight = EVENT_TEMPLATES.reduce((sum, e) => sum + e.weight, 0);
   154→  let random = Math.random() * totalWeight;
   155→  let selected: EventTemplate | null = null;
   156→  for (const event of EVENT_TEMPLATES) {
   157→    random -= event.weight;
   158→    if (random <= 0) {
   159→      selected = event;
   160→      break;
   161→    }
   162→  }
   163→  if (!selected) selected = EVENT_TEMPLATES[0];
   164→
   165→  const now = new Date();
   166→  const endAt = new Date(now.getTime() + selected.durationDays * 60 * 1000);
   167→  await db.gameEvent.create({
   168→    data: {
   169→      title: selected.title,
   170→      description: selected.description,
   171→      type: selected.type,
   172→      effects: JSON.stringify(selected.effects),
   173→      active: true,
   174→      startsAt: now,
   175→      endsAt: endAt,
   176→    },
   177→  });
   178→}

   186→export async function updateMarketPrices(): Promise<void> {
   187→  const allProducts = Object.values(PRODUCTS).flat();
   188→  const cities = CITIES.map(c => c.id);
   189→
   190→  // Get active events for price modifiers
   191→  const activeEvents = await db.gameEvent.findMany({ where: { active: true } });
   192→  const categoryPriceMod: Record<string, number> = {};
   193→  let allPriceMod = 0;
   194→  for (const event of activeEvents) {
   195→    const effects = JSON.parse(event.effects) as Record<string, number>;
   196→    for (const [key, value] of Object.entries(effects)) {
   197→      if (key === 'all_price' || key === 'all_imported') allPriceMod += value;
   198→      if (key.endsWith('_price')) {
   199→        const cat = key.replace('_price', '');
   200→        categoryPriceMod[cat] = (categoryPriceMod[cat] || 0) + value;
   201→      }
   202→    }
   203→  }
   204→
   205→  const updates: { productName: string; city: string; priceMultiplier: number; demandMultiplier: number }[] = [];
   206→
   207→  for (const city of cities) {
   208→    for (const product of allProducts) {
   209→      // Get current market price (for retention)
   210→      const existing = await db.marketPrice.findUnique({
   211→        where: { productName_city: { productName: product.name, city } },
   212→      });
   213→      const previousPriceMult = existing?.priceMultiplier ?? 1;
   214→
   215→      // Calculate event price modifier for this product
   216→      const eventPriceMod = allPriceMod + (categoryPriceMod[product.category] || 0);
   217→
   218→      // Use Phase 1 retention-based update
   219→      const { priceMultiplier, demandMultiplier } = calculateMarketPriceUpdate(
   220→        previousPriceMult,
   221→        eventPriceMod
   222→      );
   223→
   224→      updates.push({ productName: product.name, city, priceMultiplier, demandMultiplier });
   225→    }
   226→  }
   227→
   228→  // Batch update in a transaction
   229→  await db.$transaction(async (tx) => {
   230→    for (const upd of updates) {
   231→      await tx.marketPrice.upsert({
   232→        where: { productName_city: { productName: upd.productName, city: upd.city } },
   233→        create: upd,
   234→        update: {
   235→          priceMultiplier: upd.priceMultiplier,
   236→          demandMultiplier: upd.demandMultiplier,
   237→          updatedAt: new Date(),
   238→        },
   239→      });
   240→    }
   241→  });
   242→}

   276→export async function simulateBusinessTick(businessId: string): Promise<void> {
   277→  const business = await db.business.findUnique({
   278→    where: { id: businessId },
   279→    include: {
   280→      inventories: true,
   281→      employees: true,
   282→      player: { select: { id: true, cash: true, netWorth: true } },
   283→    },
   284→  });
   285→
   286→  if (!business) return;
   287→
   288→  const businessType = getBusinessType(business.type);
   289→  const city = getCity(business.city);
   290→  if (!businessType || !city) return;
   291→
   292→  const economyConfig = getBusinessEconomyConfig(business.type);
   293→
   294→  // Get active event effects (combined)
   295→  const activeEvents = await db.gameEvent.findMany({
   296→    where: { active: true },
   297→    select: { effects: true },
   298→  });
   299→  const combinedEffects: Record<string, number> = {};
   300→  for (const event of activeEvents) {
   301→    const effects = JSON.parse(event.effects) as Record<string, number>;
   302→    for (const [key, value] of Object.entries(effects)) {
   303→      combinedEffects[key] = (combinedEffects[key] || 0) + value;
   304→    }
   305→  }
   306→
   307→  // Get market prices for this city
   308→  const marketPrices = await db.marketPrice.findMany({
   309→    where: { city: business.city },
   310→  });
   311→  const marketPriceMap: Record<string, { priceMultiplier: number; demandMultiplier: number }> = {};
   312→  for (const mp of marketPrices) {
   313→    marketPriceMap[mp.productName] = {
   314→      priceMultiplier: mp.priceMultiplier,
   315→      demandMultiplier: mp.demandMultiplier,
   316→    };
   317→  }
   318→
   319→  // ---- Step 1: Calculate potential customers ----
   320→  const totalStock = business.inventories.reduce((sum, inv) => sum + inv.quantity, 0);
   321→  // Max stock capacity: sum of product maxStock values for this business type
   322→  const productDefs = PRODUCTS[business.type] || [];
   323→  const maxStockCapacity = productDefs.reduce((sum, p) => sum + p.maxStock, 0);
   324→  const avgEmployeeSkill = business.employees.length > 0
   325→    ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   326→    : 0;
   327→
   328→  const potentialCustomers = calculatePotentialCustomers({
   329→    baseCustomers: businessType.baseCustomers,
   330→    cityMultiplier: city.customerMultiplier,
   331→    level: business.level,
   332→    reputation: business.reputation,
   333→    totalStock,
   334→    maxStockCapacity,
   335→    employeeCount: business.employees.length,
   336→    avgEmployeeSkill,
   337→    eventCustomerEffect: combinedEffects['all_customers'] || 0,
   338→    businessDemandEffect: combinedEffects[`${businessType.id}_demand`] || 0,
   339→    businessTypeId: business.type,
   340→  });
   341→
   342→  // ---- Step 2: Product-level sales simulation ----
   343→  let totalRevenue = 0;
   344→  let totalCOGS = 0;
   345→  const inventoryUpdates: { id: string; quantityDecrement: number }[] = [];
   346→  const salesResults: ProductSalesResult[] = [];
   347→
   348→  for (const inventory of business.inventories) {
   349→    if (inventory.quantity <= 0) continue;
   350→
   351→    const productDef = productDefs.find(p => p.name === inventory.productName);
   352→    if (!productDef) continue;
   353→
   354→    // Product demand config
   355→    const prodDemandConfig = getProductDemandConfig(inventory.productName);
   356→
   357→    // Market reference price = typical retail price (basePrice × (1 + suggestedMarkup) × priceMultiplier)
   358→    // This represents what the MARKET typically charges, not the wholesale cost.
   359→    // Players compare their price to market retail price, not to their own cost.
   360→    const mp = marketPriceMap[inventory.productName];
   361→    const typicalRetailPrice = productDef.basePrice * (1 + productDef.suggestedMarkup);
   362→    const marketReferencePrice = Math.round(
   363→      typicalRetailPrice * (mp?.priceMultiplier || 1)
   364→    );
   365→
   366→    // Market demand multiplier from events
   367→    const demandMult = mp?.demandMultiplier || 1;
   368→
   369→    // Event demand effect for this product's category
   370→    const eventDemandEffect = combinedEffects[`${productDef.category}_demand`] || 0;
   371→
   372→    // Run the complete sales pipeline
   373→    const salesResult = simulateProductSales({
   374→      inventoryId: inventory.id,
   375→      productName: inventory.productName,
   376→      quantity: inventory.quantity,
   377→      purchasePrice: inventory.purchasePrice,
   378→      sellPrice: inventory.sellPrice,
   379→      marketReferencePrice,
   380→      baseDemand: prodDemandConfig.baseDemand,
   381→      demandMultiplier: demandMult,
   382→      eventDemandEffect,
   383→      productPriceSensitivity: prodDemandConfig.priceSensitivity,
   384→      productVolatility: prodDemandConfig.volatility,
   385→      businessPriceSensitivity: economyConfig.priceSensitivity,
   386→      potentialCustomers,
   387→    });
   388→
   389→    if (salesResult.itemsSold > 0) {
   390→      totalRevenue += salesResult.revenue;
   391→      totalCOGS += salesResult.costOfGoodsSold;
   392→      inventoryUpdates.push({
   393→        id: inventory.id,
   394→        quantityDecrement: salesResult.itemsSold,
   395→      });
   396→    }
   397→
   398→    salesResults.push(salesResult);
   399→  }
   400→
   401→  // ---- Step 3: Calculate expenses ----
   402→  const totalMonthlySalaries = business.employees.reduce((sum, emp) => sum + emp.salary, 0);
   403→  const grossProfit = totalRevenue - totalCOGS;
   404→
   405→  const expenses = calculateBusinessExpenses({
   406→    baseRent: businessType.rent,
   407→    level: business.level,
   408→    cityRentMultiplier: city.rentMultiplier,
   409→    totalMonthlySalaries,
   410→    businessLevel: business.level,
   411→    businessTypeId: business.type,
   412→    revenue: totalRevenue,
   413→    grossProfit,
   414→  });
   415→
   416→  // ---- Step 4: Calculate net profit ----
   417→  const dailyProfit = calculateNetProfit(totalRevenue, totalCOGS, expenses.totalExpense);
   418→
   419→  // ---- Step 5: Calculate reputation change ----
   420→  const outOfStockRatio = business.inventories.length > 0
   421→    ? business.inventories.filter(inv => inv.quantity <= 0).length / business.inventories.length
   422→    : 0;
   423→  const managers = business.employees.filter(e => e.role === 'MANAGER');
   424→  const cleaners = business.employees.filter(e => e.role === 'CLEANER');
   425→
   426→  const reputationChange = calculateReputationChange({
   427→    dailyProfit,
   428→    outOfStockRatio,
   429→    managers: managers.map(m => ({ skill: m.skill })),
   430→    cleaners: cleaners.map(c => ({ skill: c.skill })),
   431→  });
   432→
   433→  const newReputation = Math.max(
   434→    ECONOMY_CONFIG.minReputation,
   435→    Math.min(ECONOMY_CONFIG.maxReputation, business.reputation + reputationChange)
   436→  );
   437→
   438→  // ---- Step 6: Calculate business health score ----
   439→  const healthResult = calculateBusinessHealth({
   440→    dailyProfit,
   441→    dailyRevenue: totalRevenue,
   442→    businessCash: business.cash + dailyProfit,
   443→    totalStock,
   444→    maxStockCapacity,
   445→    reputation: newReputation,
   446→    businessTypeId: business.type,
   447→  });
   448→
   449→  // ---- Step 7: Get game day for metrics ----
   450→  const gameDayState = await db.gameState.findUnique({ where: { key: 'gameDay' } });
   451→  const gameDay = parseInt(gameDayState?.value || '1', 10);
   452→
   453→  // ---- Step 8: Execute all updates in a single transaction ----
   454→  await db.$transaction(async (tx) => {
   455→    // Update inventory quantities
   456→    for (const upd of inventoryUpdates) {
   457→      await tx.inventory.update({
   458→        where: { id: upd.id },
   459→        data: { quantity: { decrement: upd.quantityDecrement } },
   460→      });
   461→    }
   462→
   463→    // Phase 1 FIX: Distribute business profit to player cash
   464→    // Previously, profit accumulated in business.cash but was never accessible.
   465→    // Now: business profit is transferred to the player each tick.
   466→    const profitToDistribute = dailyProfit;
   467→
   468→    // Update business
   469→    await tx.business.update({
   470→      where: { id: business.id },
   471→      data: {
   472→        dailyRevenue: roundTaka(totalRevenue),
   473→        dailyExpense: roundTaka(expenses.totalExpense),
   474→        dailyProfit: roundTaka(dailyProfit),
   475→        dailyCOGS: roundTaka(totalCOGS),
   476→        dailyCustomers: potentialCustomers,
   477→        cash: business.cash + roundTaka(dailyProfit),
   478→        totalRevenue: business.totalRevenue + roundTaka(totalRevenue),
   479→        totalProfit: business.totalProfit + roundTaka(dailyProfit),
   480→        reputation: newReputation,
   481→        healthScore: healthResult.score,
   482→      },
   483→    });
   484→
   485→    // Distribute profit to player cash
   486→    await tx.player.update({
   487→      where: { id: business.playerId },
   488→      data: { cash: { increment: roundTaka(profitToDistribute) } },
   489→    });
   490→
   491→    // Save daily metrics for analytics
   492→    await tx.businessMetric.upsert({
   493→      where: { businessId_gameDay: { businessId: business.id, gameDay } },
   494→      create: {
   495→        businessId: business.id,
   496→        gameDay,
   497→        revenue: roundTaka(totalRevenue),
   498→        expenses: roundTaka(expenses.totalExpense),
   499→        profit: roundTaka(dailyProfit),
   500→        cogs: roundTaka(totalCOGS),
   501→        customers: potentialCustomers,
   502→        reputation: newReputation,
   503→      },
   504→      update: {
   505→        revenue: roundTaka(totalRevenue),
   506→        expenses: roundTaka(expenses.totalExpense),
   507→        profit: roundTaka(dailyProfit),
   508→        cogs: roundTaka(totalCOGS),
   509→        customers: potentialCustomers,
   510→        reputation: newReputation,
   511→      },
   512→    });
   513→
   514→    // Log
   515→    const cogsStr = totalCOGS > 0 ? ` | COGS ৳${roundTaka(totalCOGS).toLocaleString()}` : '';
   516→    await tx.gameLog.create({
   517→      data: {
   518→        playerId: business.playerId,
   519→        businessId: business.id,
   520→        type: dailyProfit >= 0 ? 'PROFIT' : 'LOSS',
   521→        message: `${business.name}: Rev ৳${roundTaka(totalRevenue).toLocaleString()}${cogsStr} | Exp ৳${roundTaka(expenses.totalExpense).toLocaleString()} | ${dailyProfit >= 0 ? 'Profit' : 'Loss'} ৳${Math.abs(roundTaka(dailyProfit)).toLocaleString()} | ${potentialCustomers} customers | Health ${healthResult.score}`,
   522→        amount: dailyProfit,
   523→      },
   524→    });
   525→
   526→    // Recalculate net worth within the same transaction
   527→    await recalculateNetWorth(tx, business.playerId);
   528→  });
   529→}

   533→async function processLoanPayments(): Promise<void> {
   534→  const activeLoans = await db.loan.findMany({
   535→    where: { status: 'ACTIVE' },
   536→  });
   537→
   538→  if (activeLoans.length === 0) return;
   539→
   540→  for (const loan of activeLoans) {
   541→    const payment = loan.dailyPayment;
   542→    const isPaidOff = loan.daysRemaining - 1 <= 0;
   543→    const actualPayment = isPaidOff ? loan.remainingDebt : payment;
   544→
   545→    await db.$transaction(async (tx) => {
   546→      const player = await tx.player.findUnique({
   547→        where: { id: loan.playerId },
   548→        select: { cash: true },
   549→      });
   550→      if (!player) {
   551→        console.error(`[GameEngine] processLoanPayments: Player ${loan.playerId} not found for loan ${loan.id}`);
   552→        return;
   553→      }
   554→
   555→      const deductAmount = Math.min(actualPayment, Math.max(0, player.cash));
   556→      const newRemainingDebt = Math.max(loan.remainingDebt - deductAmount, 0);
   557→
   558→      await tx.loan.update({
   559→        where: { id: loan.id },
   560→        data: {
   561→          remainingDebt: newRemainingDebt,
   562→          daysRemaining: Math.max(loan.daysRemaining - 1, 0),
   563→          status: newRemainingDebt <= 0 ? 'PAID_OFF' : 'ACTIVE',
   564→        },
   565→      });
   566→
   567→      await tx.player.update({
   568→        where: { id: loan.playerId },
   569→        data: { cash: { decrement: deductAmount } },
   570→      });
   571→
   572→      await tx.gameLog.create({
   573→        data: {
   574→          playerId: loan.playerId,
   575→          type: newRemainingDebt <= 0 ? 'LOAN_PAID' : 'LOAN_PAYMENT',
   576→          message: newRemainingDebt <= 0
   577→            ? `Loan of ৳${loan.amount.toLocaleString()} fully repaid! Final deduction: ৳${Math.round(deductAmount).toLocaleString()}`
   578→            : `Loan payment: ৳${Math.round(deductAmount).toLocaleString()} deducted. Remaining debt: ৳${Math.round(newRemainingDebt).toLocaleString()} (${loan.daysRemaining - 1} days left)`,
   579→          amount: -deductAmount,
   580→        },
   581→      });
   582→    });
   583→  }
   584→}

   588→export async function gameTick(): Promise<void> {
   589→  // NOTE: The tick lock is managed by the API route handler.
   590→  // gameTick() is the pure simulation function and does NOT acquire/release the lock.
   591→
   592→  // Log tick start
   593→  await db.gameLog.create({
   594→    data: {
   595→      type: 'GAME_TICK_STARTED',
   596→      message: `Game tick started at ${new Date().toISOString()}`,
   597→    },
   598→  });
   599→
   600→  // 0. Game state housekeeping
   601→  const tickState = await db.gameState.findUnique({ where: { key: 'tickCount' } });
   602→  const tickNum = parseInt(tickState?.value || '0');
   603→
   604→  await db.$transaction(async (tx) => {
   605→    // Expire old events
   606→    await tx.gameEvent.updateMany({
   607→      where: { active: true, endsAt: { lt: new Date() } },
   608→      data: { active: false },
   609→    });
   610→
   611→    // Update game state
   612→    const now = new Date().toISOString();
   613→    await tx.gameState.upsert({
   614→      where: { key: 'lastTick' },
   615→      create: { key: 'lastTick', value: now },
   616→      update: { value: now },
   617→    });
   618→    await tx.gameState.upsert({
   619→      where: { key: 'tickCount' },
   620→      create: { key: 'tickCount', value: '1' },
   621→      update: { value: String(tickNum + 1) },
   622→    });
   623→    const dayState = await tx.gameState.findUnique({ where: { key: 'gameDay' } });
   624→    await tx.gameState.upsert({
   625→      where: { key: 'gameDay' },
   626→      create: { key: 'gameDay', value: '1' },
   627→      update: { value: String(parseInt(dayState?.value || '0') + 1) },
   628→    });
   629→  });
   630→
   631→  // Generate new events (20% chance)
   632→  if (Math.random() < 0.2) {
   633→    try {
   634→      await generateEvent();
   635→    } catch (err) {
   636→      console.error('[GameEngine] Failed to generate event:', err);
   637→    }
   638→  }
   639→
   640→  // Update market prices (every 3 ticks)
   641→  // Phase 1: This now uses retention-based updates with event effects integrated
   642→  if (tickNum % 3 === 0) {
   643→    try {
   644→      await updateMarketPrices();
   645→    } catch (err) {
   646→      console.error('[GameEngine] Failed to update market prices:', err);
   647→    }
   648→    // applyEventEffects() is now a no-op — events handled in updateMarketPrices()
   649→  }
   650→
   651→  // 1. Simulate all businesses
   652→  const businesses = await db.business.findMany({ select: { id: true } });
   653→  for (const business of businesses) {
   654→    try {
   655→      await simulateBusinessTick(business.id);
   656→    } catch (err) {
   657→      console.error(`[GameEngine] Failed to simulate business ${business.id}:`, err);
   658→    }
   659→  }
   660→
   661→  // 2. Process loan payments
   662→  try {
   663→    await processLoanPayments();
   664→  } catch (err) {
   665→    console.error('[GameEngine] Failed to process loan payments:', err);
   666→  }
   667→
   668→  // 3. Simulate AI player activity (Phase 2: Real AI decisions)
   669→  try {
   670→    const gameDayState2 = await db.gameState.findUnique({ where: { key: 'gameDay' } });
   671→    const currentGameDay = parseInt(gameDayState2?.value || '1', 10);
   672→    await simulateAIPlayersTick(currentGameDay);
   673→  } catch (err) {
   674→    console.error('[GameEngine] Failed to simulate AI tick:', err);
   675→  }
   676→
   677→  // 4. Generate news (50% chance)
   678→  if (Math.random() < 0.5) {
   679→    try {
   680→      await generateNews();
   681→    } catch (err) {
   682→      console.error('[GameEngine] Failed to generate news:', err);
   683→    }
   684→  }
   685→
   686→  // Log tick completion
   687→  await db.gameLog.create({
   688→    data: {
   689→      type: 'GAME_TICK_COMPLETED',
   690→      message: `Game tick completed at ${new Date().toISOString()}`,
   691→    },
   692→  });
   693→}

   697→async function generateNews(): Promise<void> {

   707→  const randomBusiness = BUSINESS_TYPES[Math.floor(Math.random() * BUSINESS_TYPES.length)].name;

   709→  const trend = Math.random() > 0.5 ? 'strong growth' : 'slight decline';
   710→  const weatherOptions = ['heavy rain', 'clear skies', 'heatwave', 'mild weather', 'thunderstorms'];
   711→  const weather = weatherOptions[Math.floor(Math.random() * weatherOptions.length)];

   714→  const title = template.title
   715→    .replace('{city}', randomCity)
   716→    .replace('{business_type}', randomBusiness)
   717→    .replace('{direction}', direction)
   718→    .replace('{trend}', trend)
   719→    .replace('{weather}', weather);
   720→
   721→  const contents: Record<string, string> = {
   722→    MARKET: `Sources report that commodity prices have ${direction} in ${randomCity}. Traders are advising businesses to adjust their inventory strategies accordingly.`,
   723→    BUSINESS: `The ${randomBusiness} sector is currently showing ${trend} according to industry analysts. Business owners should monitor market trends closely.`,

   725→    WEATHER: `${weather.charAt(0).toUpperCase() + weather.slice(1)} conditions are expected in the ${randomCity} region. Businesses should prepare for potential changes in customer footfall.`,
   726→    TRADE: `International trade reports indicate that import costs have ${direction}. This may affect pricing strategies for businesses dealing with imported goods.`,
   727→  };
   728→
   729→  await db.newsArticle.create({
   730→    data: {
   731→      title,
   732→      content: contents[template.category] || contents['ECONOMY'],
   733→      category: template.category,
   734→    },
   735→  });
   736→
   737→  // Keep only last 50 news articles
   738→  const totalNews = await db.newsArticle.count();
   739→  if (totalNews > 50) {
   740→    const oldNews = await db.newsArticle.findMany({
   741→      orderBy: { createdAt: 'asc' },
   742→      take: totalNews - 50,
   743→      select: { id: true },
   744→    });
   745→    const ids = oldNews.map(n => n.id);
   746→    await db.newsArticle.deleteMany({ where: { id: { in: ids } } });
   747→  }
