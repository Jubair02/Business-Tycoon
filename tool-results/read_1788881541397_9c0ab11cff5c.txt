    81→// ---- Tick Concurrency Lock ----
    82→
    83→/**
    84→ * Atomically attempt to acquire the tick lock using the GameState key-value store.
    85→ * Returns true if the lock was acquired, false if it was already held.
    86→ *
    87→ * Stale lock recovery: if the lock has been held for longer than
    88→ * STALE_TICK_TIMEOUT_MS, the lock is stolen and a warning is logged.
    89→ */
    90→export async function acquireTickLock(): Promise<boolean> {
    91→  const now = new Date();
    92→
    93→  return db.$transaction(async (tx) => {
    94→    // Read current lock state
    95→    const lockState = await tx.gameState.findUnique({ where: { key: 'tickInProgress' } });
    96→    const startedAtState = await tx.gameState.findUnique({ where: { key: 'tickStartedAt' } });
    97→
    98→    const isLocked = lockState?.value === 'true';
    99→    const startedAt = startedAtState?.value ? new Date(startedAtState.value) : null;
   100→
   101→    if (isLocked && startedAt) {
   102→      const elapsed = now.getTime() - startedAt.getTime();
   103→      if (elapsed < STALE_TICK_TIMEOUT_MS) {
   104→        return false; // Lock is legitimately held
   105→      }
   106→      console.warn(`[GameEngine] Stale tick lock detected (held for ${Math.round(elapsed / 1000)}s). Stealing lock.`);
   107→    } else if (isLocked) {
   108→      console.warn('[GameEngine] Tick lock held with no timestamp. Stealing lock.');
   109→    }
   110→
   111→    // Acquire the lock
   112→    const nowISO = now.toISOString();
   113→    await tx.gameState.upsert({
   114→      where: { key: 'tickInProgress' },
   115→      create: { key: 'tickInProgress', value: 'true' },
   116→      update: { value: 'true' },
   117→    });
   118→    await tx.gameState.upsert({
   119→      where: { key: 'tickStartedAt' },
   120→      create: { key: 'tickStartedAt', value: nowISO },
   121→      update: { value: nowISO },
   122→    });
   123→
   124→    // Increment tick version
   125→    const versionState = await tx.gameState.findUnique({ where: { key: 'tickVersion' } });
   126→    const currentVersion = parseInt(versionState?.value || '0', 10);
   127→    await tx.gameState.upsert({
   128→      where: { key: 'tickVersion' },
   129→      create: { key: 'tickVersion', value: '1' },
   130→      update: { value: String(currentVersion + 1) },
   131→    });
   132→
   133→    return true;
   134→  });
   135→}

   145→// ---- Net Worth Recalculation ----
   146→async function recalculateNetWorth(tx: PrismaTx, playerId: string): Promise<void> {
   147→  const player = await tx.player.findUnique({
   148→    where: { id: playerId },
   149→    select: { id: true, cash: true },
   150→  });
   151→  if (!player) {
   152→    console.error(`[GameEngine] recalculateNetWorth: Player ${playerId} not found`);
   153→    return;
   154→  }
   155→
   156→  const allBusinesses = await tx.business.findMany({
   157→    where: { playerId },
   158→    select: { cash: true },
   159→  });
   160→  const totalBusinessCash = allBusinesses.reduce((sum, b) => sum + b.cash, 0);
   161→
   162→  const allInventory = await tx.inventory.findMany({
   163→    where: { business: { playerId } },
   164→    select: { quantity: true, purchasePrice: true },
   165→  });
   166→  const totalInventoryValue = allInventory.reduce((sum, inv) => sum + inv.quantity * inv.purchasePrice, 0);
   167→
   168→  const activeLoans = await tx.loan.findMany({
   169→    where: { playerId, status: 'ACTIVE' },
   170→    select: { remainingDebt: true },
   171→  });
   172→  const outstandingDebt = activeLoans.reduce((sum, loan) => sum + loan.remainingDebt, 0);
   173→
   174→  const netWorth = player.cash + totalBusinessCash + totalInventoryValue - outstandingDebt;
   175→  await tx.player.update({ where: { id: playerId }, data: { netWorth } });
   176→}

   179→export async function generateEvent(): Promise<void> {
   180→  const totalWeight = EVENT_TEMPLATES.reduce((sum, e) => sum + e.weight, 0);
   181→  let random = Math.random() * totalWeight;
   182→  let selected: EventTemplate | null = null;
   183→  for (const event of EVENT_TEMPLATES) {
   184→    random -= event.weight;
   185→    if (random <= 0) {
   186→      selected = event;
   187→      break;
   188→    }
   189→  }
   190→  if (!selected) selected = EVENT_TEMPLATES[0];
   191→
   192→  const now = new Date();
   193→  const endAt = new Date(now.getTime() + selected.durationDays * 60 * 1000);
   194→  await db.gameEvent.create({
   195→    data: {
   196→      title: selected.title,
   197→      description: selected.description,
   198→      type: selected.type,
   199→      effects: JSON.stringify(selected.effects),
   200→      active: true,
   201→      startsAt: now,
   202→      endsAt: endAt,
   203→    },
   204→  });
   205→}

   213→export async function updateMarketPrices(): Promise<void> {
   214→  const allProducts = Object.values(PRODUCTS).flat();
   215→  const cities = CITIES.map(c => c.id);
   216→
   217→  // Get active events for price modifiers
   218→  const activeEvents = await db.gameEvent.findMany({ where: { active: true } });
   219→  const categoryPriceMod: Record<string, number> = {};
   220→  let allPriceMod = 0;
   221→  for (const event of activeEvents) {
   222→    const effects = JSON.parse(event.effects) as Record<string, number>;
   223→    for (const [key, value] of Object.entries(effects)) {
   224→      if (key === 'all_price' || key === 'all_imported') allPriceMod += value;
   225→      if (key.endsWith('_price')) {
   226→        const cat = key.replace('_price', '');
   227→        categoryPriceMod[cat] = (categoryPriceMod[cat] || 0) + value;
   228→      }
   229→    }
   230→  }
   231→
   232→  const updates: { productName: string; city: string; priceMultiplier: number; demandMultiplier: number }[] = [];
   233→
   234→  for (const city of cities) {
   235→    for (const product of allProducts) {
   236→      // Get current market price (for retention)
   237→      const existing = await db.marketPrice.findUnique({
   238→        where: { productName_city: { productName: product.name, city } },
   239→      });
   240→      const previousPriceMult = existing?.priceMultiplier ?? 1;
   241→
   242→      // Calculate event price modifier for this product
   243→      const eventPriceMod = allPriceMod + (categoryPriceMod[product.category] || 0);
   244→
   245→      // Use Phase 1 retention-based update
   246→      const { priceMultiplier, demandMultiplier } = calculateMarketPriceUpdate(
   247→        previousPriceMult,
   248→        eventPriceMod
   249→      );
   250→
   251→      updates.push({ productName: product.name, city, priceMultiplier, demandMultiplier });
   252→    }
   253→  }
   254→
   255→  // Batch update in a transaction
   256→  await db.$transaction(async (tx) => {
   257→    for (const upd of updates) {
   258→      await tx.marketPrice.upsert({
   259→        where: { productName_city: { productName: upd.productName, city: upd.city } },
   260→        create: upd,
   261→        update: {
   262→          priceMultiplier: upd.priceMultiplier,
   263→          demandMultiplier: upd.demandMultiplier,
   264→          updatedAt: new Date(),
   265→        },
   266→      });
   267→    }
   268→  });
   269→}

   284→/**
   285→ * Process all active marketing campaigns for a business.
   286→ * Called within simulateBusinessTick.

   290→async function processMarketingTick(
   291→  businessId: string,
   292→  businessLevel: number,
   293→  satisfactionScore: number,
   294→  loyaltyScore: number,
   295→  repeatCustomerRate: number,
   296→  currentBrandAwareness: number,
   297→  segmentDemands: { segment: string; demandMultiplier: number }[],
   298→  gameDay: number,
   299→): Promise<{
   300→  combinedDemandModifier: number;
   301→  totalMarketingSpend: number;
   302→  newBrandAwareness: number;
   303→  totalConversions: number;
   304→  totalReach: number;
   305→}> {
   306→  // Fetch active campaigns
   307→  const activeCampaigns = await db.marketingCampaign.findMany({
   308→    where: { businessId, status: 'ACTIVE' },
   309→  });
   310→
   311→  if (activeCampaigns.length === 0) {
   312→    // No campaigns — just decay brand awareness
   313→    const newAwareness = calculateBrandAwareness(currentBrandAwareness, 0);
   314→    return {
   315→      combinedDemandModifier: 1.0,
   316→      totalMarketingSpend: 0,
   317→      newBrandAwareness: newAwareness,
   318→      totalConversions: 0,
   319→      totalReach: 0,
   320→    };
   321→  }
   322→
   323→  const campaignModifiers: number[] = [];
   324→  let totalSpend = 0;
   325→  let totalReach = 0;
   326→  let totalConversions = 0;
   327→
   328→  for (const campaign of activeCampaigns) {
   329→    // Calculate daily spend (clamped to remaining budget)
   330→    const dailySpend = calculateCampaignDailySpend(
   331→      campaign.dailyBudget,
   332→      campaign.totalSpend,
   333→      campaign.totalBudget,
   334→    );
   335→
   336→    if (dailySpend <= 0) {
   337→      // Budget exhausted — complete the campaign
   338→      await db.marketingCampaign.update({
   339→        where: { id: campaign.id },
   340→        data: { status: 'COMPLETED' },
   341→      });
   342→      continue;
   343→    }
   344→
   345→    // Calculate reach
   346→    const reach = calculateCampaignReach(
   347→      campaign.channel as any,
   348→      dailySpend,
   349→      businessLevel,
   350→    );
   351→
   352→    // Calculate conversions
   353→    const conversions = calculateCampaignConversions(
   354→      campaign.channel as any,
   355→      reach,
   356→      satisfactionScore,
   357→      campaign.targetSegment as any,
   358→      segmentDemands,
   359→    );
   360→
   361→    // Calculate this campaign's demand modifier
   362→    // Use a reasonable baseline for scaling (business type base customers × city multiplier)
   363→    const baselineForScaling = 50; // Reasonable default
   364→    const demandMod = calculateCampaignDemandModifier(conversions, baselineForScaling);
   365→    campaignModifiers.push(demandMod);
   366→
   367→    // Calculate effectiveness
   368→    const newTotalSpend = campaign.totalSpend + dailySpend;
   369→    const newTotalConversions = campaign.totalConversions + conversions;
   370→    const newTotalReach = campaign.totalReach + reach;
   371→    const effectiveness = calculateCampaignEffectiveness(
   372→      newTotalConversions,
   373→      newTotalReach,
   374→      newTotalSpend,
   375→      campaign.revenueInfluenced, // Will be updated later when we know revenue
   376→    );
   377→
   378→    // Update campaign
   379→    const newDaysRun = campaign.daysRun + 1;
   380→    const isExpired = newDaysRun >= campaign.duration;
   381→
   382→    await db.marketingCampaign.update({
   383→      where: { id: campaign.id },
   384→      data: {
   385→        daysRun: newDaysRun,
   386→        totalSpend: newTotalSpend,
   387→        totalReach: newTotalReach,
   388→        totalConversions: newTotalConversions,
   389→        effectiveness,
   390→        status: isExpired ? 'COMPLETED' : 'ACTIVE',
   391→      },
   392→    });
   393→
   394→    // Create daily metric
   395→    await db.campaignMetric.create({
   396→      data: {
   397→        campaignId: campaign.id,
   398→        gameDay,
   399→        dailySpend,
   400→        dailyReach: reach,
   401→        dailyConversions: conversions,
   402→        demandModifier: demandMod,
   403→      },
   404→    });
   405→
   406→    totalSpend += dailySpend;
   407→    totalReach += reach;
   408→    totalConversions += conversions;
   409→  }
   410→
   411→  // Combined demand modifier with stacking diminishing returns
   412→  const combinedDemandModifier = calculateCombinedMarketingModifier(campaignModifiers);
   413→
   414→  // Brand awareness update
   415→  const newBrandAwareness = calculateBrandAwareness(currentBrandAwareness, totalReach);
   416→
   417→  return {
   418→    combinedDemandModifier,
   419→    totalMarketingSpend: totalSpend,
   420→    newBrandAwareness,
   421→    totalConversions,
   422→    totalReach,
   423→  };
   424→}

   431→ * Flow:
   432→ * 1. Calculate potential customers (layered model)

   447→export async function simulateBusinessTick(businessId: string): Promise<void> {
   448→  const business = await db.business.findUnique({
   449→    where: { id: businessId },
   450→    include: {
   451→      inventories: true,
   452→      employees: true,
   453→      player: { select: { id: true, cash: true, netWorth: true } },
   454→    },
   455→  });
   456→
   457→  if (!business) return;
   458→
   459→  const businessType = getBusinessType(business.type);
   460→  const city = getCity(business.city);
   461→  if (!businessType || !city) return;
   462→
   463→  const economyConfig = getBusinessEconomyConfig(business.type);
   464→
   465→  // Get active event effects (combined)
   466→  const activeEvents = await db.gameEvent.findMany({
   467→    where: { active: true },
   468→    select: { effects: true },
   469→  });
   470→  const combinedEffects: Record<string, number> = {};
   471→  for (const event of activeEvents) {
   472→    const effects = JSON.parse(event.effects) as Record<string, number>;
   473→    for (const [key, value] of Object.entries(effects)) {
   474→      combinedEffects[key] = (combinedEffects[key] || 0) + value;
   475→    }
   476→  }
   477→
   478→  // Get market prices for this city
   479→  const marketPrices = await db.marketPrice.findMany({
   480→    where: { city: business.city },
   481→  });
   482→  const marketPriceMap: Record<string, { priceMultiplier: number; demandMultiplier: number }> = {};
   483→  for (const mp of marketPrices) {
   484→    marketPriceMap[mp.productName] = {
   485→      priceMultiplier: mp.priceMultiplier,
   486→      demandMultiplier: mp.demandMultiplier,
   487→    };
   488→  }
   489→
   490→  // ---- Step 0.5: CX Demand Modifier (from previous tick) ----
   491→  // Use the business's existing satisfaction/loyalty/reputation to calculate
   492→  // how CX affects THIS tick's customer count. This avoids circular dependency
   493→  // because we use previous-tick CX values to influence current-tick demand.
   494→  const previousCxDemandModifier = calculateCXDemandModifier(
   495→    business.satisfactionScore,
   496→    business.repeatCustomerRate,
   497→    getLoyaltyTier(business.loyaltyScore).multiplier,
   498→  );
   499→
   500→  // Calculate segment demand modifier from previous-tick data
   501→  // Segments react differently to quality, service, reputation
   502→  // NOTE: priceCompetitiveness is NOT passed to segment demands because price is already
   503→  // handled by per-product priceDemandMultiplier AND satisfaction's priceCompetitiveness.
   504→  // Passing it here would triple-count price effects.
   505→  // Similarly, reputation is already in calculatePotentialCustomers Layer 3 AND satisfaction's
   506→  // atmosphere factor, so we use a moderate value to avoid quadruple-counting.
   507→  const previousCxServiceQuality = calculateServiceQuality(
   508→    business.employees.length,
   509→    business.employees.length > 0
   510→      ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   511→      : 0,
   512→    2 + business.level,
   513→  );
   514→  const segmentDemands = calculateSegmentDemands({
   515→    priceCompetitiveness: 1.0,  // Price already handled by priceDemandMultiplier + satisfaction
   516→    productQuality: Math.max(0, Math.min(1, business.healthScore / 100)), // Use health as quality proxy from previous tick
   517→    serviceQuality: previousCxServiceQuality,
   518→    reputation: Math.min(100, business.reputation * 0.5 + 50),  // Dampened: avg 50 ± 25 from reputation
   519→  });
   520→  // Aggregate segment demand: sum of demand multipliers (NOT weighted by share again)
   521→  // demandMultiplier already includes baseShare, so we just sum them
   522→  const segmentDemandModifier = segmentDemands.reduce(
   523→    (sum, seg) => sum + seg.demandMultiplier, 0
   524→  );
   525→
   526→  // ---- Step 0.6: Marketing Demand Modifier (Phase 4) ----
   527→  // Fetch gameDay early (needed for marketing metrics and later for business metrics)
   528→  const gameDayState = await db.gameState.findUnique({ where: { key: 'gameDay' } });
   529→  const gameDay = parseInt(gameDayState?.value || '1', 10);
   530→
   531→  const marketingEffect = await processMarketingTick(
   532→    business.id,
   533→    business.level,
   534→    business.satisfactionScore,
   535→    business.loyaltyScore,
   536→    business.repeatCustomerRate,
   537→    business.brandAwareness,
   538→    segmentDemands,
   539→    gameDay,
   540→  );
   541→
   542→  // ---- Step 1: Calculate potential customers ----
   543→  const totalStock = business.inventories.reduce((sum, inv) => sum + inv.quantity, 0);
   544→  // Max stock capacity: sum of product maxStock values for this business type
   545→  const productDefs = PRODUCTS[business.type] || [];
   546→  const maxStockCapacity = productDefs.reduce((sum, p) => sum + p.maxStock, 0);
   547→  const avgEmployeeSkill = business.employees.length > 0
   548→    ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   549→    : 0;
   550→
   551→  const basePotentialCustomers = calculatePotentialCustomers({
   552→    baseCustomers: businessType.baseCustomers,
   553→    cityMultiplier: city.customerMultiplier,
   554→    level: business.level,
   555→    reputation: business.reputation,
   556→    totalStock,
   557→    maxStockCapacity,
   558→    employeeCount: business.employees.length,
   559→    avgEmployeeSkill,
   560→    eventCustomerEffect: combinedEffects['all_customers'] || 0,
   561→    businessDemandEffect: combinedEffects[`${businessType.id}_demand`] || 0,
   562→    businessTypeId: business.type,
   563→  });
   564→
   565→  // Apply CX demand modifier, segment demand modifier, and marketing modifiers
   566→  // CX modifier: satisfaction/loyalty affect how many customers visit (0.2-2.0 range)
   567→  // Segment modifier: different customer segments find the business more/less attractive
   568→  // Phase 4: Marketing demand modifier and brand awareness bonus
   569→  const marketingDemandModifier = marketingEffect.combinedDemandModifier;
   570→  const brandAwarenessBonus = calculateBrandAwarenessDemandBonus(marketingEffect.newBrandAwareness);
   571→
   572→  const potentialCustomers = Math.max(0, Math.floor(
   573→    basePotentialCustomers * previousCxDemandModifier * segmentDemandModifier * marketingDemandModifier * brandAwarenessBonus
   574→  ));
   575→
   576→  // ---- Step 2: Product-level sales simulation ----
   577→  let totalRevenue = 0;
   578→  let totalCOGS = 0;
   579→  const inventoryUpdates: { id: string; quantityDecrement: number }[] = [];
   580→  const salesResults: ProductSalesResult[] = [];
   581→
   582→  for (const inventory of business.inventories) {
   583→    if (inventory.quantity <= 0) continue;
   584→
   585→    const productDef = productDefs.find(p => p.name === inventory.productName);
   586→    if (!productDef) continue;
   587→
   588→    // Product demand config
   589→    const prodDemandConfig = getProductDemandConfig(inventory.productName);
   590→
   591→    // Market reference price = typical retail price (basePrice × (1 + suggestedMarkup) × priceMultiplier)
   592→    // This represents what the MARKET typically charges, not the wholesale cost.
   593→    // Players compare their price to market retail price, not to their own cost.
   594→    const mp = marketPriceMap[inventory.productName];
   595→    const typicalRetailPrice = productDef.basePrice * (1 + productDef.suggestedMarkup);
   596→    const marketReferencePrice = Math.round(
   597→      typicalRetailPrice * (mp?.priceMultiplier || 1)
   598→    );
   599→
   600→    // Market demand multiplier from events
   601→    const demandMult = mp?.demandMultiplier || 1;
   602→
   603→    // Event demand effect for this product's category
   604→    const eventDemandEffect = combinedEffects[`${productDef.category}_demand`] || 0;
   605→
   606→    // Run the complete sales pipeline
   607→    const salesResult = simulateProductSales({
   608→      inventoryId: inventory.id,
   609→      productName: inventory.productName,
   610→      quantity: inventory.quantity,
   611→      purchasePrice: inventory.purchasePrice,
   612→      sellPrice: inventory.sellPrice,
   613→      marketReferencePrice,
   614→      baseDemand: prodDemandConfig.baseDemand,
   615→      demandMultiplier: demandMult,
   616→      eventDemandEffect,
   617→      productPriceSensitivity: prodDemandConfig.priceSensitivity,
   618→      productVolatility: prodDemandConfig.volatility,
   619→      businessPriceSensitivity: economyConfig.priceSensitivity,
   620→      potentialCustomers,
   621→    });
   622→
   623→    if (salesResult.itemsSold > 0) {
   624→      totalRevenue += salesResult.revenue;
   625→      totalCOGS += salesResult.costOfGoodsSold;
   626→      inventoryUpdates.push({
   627→        id: inventory.id,
   628→        quantityDecrement: salesResult.itemsSold,
   629→      });
   630→    }
   631→
   632→    salesResults.push(salesResult);
   633→  }
   634→
   635→  // ---- Step 3: Calculate expenses ----
   636→  const totalMonthlySalaries = business.employees.reduce((sum, emp) => sum + emp.salary, 0);
   637→  const grossProfit = totalRevenue - totalCOGS;
   638→
   639→  const expenses = calculateBusinessExpenses({
   640→    baseRent: businessType.rent,
   641→    level: business.level,
   642→    cityRentMultiplier: city.rentMultiplier,
   643→    totalMonthlySalaries,
   644→    businessLevel: business.level,
   645→    businessTypeId: business.type,
   646→    revenue: totalRevenue,
   647→    grossProfit,
   648→  });
   649→
   650→  // ---- Step 4: Calculate net profit ----
   651→  // Phase 4: Include marketing spend in total expenses
   652→  const totalExpenseWithMarketing = expenses.totalExpense + marketingEffect.totalMarketingSpend;
   653→  const dailyProfit = calculateNetProfit(totalRevenue, totalCOGS, totalExpenseWithMarketing);
   654→
   655→  // ---- Step 5: Calculate reputation change ----
   656→  const outOfStockRatio = business.inventories.length > 0
   657→    ? business.inventories.filter(inv => inv.quantity <= 0).length / business.inventories.length
   658→    : 0;
   659→  const managers = business.employees.filter(e => e.role === 'MANAGER');
   660→  const cleaners = business.employees.filter(e => e.role === 'CLEANER');
   661→
   662→  const reputationChange = calculateReputationChange({
   663→    dailyProfit,
   664→    outOfStockRatio,
   665→    managers: managers.map(m => ({ skill: m.skill })),
   666→    cleaners: cleaners.map(c => ({ skill: c.skill })),
   667→  });
   668→
   669→  const newReputation = Math.max(
   670→    ECONOMY_CONFIG.minReputation,
   671→    Math.min(ECONOMY_CONFIG.maxReputation, business.reputation + reputationChange)
   672→  );
   673→
   674→  // ---- Step 6: Calculate business health score ----
   675→  const healthResult = calculateBusinessHealth({
   676→    dailyProfit,
   677→    dailyRevenue: totalRevenue,
   678→    businessCash: business.cash + dailyProfit,
   679→    totalStock,
   680→    maxStockCapacity,
   681→    reputation: newReputation,
   682→    businessTypeId: business.type,
   683→  });
   684→
   685→  // ---- Step 6.5: Customer Experience (Phase 3) ----
   686→  // Calculate price competitiveness across all products
   687→  const cxPriceCompetitiveness = calculatePriceCompetitiveness(
   688→    business.inventories.map(inv => ({ productName: inv.productName, sellPrice: inv.sellPrice })),
   689→    productDefs.map(p => ({ name: p.name, basePrice: p.basePrice, suggestedMarkup: p.suggestedMarkup })),
   690→    marketPriceMap,
   691→  );
   692→
   693→  // Calculate service quality from employees
   694→  const idealEmployeeCount = 2 + business.level; // Base 2 + 1 per level
   695→  const cxServiceQuality = calculateServiceQuality(
   696→    business.employees.length,
   697→    avgEmployeeSkill,
   698→    idealEmployeeCount,
   699→  );
   700→
   701→  // Calculate product quality from margin (higher margin = better products)
   702→  const cxProductQuality = totalRevenue > 0
   703→    ? Math.min(1, Math.max(0, (totalRevenue - totalCOGS) / totalRevenue / 0.3))
   704→    : 0.5;
   705→
   706→  // Calculate satisfaction
   707→  const cxSatisfaction = calculateSatisfaction({
   708→    priceCompetitiveness: cxPriceCompetitiveness,
   709→    stockAvailability: maxStockCapacity > 0 ? totalStock / maxStockCapacity : 0,
   710→    serviceQuality: cxServiceQuality,
   711→    productQuality: cxProductQuality,
   712→    atmosphere: newReputation / 100,
   713→    currentSatisfaction: business.satisfactionScore,
   714→  });
   715→
   716→  // Determine if this was a positive experience
   717→  const wasPositiveExperience = cxSatisfaction.overall >= 50 && dailyProfit >= 0;
   718→
   719→  // Calculate loyalty
   720→  const cxLoyalty = calculateLoyalty({
   721→    currentLoyaltyScore: business.loyaltyScore,
   722→    currentRepeatRate: business.repeatCustomerRate,
   723→    wasPositiveExperience,
   724→    totalCustomers: potentialCustomers,
   725→    numberOfReviews: business.totalReviews,
   726→    averageRating: business.avgReviewRating,
   727→    loyaltyMultiplier: getLoyaltyTier(business.loyaltyScore).multiplier,
   728→  });
   729→
   730→  // ---- Step 7: Get game day for metrics ----
   731→  // (gameDay already fetched at Step 0.6 for marketing processing)
   732→
   733→  // ---- Step 8: Execute all updates in a single transaction ----
   734→  await db.$transaction(async (tx) => {
   735→    // Update inventory quantities
   736→    for (const upd of inventoryUpdates) {
   737→      await tx.inventory.update({
   738→        where: { id: upd.id },
   739→        data: { quantity: { decrement: upd.quantityDecrement } },
   740→      });
   741→    }
   742→
   743→    // Phase 1 FIX: Distribute business profit to player cash
   744→    // Previously, profit accumulated in business.cash but was never accessible.
   745→    // Now: business profit is transferred to the player each tick.
   746→    const profitToDistribute = dailyProfit;
   747→
   748→    // Update business (including Phase 3 CX fields and Phase 4 Marketing fields)
   749→    await tx.business.update({
   750→      where: { id: business.id },
   751→      data: {
   752→        dailyRevenue: roundTaka(totalRevenue),
   753→        dailyExpense: roundTaka(totalExpenseWithMarketing),
   754→        dailyProfit: roundTaka(dailyProfit),
   755→        dailyCOGS: roundTaka(totalCOGS),
   756→        dailyCustomers: potentialCustomers,
   757→        cash: business.cash + roundTaka(dailyProfit),
   758→        totalRevenue: business.totalRevenue + roundTaka(totalRevenue),
   759→        totalProfit: business.totalProfit + roundTaka(dailyProfit),
   760→        reputation: newReputation,
   761→        healthScore: healthResult.score,
   762→        // Phase 3: CX fields
   763→        satisfactionScore: cxSatisfaction.overall,
   764→        loyaltyScore: cxLoyalty.loyaltyScore,
   765→        repeatCustomerRate: cxLoyalty.repeatCustomerRate,
   766→        // Phase 4: Marketing fields
   767→        brandAwareness: marketingEffect.newBrandAwareness,
   768→      },
   769→    });
   770→
   771→    // Distribute profit to player cash
   772→    await tx.player.update({
   773→      where: { id: business.playerId },
   774→      data: { cash: { increment: roundTaka(profitToDistribute) } },
   775→    });
   776→
   777→    // Phase 4: Deduct marketing spend from player cash
   778→    // Marketing spend is separate from business expenses (it's a direct player cost)
   779→    if (marketingEffect.totalMarketingSpend > 0) {
   780→      await tx.player.update({
   781→        where: { id: business.playerId },
   782→        data: { cash: { decrement: roundTaka(marketingEffect.totalMarketingSpend) } },
   783→      });
   784→
   785→      // Phase 4: Attribute revenue to active campaigns (proportional to their spend)
   786→      // This gives campaigns ROI data for analytics
   787→      if (totalRevenue > 0 && marketingEffect.totalConversions > 0) {
   788→        const activeCampaigns = await tx.marketingCampaign.findMany({
   789→          where: { businessId: business.id, status: 'ACTIVE' },
   790→          select: { id: true, totalSpend: true },
   791→        });
   792→        const totalSpend = activeCampaigns.reduce((sum, c) => sum + c.totalSpend, 0);
   793→        if (totalSpend > 0) {
   794→          for (const campaign of activeCampaigns) {
   795→            // Attribute revenue proportional to this campaign's spend share
   796→            const spendShare = campaign.totalSpend / totalSpend;
   797→            // Only attribute the "extra" revenue from marketing (not base revenue)
   798→            const marketingRevenueBoost = totalRevenue * (marketingDemandModifier - 1) * (brandAwarenessBonus - 1 + 1);
   799→            const attributedRevenue = Math.max(0, marketingRevenueBoost * spendShare);
   800→            if (attributedRevenue > 0) {
   801→              await tx.marketingCampaign.update({
   802→                where: { id: campaign.id },
   803→                data: { revenueInfluenced: { increment: roundTaka(attributedRevenue) } },
   804→              });
   805→            }
   806→          }
   807→        }
   808→      }
   809→    }
   810→
   811→    // Save daily metrics for analytics (including Phase 3 CX metrics)
   812→    await tx.businessMetric.upsert({
   813→      where: { businessId_gameDay: { businessId: business.id, gameDay } },
   814→      create: {
   815→        businessId: business.id,
   816→        gameDay,
   817→        revenue: roundTaka(totalRevenue),
   818→        expenses: roundTaka(totalExpenseWithMarketing),
   819→        profit: roundTaka(dailyProfit),
   820→        cogs: roundTaka(totalCOGS),
   821→        customers: potentialCustomers,
   822→        reputation: newReputation,
   823→        // Phase 3: CX metrics
   824→        satisfaction: cxSatisfaction.overall,
   825→        loyalty: cxLoyalty.loyaltyScore,
   826→        nps: business.npsScore,
   827→      },
   828→      update: {
   829→        revenue: roundTaka(totalRevenue),
   830→        expenses: roundTaka(totalExpenseWithMarketing),
   831→        profit: roundTaka(dailyProfit),
   832→        cogs: roundTaka(totalCOGS),
   833→        customers: potentialCustomers,
   834→        reputation: newReputation,
   835→        // Phase 3: CX metrics
   836→        satisfaction: cxSatisfaction.overall,
   837→        loyalty: cxLoyalty.loyaltyScore,
   838→        nps: business.npsScore,
   839→      },
   840→    });
   841→
   842→    // Log (with Phase 3 CX info)
   843→    const cogsStr = totalCOGS > 0 ? ` | COGS ৳${roundTaka(totalCOGS).toLocaleString()}` : '';
   844→    await tx.gameLog.create({
   845→      data: {
   846→        playerId: business.playerId,
   847→        businessId: business.id,
   848→        type: dailyProfit >= 0 ? 'PROFIT' : 'LOSS',
   849→        message: `${business.name}: Rev ৳${roundTaka(totalRevenue).toLocaleString()}${cogsStr} | Exp ৳${roundTaka(totalExpenseWithMarketing).toLocaleString()} | ${dailyProfit >= 0 ? 'Profit' : 'Loss'} ৳${Math.abs(roundTaka(dailyProfit)).toLocaleString()} | ${potentialCustomers} customers | Health ${healthResult.score} | Sat ${Math.round(cxSatisfaction.overall)} | Loy ${cxLoyalty.tier} | Mkt ৳${roundTaka(marketingEffect.totalMarketingSpend).toLocaleString()}`,
   850→        amount: dailyProfit,
   851→      },
   852→    });
   853→
   854→    // Phase 3: Generate and persist customer reviews
   855→    const cxReviews = generateReviews(business.id, gameDay, cxSatisfaction, REVIEW_CONFIG.maxPerTick);
   856→    for (const review of cxReviews) {
   857→      await tx.customerReview.create({
   858→        data: {
   859→          businessId: review.businessId,
   860→          gameDay: review.gameDay,
   861→          rating: review.rating,
   862→          sentiment: review.sentiment,
   863→          category: review.category,
   864→          comment: review.comment,
   865→          segment: review.segment,
   866→        },
   867→      });
   868→    }
   869→
   870→    // Phase 3: Update NPS and review stats
   871→    if (cxReviews.length > 0) {
   872→      const allReviews = await tx.customerReview.findMany({
   873→        where: { businessId: business.id },
   874→        select: { rating: true },
   875→        orderBy: { createdAt: 'desc' },
   876→        take: REVIEW_CONFIG.keepLast,
   877→      });
   878→      const npsResult = calculateNPS(allReviews);
   879→      const totalReviews = allReviews.length;
   880→      const avgRating = totalReviews > 0
   881→        ? allReviews.reduce((sum, r) => sum + r.rating, 0) / totalReviews
   882→        : 0;
   883→
   884→      await tx.business.update({
   885→        where: { id: business.id },
   886→        data: {
   887→          npsScore: npsResult.nps,
   888→          totalReviews,
   889→          avgReviewRating: Math.round(avgRating * 100) / 100,
   890→        },
   891→      });
   892→
   893→      // Prune old reviews (keep only last N)
   894→      const reviewCount = await tx.customerReview.count({
   895→        where: { businessId: business.id },
   896→      });
   897→      if (reviewCount > REVIEW_CONFIG.keepLast) {
   898→        const oldReviews = await tx.customerReview.findMany({
   899→          where: { businessId: business.id },
   900→          orderBy: { createdAt: 'asc' },
   901→          take: reviewCount - REVIEW_CONFIG.keepLast,
   902→          select: { id: true },
   903→        });
   904→        for (const old of oldReviews) {
   905→          await tx.customerReview.delete({ where: { id: old.id } });
   906→        }
   907→      }
   908→    }
   909→
   910→    // Recalculate net worth within the same transaction
   911→    await recalculateNetWorth(tx, business.playerId);
   912→  });
   913→}

   915→// ---- Loan Payment Processing ----

   917→async function processLoanPayments(): Promise<void> {
   918→  const activeLoans = await db.loan.findMany({
   919→    where: { status: 'ACTIVE' },
   920→  });
   921→
   922→  if (activeLoans.length === 0) return;
   923→
   924→  for (const loan of activeLoans) {
   925→    const payment = loan.dailyPayment;
   926→    const isPaidOff = loan.daysRemaining - 1 <= 0;
   927→    const actualPayment = isPaidOff ? loan.remainingDebt : payment;
   928→
   929→    await db.$transaction(async (tx) => {
   930→      const player = await tx.player.findUnique({
   931→        where: { id: loan.playerId },
   932→        select: { cash: true },
   933→      });
   934→      if (!player) {
   935→        console.error(`[GameEngine] processLoanPayments: Player ${loan.playerId} not found for loan ${loan.id}`);
   936→        return;
   937→      }
   938→
   939→      const deductAmount = Math.min(actualPayment, Math.max(0, player.cash));
   940→      const newRemainingDebt = Math.max(loan.remainingDebt - deductAmount, 0);
   941→
   942→      await tx.loan.update({
   943→        where: { id: loan.id },
   944→        data: {
   945→          remainingDebt: newRemainingDebt,
   946→          daysRemaining: Math.max(loan.daysRemaining - 1, 0),
   947→          status: newRemainingDebt <= 0 ? 'PAID_OFF' : 'ACTIVE',
   948→        },
   949→      });
   950→
   951→      await tx.player.update({
   952→        where: { id: loan.playerId },
   953→        data: { cash: { decrement: deductAmount } },
   954→      });
   955→
   956→      await tx.gameLog.create({
   957→        data: {
   958→          playerId: loan.playerId,
   959→          type: newRemainingDebt <= 0 ? 'LOAN_PAID' : 'LOAN_PAYMENT',
   960→          message: newRemainingDebt <= 0
   961→            ? `Loan of ৳${loan.amount.toLocaleString()} fully repaid! Final deduction: ৳${Math.round(deductAmount).toLocaleString()}`
   962→            : `Loan payment: ৳${Math.round(deductAmount).toLocaleString()} deducted. Remaining debt: ৳${Math.round(newRemainingDebt).toLocaleString()} (${loan.daysRemaining - 1} days left)`,
   963→          amount: -deductAmount,
   964→        },
   965→      });
   966→    });
   967→  }
   968→}

   972→export async function gameTick(): Promise<void> {
   973→  // NOTE: The tick lock is managed by the API route handler.
   974→  // gameTick() is the pure simulation function and does NOT acquire/release the lock.
   975→
   976→  // Log tick start

   978→    data: {
   979→      type: 'GAME_TICK_STARTED',
   980→      message: `Game tick started at ${new Date().toISOString()}`,
   981→    },
   982→  });
   983→
   984→  // 0. Game state housekeeping
   985→  const tickState = await db.gameState.findUnique({ where: { key: 'tickCount' } });
   986→  const tickNum = parseInt(tickState?.value || '0');
   987→
   988→  await db.$transaction(async (tx) => {
   989→    // Expire old events
   990→    await tx.gameEvent.updateMany({
   991→      where: { active: true, endsAt: { lt: new Date() } },
   992→      data: { active: false },
   993→    });
   994→
   995→    // Update game state
   996→    const now = new Date().toISOString();
   997→    await tx.gameState.upsert({
   998→      where: { key: 'lastTick' },
   999→      create: { key: 'lastTick', value: now },
  1000→      update: { value: now },
  1001→    });
  1002→    await tx.gameState.upsert({
  1003→      where: { key: 'tickCount' },
  1004→      create: { key: 'tickCount', value: '1' },
  1005→      update: { value: String(tickNum + 1) },
  1006→    });
  1007→    const dayState = await tx.gameState.findUnique({ where: { key: 'gameDay' } });
  1008→    await tx.gameState.upsert({
  1009→      where: { key: 'gameDay' },
  1010→      create: { key: 'gameDay', value: '1' },
  1011→      update: { value: String(parseInt(dayState?.value || '0') + 1) },
  1012→    });
  1013→  });
  1014→
  1015→  // Generate new events (20% chance)
  1016→  if (Math.random() < 0.2) {
  1017→    try {
  1018→      await generateEvent();
  1019→    } catch (err) {
  1020→      console.error('[GameEngine] Failed to generate event:', err);
  1021→    }
  1022→  }
  1023→
  1024→  // Update market prices (every 3 ticks)
  1025→  // Phase 1: This now uses retention-based updates with event effects integrated
  1026→  if (tickNum % 3 === 0) {
  1027→    try {
  1028→      await updateMarketPrices();
  1029→    } catch (err) {
  1030→      console.error('[GameEngine] Failed to update market prices:', err);
  1031→    }
  1032→    // applyEventEffects() is now a no-op — events handled in updateMarketPrices()
  1033→  }
  1034→
  1035→  // 1. Simulate all businesses
  1036→  const businesses = await db.business.findMany({ select: { id: true } });
  1037→  for (const business of businesses) {
  1038→    try {
  1039→      await simulateBusinessTick(business.id);
  1040→    } catch (err) {
  1041→      console.error(`[GameEngine] Failed to simulate business ${business.id}:`, err);
  1042→    }
  1043→  }
  1044→
  1045→  // 2. Process loan payments
  1046→  try {
  1047→    await processLoanPayments();
  1048→  } catch (err) {
  1049→    console.error('[GameEngine] Failed to process loan payments:', err);
  1050→  }
  1051→
  1052→  // 3. Simulate AI player activity (Phase 2: Real AI decisions)
  1053→  try {
  1054→    const gameDayState2 = await db.gameState.findUnique({ where: { key: 'gameDay' } });
  1055→    const currentGameDay = parseInt(gameDayState2?.value || '1', 10);
  1056→    await simulateAIPlayersTick(currentGameDay);
  1057→  } catch (err) {
  1058→    console.error('[GameEngine] Failed to simulate AI tick:', err);
  1059→  }
  1060→
  1061→  // 4. Generate news (50% chance)
  1062→  if (Math.random() < 0.5) {
  1063→    try {
  1064→      await generateNews();
  1065→    } catch (err) {
  1066→      console.error('[GameEngine] Failed to generate news:', err);
  1067→    }
  1068→  }
  1069→
  1070→  // Log tick completion
  1071→  await db.gameLog.create({
  1072→    data: {
  1073→      type: 'GAME_TICK_COMPLETED',
  1074→      message: `Game tick completed at ${new Date().toISOString()}`,
  1075→    },
  1076→  });
  1077→}
