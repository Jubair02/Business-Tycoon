   107→      setLogsLoading(false);

   121→      // silent

   135→    } catch {

   137→    } finally {
   138→      setCxLoading(false);

   171→        toast.success(`${productName}: price set to ${formatTaka(price)}`);

   173→        refreshPlayer();

   175→        const err = await res.json();

   420→            </div>

   437→      <div className="p-3 md:p-4">
