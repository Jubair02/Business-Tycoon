    41→describe('calculatePriceCompetitiveness', () => {

    43→    { name: 'Tea', basePrice: 5, suggestedMarkup: 0.5 },

    47→  const neutralMarket = {
    48→    Tea: { priceMultiplier: 1.0 },
    49→    Rice: { priceMultiplier: 1.0 },
    50→    Soap: { priceMultiplier: 1.0 },
    51→  };

    54→    expect(calculatePriceCompetitiveness([], productDefs, neutralMarket)).toBe(0.5);
    55→  });

    58→    // marketRef = basePrice * (1 + suggestedMarkup)
    59→    // Tea: 5 * 1.5 = 7.5, Rice: 50 * 1.3 = 65, Soap: 20 * 1.4 = 28
    60→    const inventories = [
    61→      { productName: 'Tea', sellPrice: 7.5 },
    62→      { productName: 'Rice', sellPrice: 65 },
    63→      { productName: 'Soap', sellPrice: 28 },
    64→    ];
    65→    const result = calculatePriceCompetitiveness(inventories, productDefs, neutralMarket);
    66→    // ratio = 1.0 for each, competitiveness = 1 - (1-1)/2 = 1.0
    67→    expect(result).toBeCloseTo(1.0, 2);
    68→  });
    69→

    71→    const inventories = [
    72→      { productName: 'Tea', sellPrice: 7.5 * 3 },

    74→      { productName: 'Soap', sellPrice: 28 * 3 },
