    13→// ⚠️  the real tick pipeline (game-engine.ts). DO NOT use this

    18→//
    19→// 1. EVENTS: No game events are generated or applied.
    20→//    eventCustomerEffect=0, eventDemandEffect=0, activeEvents=[].
    21→//    Real engine: events affect customer counts, demand, and market prices.
    22→//
    23→// 2. MARKET PRICES: No per-city market price multipliers or retention-based
    24→//    updates. Real engine: MarketPrice table updated every 3 ticks with
    25→//    retention blending and event effects.
    26→//
    27→// 3. PRODUCT SALES: Uses individual formula calls (calculateProductDemand,
    28→//    calculatePriceDemandMultiplier, etc.) instead of simulateProductSales()
    29→//    which is the unified pipeline used by the real engine. The unified
    30→//    function adds random variation/volatility per product per day.
    31→//
    32→// 4. TICK ORDER: Real engine order is:
    33→//    Expire events → Update game state → Generate events → Update market prices
    34→//    → Simulate ALL businesses → Process loans → AI tick → Generate news
    35→//    Simulation order (per-player):
    36→//    Simulate businesses → Process loans → Mandatory restock → AI decision
    37→//
    38→// 5. REPUTATION: Uses simplified reputation update (profitable +0.5,
    39→//    unprofitable -0.8, stockout penalty, decay). Real engine uses
    40→//    calculateReputationChange() which includes manager/cleaner skill bonuses.
    41→//
    42→// 6. BUSINESS METRICS: No BusinessMetric recording. Real engine records
    43→//    daily metrics per business for trend analysis.
    44→//
    45→// 7. MARKET SHARE: No market share calculation. Real engine computes
    46→//    attractiveness scores and share percentages per city+type.
    47→//
    48→// 8. AI ACTIONS: In-memory execution is simplified. Real engine uses
    49→//    executeAIAction() with database transactions and full validation.
    50→//
    51→// 9. DETERMINISM: Uses SeededRNG (Mulberry32) for reproducibility.
    52→//    Real engine uses Math.random() — results vary per run.

    54→// 10. NO HUMAN PLAYERS: Only AI players are simulated. Real engine

    56→//
    57→// 11. NO CONCURRENCY: No tick lock, stale recovery, or transaction safety.

    59→// 12. CASH FLOOR: Has artificial floor at -1,000,000 ৳ to prevent

    91→// ---- Seeded PRNG (deterministic) ----
    92→class SeededRNG {
    93→  private seed: number;
    94→  constructor(seed: number) {
    95→    this.seed = seed;
    96→  }
    97→  /** Mulberry32 PRNG */
    98→  next(): number {
    99→    let t = (this.seed += 0x6d2b79f5);
   100→    t = Math.imul(t ^ (t >>> 15), t | 1);
   101→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
   102→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   103→  }
   104→  /** Random float in [0, 1) */
   105→  random(): number {
   106→    return this.next();
   107→  }
   108→  /** Random int in [min, max) */
   109→  randInt(min: number, max: number): number {
   110→    return min + Math.floor(this.random() * (max - min));
   111→  }
   112→  /** Random float in [min, max) */
   113→  randFloat(min: number, max: number): number {
   114→    return min + this.random() * (max - min);
   115→  }
   116→  /** Pick random element from array */
   117→  pick<T>(arr: T[]): T {
   118→    return arr[Math.floor(this.random() * arr.length)];
   119→  }
   120→}

   188→// ---- Create 8 AI Players (matching seedAIPlayers) ----
   189→function createAIPlayers(rng: SeededRNG): SimPlayer[] {
   190→  const definitions = [
   191→    { name: 'Rahim Enterprises', personality: 'CONSERVATIVE' as AIPersonality },
   192→    { name: 'Fatima Holdings', personality: 'AGGRESSIVE' as AIPersonality },
   193→    { name: 'Khan & Sons', personality: 'BALANCED' as AIPersonality },
   194→    { name: 'Chowdhury Corp', personality: 'TRADER' as AIPersonality },
   195→    { name: 'Sylhet Trading Co', personality: 'EXPANSIONIST' as AIPersonality },
   196→    { name: 'Dhaka Business Group', personality: 'BALANCED' as AIPersonality },
   197→    { name: 'Bengal Ventures', personality: 'AGGRESSIVE' as AIPersonality },
   198→    { name: 'Padma Industries', personality: 'CONSERVATIVE' as AIPersonality },
   199→  ];
   200→
   201→  const players: SimPlayer[] = [];
   202→  let nextId = 1;
   203→
   204→  for (const def of definitions) {
   205→    const baseCash = 600000 + rng.random() * 600000;
   206→    const player: SimPlayer = {
   207→      id: `player-${nextId++}`,
   208→      name: def.name,
   209→      personality: def.personality,
   210→      cash: baseCash,
   211→      netWorth: baseCash,
   212→      businesses: [],
   213→      loans: [],
   214→      lastActionAt: 0,
   215→    };
   216→
   217→    // Each AI starts with 1-2 businesses
   218→    const numBusinesses = def.personality === 'EXPANSIONIST' ? 2 : 1;
   219→    const usedTypes = new Set<string>();
   220→
   221→    for (let i = 0; i < numBusinesses; i++) {
   222→      let bType: typeof BUSINESS_TYPES[number] | undefined;
   223→
   224→      if (i === 0) {
   225→        // First business: prefer cheaper, stable types (TEA_STALL, GROCERY)
   226→        const starterTypes = BUSINESS_TYPES.filter(b => b.investment <= 300000);
   227→        bType = starterTypes.length > 0
   228→          ? rng.pick(starterTypes)
   229→          : BUSINESS_TYPES[0]; // fallback
   230→      } else {
   231→        // Second business (EXPANSIONIST only): any affordable type keeping 200K cash reserve
   232→        const affordableTypes = BUSINESS_TYPES.filter(b => !usedTypes.has(b.id) && player.cash > b.investment + 200000);
   233→        if (affordableTypes.length === 0) break;
   234→        bType = rng.pick(affordableTypes);
   235→      }
   236→
   237→      if (!bType) break;
   238→      usedTypes.add(bType.id);
   239→
   240→      const city = rng.pick(CITIES);
   241→      const investmentCost = bType.investment;
   242→      if (player.cash - investmentCost < 0) continue;
   243→
   244→      player.cash -= investmentCost;
   245→
   246→      const biz: SimBusiness = {
   247→        id: `biz-${nextId++}`,
   248→        type: bType.id,
   249→        city: city.id,
   250→        name: `${bType.name} - ${city.name}`,
   251→        level: 1,
   252→        reputation: 40 + rng.random() * 20,
   253→        cash: 0,
   254→        dailyRevenue: 0,
   255→        dailyExpense: 0,
   256→        dailyProfit: 0,
   257→        totalProfit: 0,
   258→        healthScore: 50,
   259→        inventories: [],
   260→        employees: [],
   261→      };
   262→
   263→      // Create initial inventory
   264→      const productDefs = PRODUCTS[bType.id] || [];
   265→      const config = getPersonalityConfig(def.personality);
   266→      for (const prod of productDefs) {
   267→        const initialStock = Math.floor(prod.maxStock * 0.4);
   268→        const marketRef = prod.basePrice * (1 + prod.suggestedMarkup);
   269→        const sellPrice = calculateAIPrice(marketRef, def.personality, config.defaultPricingStrategy, 50, 0.4);
   270→
   271→        biz.inventories.push({
   272→          id: `inv-${nextId++}`,
   273→          productName: prod.name,
   274→          category: prod.category,
   275→          quantity: initialStock,
   276→          maxStock: prod.maxStock,
   277→          purchasePrice: prod.basePrice,
   278→          sellPrice,
   279→        });
   280→
   281→        player.cash -= prod.basePrice * initialStock;
   282→      }
   283→
   284→      player.businesses.push(biz);
   285→    }
   286→
   287→    // Ensure cash is not negative
   288→    player.cash = Math.max(0, player.cash);
   289→    player.netWorth = calculateNetWorth(player);
   290→    players.push(player);
   291→  }
   292→
   293→  return players;
   294→}

   296→// ---- Calculate Net Worth ----
   297→function calculateNetWorth(player: SimPlayer): number {
   298→  const businessCash = player.businesses.reduce((sum, b) => sum + b.cash, 0);
   299→  const inventoryValue = player.businesses.reduce(
   300→    (sum, b) => sum + b.inventories.reduce((iSum, inv) => iSum + inv.quantity * inv.purchasePrice, 0),
   301→    0,
   302→  );
   303→  const outstandingDebt = player.loans.reduce((sum, l) => sum + l.remainingDebt, 0);
   304→  return roundTaka(player.cash + businessCash + inventoryValue - outstandingDebt);
   305→}

   307→// ---- Simulate One Business Day (Economy) ----
   308→function simulateBusinessDay(biz: SimBusiness, rng: SeededRNG): void {
   309→  const bType = BUSINESS_TYPES.find(b => b.id === biz.type);
   310→  const cityData = CITIES.find(c => c.id === biz.city);
   311→  if (!bType || !cityData) return;
   312→
   313→  const bizConfig = getBusinessEconomyConfig(biz.type);
   314→
   315→  // Calculate potential customers
   316→  const totalStock = biz.inventories.reduce((sum, inv) => sum + inv.quantity, 0);
   317→  const maxStockCapacity = biz.inventories.reduce((sum, inv) => sum + inv.maxStock, 0);
   318→  const avgEmployeeSkill = biz.employees.length > 0
   319→    ? biz.employees.reduce((sum, e) => sum + e.skill, 0) / biz.employees.length
   320→    : 0;
   321→
   322→  const potentialCustomers = calculatePotentialCustomers({
   323→    baseCustomers: bType.baseCustomers,
   324→    cityMultiplier: cityData.customerMultiplier,
   325→    level: biz.level,
   326→    reputation: biz.reputation,
   327→    totalStock,
   328→    maxStockCapacity,
   329→    employeeCount: biz.employees.length,
   330→    avgEmployeeSkill,
   331→    eventCustomerEffect: 0, // No events in simulation
   332→    businessDemandEffect: 0,
   333→    businessTypeId: biz.type,
   334→  });
   335→
   336→  // Simulate sales for each product
   337→  let totalRevenue = 0;
   338→  let totalCOGS = 0;
   339→
   340→  for (const inv of biz.inventories) {
   341→    const prodDef = (PRODUCTS[biz.type] || []).find((p: ProductDef) => p.name === inv.productName);
   342→    if (!prodDef) continue;
   343→
   344→    const demandConfig = getProductDemandConfig(inv.productName);
   345→    const marketRef = prodDef.basePrice * (1 + prodDef.suggestedMarkup);
   346→
   347→    // Product demand
   348→    const demand = calculateProductDemand({
   349→      baseDemand: demandConfig.baseDemand,
   350→      demandMultiplier: 1.0,
   351→      eventDemandEffect: 0,
   352→      productVolatility: demandConfig.volatility,
   353→    });
   354→
   355→    // Price demand
   356→    const priceMult = calculatePriceDemandMultiplier({
   357→      sellPrice: inv.sellPrice,
   358→      marketReferencePrice: marketRef,
   359→      productPriceSensitivity: demandConfig.priceSensitivity,
   360→      businessPriceSensitivity: bizConfig.priceSensitivity,
   361→    });
   362→
   363→    // Total demand
   364→    const totalDemand = potentialCustomers * demand * priceMult;
   365→    const itemsSold = calculateItemsSold(totalDemand, inv.quantity);
   366→    const revenue = calculateRevenue(itemsSold, inv.sellPrice);
   367→    const cogs = calculateCostOfGoodsSold(itemsSold, inv.purchasePrice);
   368→
   369→    // Update inventory
   370→    inv.quantity = Math.max(0, inv.quantity - itemsSold);
   371→
   372→    totalRevenue += revenue;
   373→    totalCOGS += cogs;
   374→  }
   375→
   376→  // Calculate expenses
   377→  const totalMonthlySalaries = biz.employees.reduce((sum, e) => sum + e.salary, 0);
   378→  const grossProfit = totalRevenue - totalCOGS;
   379→
   380→  const expenses = calculateBusinessExpenses({
   381→    baseRent: bType.rent,
   382→    level: biz.level,
   383→    cityRentMultiplier: cityData.rentMultiplier,
   384→    totalMonthlySalaries,
   385→    businessLevel: biz.level,
   386→    businessTypeId: biz.type,
   387→    revenue: totalRevenue,
   388→    grossProfit,
   389→  });
   390→
   391→  const netProfit = calculateNetProfit(totalRevenue, totalCOGS, expenses.totalExpense);
   392→
   393→  // Update business state
   394→  biz.dailyRevenue = totalRevenue;
   395→  biz.dailyExpense = expenses.totalExpense;
   396→  biz.dailyProfit = netProfit;
   397→  biz.totalProfit += netProfit;
   398→  biz.cash += netProfit;
   399→
   400→  // Update health score
   401→  const healthResult = calculateBusinessHealth({
   402→    dailyProfit: netProfit,
   403→    dailyRevenue: totalRevenue,
   404→    businessCash: biz.cash,
   405→    totalStock: biz.inventories.reduce((sum, inv) => sum + inv.quantity, 0),
   406→    maxStockCapacity,
   407→    reputation: biz.reputation,
   408→    businessTypeId: biz.type,
   409→  });
   410→  biz.healthScore = healthResult.score;
   411→
   412→  // Update reputation
   413→  if (netProfit > 0) {
   414→    biz.reputation = Math.min(100, biz.reputation + ECONOMY_CONFIG.reputationGainProfitable);
   415→  } else {
   416→    biz.reputation = Math.max(0, biz.reputation - ECONOMY_CONFIG.reputationLossUnprofitable);
   417→  }
   418→  const outOfStockRatio = biz.inventories.filter(inv => inv.quantity === 0).length / Math.max(biz.inventories.length, 1);
   419→  biz.reputation = Math.max(0, biz.reputation - ECONOMY_CONFIG.reputationLossStockout * outOfStockRatio);
   420→  biz.reputation = Math.max(0, biz.reputation - ECONOMY_CONFIG.reputationDecay);
   421→}

   423→// ---- Execute AI Action (In-Memory) ----
   424→function executeSimAction(player: SimPlayer, action: ScoredAction, rng: SeededRNG): void {
   425→  const config = getPersonalityConfig(player.personality);
   426→
   427→  switch (action.action) {
   428→    case 'BUY_INVENTORY': {
   429→      const bizId = action.target;
   430→      const biz = player.businesses.find(b => b.id === bizId);
   431→      if (!biz) break;
   432→
   433→      const productDefs = PRODUCTS[biz.type] || [];
   434→      let totalSpent = 0;
   435→
   436→      for (const inv of biz.inventories) {
   437→        const prodDef = productDefs.find((p: ProductDef) => p.name === inv.productName);
   438→        if (!prodDef) continue;
   439→
   440→        const stockRatio = inv.quantity / inv.maxStock;
   441→        if (stockRatio >= config.inventoryBuyThreshold) continue;
   442→
   443→        const targetStock = Math.floor(inv.maxStock * (config.inventoryBuyThreshold + 0.2));
   444→        const needed = targetStock - inv.quantity;
   445→        if (needed <= 0) continue;
   446→
   447→        const costPerUnit = prodDef.basePrice;
   448→        const cost = costPerUnit * needed;
   449→
   450→        if (player.cash - totalSpent < cost) continue;
   451→
   452→        inv.quantity += needed;
   453→        inv.purchasePrice = costPerUnit;
   454→        totalSpent += cost;
   455→      }
   456→
   457→      if (totalSpent > 0) {
   458→        player.cash -= roundTaka(totalSpent);
   459→      }
   460→      break;
   461→    }
   462→
   463→    case 'CHANGE_PRICE': {
   464→      const bizId = action.target;
   465→      const biz = player.businesses.find(b => b.id === bizId);
   466→      if (!biz) break;
   467→
   468→      const productDefs = PRODUCTS[biz.type] || [];
   469→      for (const inv of biz.inventories) {
   470→        const prodDef = productDefs.find((p: ProductDef) => p.name === inv.productName);
   471→        if (!prodDef) continue;
   472→
   473→        const marketRef = prodDef.basePrice * (1 + prodDef.suggestedMarkup);
   474→        const stockRatio = inv.quantity / inv.maxStock;
   475→        const strategy = selectPricingStrategy(player.personality, biz.healthScore, stockRatio);
   476→        const newPrice = calculateAIPrice(marketRef, player.personality, strategy, biz.healthScore, stockRatio);
   477→
   478→        if (Math.abs(newPrice - inv.sellPrice) / inv.sellPrice > 0.05) {
   479→          inv.sellPrice = newPrice;
   480→        }
   481→      }
   482→      break;
   483→    }
   484→
   485→    case 'HIRE_EMPLOYEE': {
   486→      const bizId = action.target;
   487→      const biz = player.businesses.find(b => b.id === bizId);
   488→      if (!biz || biz.employees.length >= 5) break;
   489→
   490→      const existingRoles = new Set(biz.employees.map(e => e.role));
   491→      const hirePriority = ['SALESPERSON', 'CASHIER', 'MANAGER', 'CLEANER', 'DELIVERY_RIDER'];
   492→      const roleToHire = hirePriority.find(r => !existingRoles.has(r)) || 'CASHIER';
   493→      const roleDef = EMPLOYEE_ROLES.find(r => r.role === roleToHire);
   494→      if (!roleDef) break;
   495→
   496→      if (player.cash < roleDef.baseSalary * 2) break;
   497→
   498→      const skill = 3 + rng.random() * 7;
   499→      const efficiency = 0.5 + rng.random() * 0.5;
   500→
   501→      biz.employees.push({
   502→        role: roleDef.role,
   503→        salary: roleDef.baseSalary,
   504→        skill: roundTaka(skill * 10) / 10,
   505→        efficiency: roundTaka(efficiency * 100) / 100,
   506→      });
   507→      break;
   508→    }
   509→
   510→    case 'UPGRADE_BUSINESS': {
   511→      const bizId = action.target;
   512→      const biz = player.businesses.find(b => b.id === bizId);
   513→      if (!biz || biz.level >= 10) break;
   514→
   515→      const bType = BUSINESS_TYPES.find(b => b.id === biz.type);
   516→      if (!bType) break;
   517→
   518→      const upgradeCost = Math.round(bType.investment * biz.level * 0.5);
   519→      if (player.cash < upgradeCost) break;
   520→
   521→      player.cash -= upgradeCost;
   522→      biz.level += 1;
   523→      break;
   524→    }
   525→
   526→    case 'CREATE_BUSINESS': {
   527→      const businessType = action.params?.businessType as string | undefined;
   528→      if (!businessType) break;
   529→
   530→      const bType = BUSINESS_TYPES.find(b => b.id === businessType);
   531→      if (!bType) break;
   532→
   533→      if (player.cash < bType.investment * 1.2) break;
   534→
   535→      const existingCities = player.businesses.map(b => b.city);
   536→      const newCities = CITIES.filter(c => !existingCities.includes(c.id));
   537→      const cityPool = newCities.length > 0 ? newCities : CITIES;
   538→      const city = rng.pick(cityPool);
   539→
   540→      player.cash -= bType.investment;
   541→
   542→      const newBiz: SimBusiness = {
   543→        id: `biz-new-${Date.now()}-${Math.floor(rng.random() * 10000)}`,
   544→        type: bType.id,
   545→        city: city.id,
   546→        name: `${bType.name} - ${city.name}`,
   547→        level: 1,
   548→        reputation: 50,
   549→        cash: 0,
   550→        dailyRevenue: 0,
   551→        dailyExpense: 0,
   552→        dailyProfit: 0,
   553→        totalProfit: 0,
   554→        healthScore: 50,
   555→        inventories: [],
   556→        employees: [],
   557→      };
   558→
   559→      const productDefs = PRODUCTS[bType.id] || [];
   560→      let invCost = 0;
   561→      for (const prod of productDefs) {
   562→        const initialStock = Math.floor(prod.maxStock * 0.4);
   563→        const marketRef = prod.basePrice * (1 + prod.suggestedMarkup);
   564→        const sellPrice = calculateAIPrice(marketRef, player.personality, config.defaultPricingStrategy, 50, 0.4);
   565→
   566→        newBiz.inventories.push({
   567→          id: `inv-new-${Date.now()}-${Math.floor(rng.random() * 10000)}`,
   568→          productName: prod.name,
   569→          category: prod.category,
   570→          quantity: initialStock,
   571→          maxStock: prod.maxStock,
   572→          purchasePrice: prod.basePrice,
   573→          sellPrice,
   574→        });
   575→        invCost += prod.basePrice * initialStock;
   576→      }
   577→
   578→      player.cash -= invCost;
   579→      player.businesses.push(newBiz);
   580→      break;
   581→    }
   582→
   583→    case 'SELL_BUSINESS': {
   584→      const bizId = action.target;
   585→      const bizIdx = player.businesses.findIndex(b => b.id === bizId);
   586→      if (bizIdx === -1) break;
   587→
   588→      const biz = player.businesses[bizIdx];
   589→      const bType = BUSINESS_TYPES.find(b => b.id === biz.type);
   590→      if (!bType) break;
   591→
   592→      const inventoryValue = biz.inventories.reduce((sum, inv) => sum + inv.quantity * inv.purchasePrice, 0);
   593→      const sellPrice = Math.round(bType.investment * 0.4 + inventoryValue * 0.5);
   594→
   595→      player.cash += sellPrice;
   596→      player.businesses.splice(bizIdx, 1);
   597→      break;
   598→    }
   599→
   600→    case 'TAKE_LOAN': {
   601→      if (player.loans.length >= 2) break;
   602→
   603→      const loanAmount = Math.min(
   604→        roundTaka(player.netWorth * (0.3 + rng.random() * 0.3)),
   605→        2000000,
   606→      );
   607→      if (loanAmount < 100000) break;
   608→
   609→      const interestRate = 0.05 + rng.random() * 0.03;
   610→      const durationDays = 30 + Math.floor(rng.random() * 30);
   611→      const totalOwed = roundTaka(loanAmount * (1 + interestRate));
   612→      const dailyPayment = roundTaka(totalOwed / durationDays);
   613→
   614→      player.loans.push({
   615→        id: `loan-${Date.now()}-${Math.floor(rng.random() * 10000)}`,
   616→        amount: loanAmount,
   617→        remainingDebt: totalOwed,
   618→        dailyPayment,
   619→        daysRemaining: durationDays,
   620→        interestRate,
   621→      });
   622→
   623→      player.cash += loanAmount;
   624→      break;
   625→    }
   626→
   627→    case 'REPAY_LOAN': {
   628→      const loanId = action.target;
   629→      const loan = player.loans.find(l => l.id === loanId);
   630→      if (!loan) break;
   631→
   632→      const repayment = Math.min(loan.remainingDebt, Math.max(0, player.cash * 0.5));
   633→      if (repayment < loan.dailyPayment) break;
   634→
   635→      const newRemaining = Math.max(0, loan.remainingDebt - repayment);
   636→      loan.remainingDebt = newRemaining;
   637→      player.cash -= roundTaka(repayment);
   638→
   639→      if (newRemaining <= 0) {
   640→        player.loans = player.loans.filter(l => l.id !== loanId);
   641→      }
   642→      break;
   643→    }
   644→
   645→    case 'HOLD':
   646→    default:
   647→      break;
   648→  }
   649→}

   695→// ---- Process Loan Payments ----
   696→function processLoanPayments(player: SimPlayer): void {
   697→  const remainingLoans: SimLoan[] = [];
   698→  for (const loan of player.loans) {
   699→    player.cash -= loan.dailyPayment;
   700→    loan.remainingDebt -= loan.dailyPayment;
   701→    loan.daysRemaining -= 1;
   702→    if (loan.remainingDebt <= 0 || loan.daysRemaining <= 0) {
   703→      // Loan paid off or expired
   704→      if (loan.remainingDebt > 0) {
   705→        player.cash -= loan.remainingDebt; // Pay remaining
   706→      }
   707→    } else {
   708→      remainingLoans.push(loan);
   709→    }
   710→  }
   711→  player.loans = remainingLoans;

   714→// ---- Mandatory Restock (Simulation Version) ----
   715→function performSimMandatoryRestock(player: SimPlayer): void {
   716→  const MANDATORY_RESTOCK_THRESHOLD = 0.20; // Below 20% → restock
   717→  const MANDATORY_RESTOCK_TARGET = 0.40;    // Restock to 40% of maxStock
   718→
   719→  if (player.cash <= 0) return;
   720→
   721→  let totalSpent = 0;
   722→
   723→  for (const biz of player.businesses) {
   724→    const productDefs = PRODUCTS[biz.type] || [];
   725→
   726→    for (const inv of biz.inventories) {
   727→      const prodDef = productDefs.find((p: ProductDef) => p.name === inv.productName);
   728→      if (!prodDef) continue;
   729→
   730→      const maxStock = inv.maxStock;
   731→      const stockRatio = inv.quantity / maxStock;
   732→
   733→      // Only restock if below threshold
   734→      if (stockRatio >= MANDATORY_RESTOCK_THRESHOLD) continue;
   735→
   736→      // Calculate how much to buy (bring to 40% of maxStock)
   737→      const targetQuantity = Math.floor(maxStock * MANDATORY_RESTOCK_TARGET);
   738→      const quantityToAdd = Math.max(0, targetQuantity - inv.quantity);
   739→      if (quantityToAdd <= 0) continue;
   740→
   741→      const costPerUnit = prodDef.basePrice;
   742→      const cost = costPerUnit * quantityToAdd;
   743→
   744→      // Check if player can afford this restock
   745→      if (player.cash - totalSpent - cost < 0) continue;
   746→
   747→      inv.quantity += quantityToAdd;
   748→      totalSpent += cost;
   749→    }
   750→  }
   751→
   752→  if (totalSpent > 0) {
   753→    player.cash -= roundTaka(totalSpent);
   754→  }
   755→}

   758→function runSimulation(): void {

   760→  const players = createAIPlayers(rng);
   761→
   762→  console.log('='.repeat(80));

   768→  }

   770→  const metrics: DayMetrics[] = [];

   782→
   783→      // 2. Process loan payments
   784→      processLoanPayments(player);

   799→      player.netWorth = calculateNetWorth(player);
   800→
   801→      // 6. Safety: prevent infinite negative
   802→      if (player.cash < -1000000) {
   803→        player.cash = -1000000; // Cap losses
   804→      }
   805→    }
   806→
   807→    // Record metrics
   808→    const dayMetrics: DayMetrics = {
   809→      day,
   810→      playerNetWorth: {},

   813→      playerBusinessCount: {},

   820→      dayMetrics.playerDailyProfit[p.id] = p.businesses.reduce((s, b) => s + b.dailyProfit, 0);
   821→      dayMetrics.playerBusinessCount[p.id] = p.businesses.length;
   822→      dayMetrics.playerLoanCount[p.id] = p.loans.length;
   823→      const totalStock = p.businesses.reduce((s, b) => s + b.inventories.reduce((is2, inv) => is2 + inv.quantity, 0), 0);
   824→      const totalMax = p.businesses.reduce((s, b) => s + b.inventories.reduce((is2, inv) => is2 + inv.maxStock, 0), 0);
   825→      dayMetrics.playerInventoryAvg[p.id] = totalMax > 0 ? totalStock / totalMax : 0;
   826→    }
   827→    metrics.push(dayMetrics);
   828→  }
   829→
   830→  // ---- Print Results ----
   831→  printSummary('30-DAY SUMMARY', players, metrics, 30);
   832→  printSummary('100-DAY SUMMARY', players, metrics, 100);
   833→  checkBalance(players, metrics);
   834→}
