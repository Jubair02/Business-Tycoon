    90→    90→const STALE_TICK_TIMEOUT_MS = 2 * 60 * 1000; // 2 minutes
    91→    91→
    92→    92→// ---- Tick Concurrency Lock ----
    93→    93→
    94→    94→/**
    95→    95→ * Atomically attempt to acquire the tick lock using the GameState key-value store.
    96→    96→ * Returns true if the lock was acquired, false if it was already held.
    97→    97→ *
    98→    98→ * Stale lock recovery: if the lock has been held for longer than
    99→    99→ * STALE_TICK_TIMEOUT_MS, the lock is stolen and a warning is logged.
   100→   100→ */
   101→   101→export async function acquireTickLock(): Promise<boolean> {
   102→   102→  const now = new Date();
   103→   103→
   104→   104→  return db.$transaction(async (tx) => {
   105→   105→    // Read current lock state
   106→   106→    const lockState = await tx.gameState.findUnique({ where: { key: 'tickInProgress' } });
   107→   107→    const startedAtState = await tx.gameState.findUnique({ where: { key: 'tickStartedAt' } });
   108→   108→
   109→   109→    const isLocked = lockState?.value === 'true';
   110→   110→    const startedAt = startedAtState?.value ? new Date(startedAtState.value) : null;
   111→   111→
   112→   112→    if (isLocked && startedAt) {
   113→   113→      const elapsed = now.getTime() - startedAt.getTime();
   114→   114→      if (elapsed < STALE_TICK_TIMEOUT_MS) {
   115→   115→        return false; // Lock is legitimately held
   116→   116→      }
   117→   117→      console.warn(`[GameEngine] Stale tick lock detected (held for ${Math.round(elapsed / 1000)}s). Stealing lock.`);
   118→   118→    } else if (isLocked) {
   119→   119→      console.warn('[GameEngine] Tick lock held with no timestamp. Stealing lock.');
   120→   120→    }
   121→   121→
   122→   122→    // Acquire the lock
   123→   123→    const nowISO = now.toISOString();
   124→   124→    await tx.gameState.upsert({
   125→   125→      where: { key: 'tickInProgress' },
   126→   126→      create: { key: 'tickInProgress', value: 'true' },
   127→   127→      update: { value: 'true' },
   128→   128→    });
   129→   129→    await tx.gameState.upsert({
   130→   130→      where: { key: 'tickStartedAt' },
   131→   131→      create: { key: 'tickStartedAt', value: nowISO },
   132→   132→      update: { value: nowISO },
   133→   133→    });
   134→   134→
   135→   135→    // Increment tick version
   136→   136→    const versionState = await tx.gameState.findUnique({ where: { key: 'tickVersion' } });
   137→   137→    const currentVersion = parseInt(versionState?.value || '0', 10);
   138→   138→    await tx.gameState.upsert({
   139→   139→      where: { key: 'tickVersion' },
   140→   140→      create: { key: 'tickVersion', value: '1' },
   141→   141→      update: { value: String(currentVersion + 1) },
   142→   142→    });
   143→   143→
   144→   144→    return true;
   145→   145→  });
   146→   146→}

   157→   157→async function recalculateNetWorth(tx: PrismaTx, playerId: string): Promise<void> {
   158→   158→  const player = await tx.player.findUnique({
   159→   159→    where: { id: playerId },
   160→   160→    select: { id: true, cash: true },
   161→   161→  });
   162→   162→  if (!player) {
   163→   163→    console.error(`[GameEngine] recalculateNetWorth: Player ${playerId} not found`);
   164→   164→    return;
   165→   165→  }
   166→   166→
   167→   167→  const allBusinesses = await tx.business.findMany({
   168→   168→    where: { playerId },
   169→   169→    select: { cash: true },
   170→   170→  });
   171→   171→  const totalBusinessCash = allBusinesses.reduce((sum, b) => sum + b.cash, 0);
   172→   172→
   173→   173→  const allInventory = await tx.inventory.findMany({
   174→   174→    where: { business: { playerId } },
   175→   175→    select: { quantity: true, purchasePrice: true },
   176→   176→  });
   177→   177→  const totalInventoryValue = allInventory.reduce((sum, inv) => sum + inv.quantity * inv.purchasePrice, 0);
   178→   178→
   179→   179→  const activeLoans = await tx.loan.findMany({
   180→   180→    where: { playerId, status: 'ACTIVE' },
   181→   181→    select: { remainingDebt: true },
   182→   182→  });
   183→   183→  const outstandingDebt = activeLoans.reduce((sum, loan) => sum + loan.remainingDebt, 0);
   184→   184→
   185→   185→  const netWorth = player.cash + totalBusinessCash + totalInventoryValue - outstandingDebt;
   186→   186→  await tx.player.update({ where: { id: playerId }, data: { netWorth } });
   187→   187→}

   190→   190→export async function generateEvent(): Promise<void> {
   191→   191→  const totalWeight = EVENT_TEMPLATES.reduce((sum, e) => sum + e.weight, 0);
   192→   192→  let random = Math.random() * totalWeight;
   193→   193→  let selected: EventTemplate | null = null;
   194→   194→  for (const event of EVENT_TEMPLATES) {
   195→   195→    random -= event.weight;
   196→   196→    if (random <= 0) {
   197→   197→      selected = event;
   198→   198→      break;
   199→   199→    }
   200→   200→  }
   201→   201→  if (!selected) selected = EVENT_TEMPLATES[0];
   202→   202→
   203→   203→  const now = new Date();
   204→   204→  const endAt = new Date(now.getTime() + selected.durationDays * 60 * 1000);
   205→   205→  await db.gameEvent.create({
   206→   206→    data: {
   207→   207→      title: selected.title,
   208→   208→      description: selected.description,
   209→   209→      type: selected.type,
   210→   210→      effects: JSON.stringify(selected.effects),
   211→   211→      active: true,
   212→   212→      startsAt: now,
   213→   213→      endsAt: endAt,
   214→   214→    },
   215→   215→  });

   224→   224→export async function updateMarketPrices(): Promise<void> {
   225→   225→  const allProducts = Object.values(PRODUCTS).flat();
   226→   226→  const cities = CITIES.map(c => c.id);
   227→   227→
   228→   228→  // Get active events for price modifiers
   229→   229→  const activeEvents = await db.gameEvent.findMany({ where: { active: true } });
   230→   230→  const categoryPriceMod: Record<string, number> = {};
   231→   231→  let allPriceMod = 0;
   232→   232→  for (const event of activeEvents) {
   233→   233→    const effects = JSON.parse(event.effects) as Record<string, number>;
   234→   234→    for (const [key, value] of Object.entries(effects)) {
   235→   235→      if (key === 'all_price' || key === 'all_imported') allPriceMod += value;
   236→   236→      if (key.endsWith('_price')) {
   237→   237→        const cat = key.replace('_price', '');
   238→   238→        categoryPriceMod[cat] = (categoryPriceMod[cat] || 0) + value;
   239→   239→      }
   240→   240→    }
   241→   241→  }
   242→   242→
   243→   243→  const updates: { productName: string; city: string; priceMultiplier: number; demandMultiplier: number }[] = [];
   244→   244→
   245→   245→  for (const city of cities) {
   246→   246→    for (const product of allProducts) {
   247→   247→      // Get current market price (for retention)
   248→   248→      const existing = await db.marketPrice.findUnique({
   249→   249→        where: { productName_city: { productName: product.name, city } },
   250→   250→      });
   251→   251→      const previousPriceMult = existing?.priceMultiplier ?? 1;
   252→   252→
   253→   253→      // Calculate event price modifier for this product
   254→   254→      const eventPriceMod = allPriceMod + (categoryPriceMod[product.category] || 0);
   255→   255→
   256→   256→      // Use Phase 1 retention-based update
   257→   257→      const { priceMultiplier, demandMultiplier } = calculateMarketPriceUpdate(
   258→   258→        previousPriceMult,
   259→   259→        eventPriceMod
   260→   260→      );
   261→   261→
   262→   262→      updates.push({ productName: product.name, city, priceMultiplier, demandMultiplier });
   263→   263→    }
   264→   264→  }
   265→   265→
   266→   266→  // Batch update in a transaction
   267→   267→  await db.$transaction(async (tx) => {
   268→   268→    for (const upd of updates) {
   269→   269→      await tx.marketPrice.upsert({
   270→   270→        where: { productName_city: { productName: upd.productName, city: upd.city } },
   271→   271→        create: upd,
   272→   272→        update: {
   273→   273→          priceMultiplier: upd.priceMultiplier,
   274→   274→          demandMultiplier: upd.demandMultiplier,
   275→   275→          updatedAt: new Date(),
   276→   276→        },
   277→   277→      });
   278→   278→    }
   279→   279→  });
   280→   280→}

   486→   486→  const combinedEffects: Record<string, number> = {};
   487→   487→  for (const event of activeEvents) {
   488→   488→    const effects = JSON.parse(event.effects) as Record<string, number>;

   916→   916→      const npsResult = calculateNPS(allReviews);

   990→   990→        where: { id: loan.playerId },
