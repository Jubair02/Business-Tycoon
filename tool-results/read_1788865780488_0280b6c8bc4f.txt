    66→const STALE_TICK_TIMEOUT_MS = 2 * 60 * 1000; // 2 minutes

    68→// ---- Tick Concurrency Lock ----
    69→
    70→/**
    71→ * Atomically attempt to acquire the tick lock using the GameState key-value store.
    72→ * Returns true if the lock was acquired, false if it was already held.
    73→ *
    74→ * Stale lock recovery: if the lock has been held for longer than
    75→ * STALE_TICK_TIMEOUT_MS, the lock is stolen and a warning is logged.
    76→ */
    77→export async function acquireTickLock(): Promise<boolean> {
    78→  const now = new Date();
    79→
    80→  return db.$transaction(async (tx) => {
    81→    // Read current lock state
    82→    const lockState = await tx.gameState.findUnique({ where: { key: 'tickInProgress' } });
    83→    const startedAtState = await tx.gameState.findUnique({ where: { key: 'tickStartedAt' } });
    84→
    85→    const isLocked = lockState?.value === 'true';
    86→    const startedAt = startedAtState?.value ? new Date(startedAtState.value) : null;
    87→
    88→    if (isLocked && startedAt) {
    89→      const elapsed = now.getTime() - startedAt.getTime();
    90→      if (elapsed < STALE_TICK_TIMEOUT_MS) {
    91→        return false; // Lock is legitimately held
    92→      }
    93→      console.warn(`[GameEngine] Stale tick lock detected (held for ${Math.round(elapsed / 1000)}s). Stealing lock.`);
    94→    } else if (isLocked) {
    95→      console.warn('[GameEngine] Tick lock held with no timestamp. Stealing lock.');
    96→    }
    97→
    98→    // Acquire the lock
    99→    const nowISO = now.toISOString();
   100→    await tx.gameState.upsert({
   101→      where: { key: 'tickInProgress' },
   102→      create: { key: 'tickInProgress', value: 'true' },
   103→      update: { value: 'true' },
   104→    });
   105→    await tx.gameState.upsert({
   106→      where: { key: 'tickStartedAt' },
   107→      create: { key: 'tickStartedAt', value: nowISO },
   108→      update: { value: nowISO },
   109→    });
   110→
   111→    // Increment tick version
   112→    const versionState = await tx.gameState.findUnique({ where: { key: 'tickVersion' } });
   113→    const currentVersion = parseInt(versionState?.value || '0', 10);
   114→    await tx.gameState.upsert({
   115→      where: { key: 'tickVersion' },
   116→      create: { key: 'tickVersion', value: '1' },
   117→      update: { value: String(currentVersion + 1) },
   118→    });
   119→
   120→    return true;
   121→  });
   122→}

   132→// ---- Net Worth Recalculation ----
   133→async function recalculateNetWorth(tx: PrismaTx, playerId: string): Promise<void> {
   134→  const player = await tx.player.findUnique({
   135→    where: { id: playerId },
   136→    select: { id: true, cash: true },
   137→  });
   138→  if (!player) {
   139→    console.error(`[GameEngine] recalculateNetWorth: Player ${playerId} not found`);
   140→    return;
   141→  }
   142→
   143→  const allBusinesses = await tx.business.findMany({
   144→    where: { playerId },
   145→    select: { cash: true },
   146→  });
   147→  const totalBusinessCash = allBusinesses.reduce((sum, b) => sum + b.cash, 0);
   148→
   149→  const allInventory = await tx.inventory.findMany({
   150→    where: { business: { playerId } },
   151→    select: { quantity: true, purchasePrice: true },
   152→  });
   153→  const totalInventoryValue = allInventory.reduce((sum, inv) => sum + inv.quantity * inv.purchasePrice, 0);
   154→
   155→  const activeLoans = await tx.loan.findMany({
   156→    where: { playerId, status: 'ACTIVE' },
   157→    select: { remainingDebt: true },
   158→  });
   159→  const outstandingDebt = activeLoans.reduce((sum, loan) => sum + loan.remainingDebt, 0);
   160→
   161→  const netWorth = player.cash + totalBusinessCash + totalInventoryValue - outstandingDebt;
   162→  await tx.player.update({ where: { id: playerId }, data: { netWorth } });
   163→}

   165→// ---- Event Generation ----
   166→export async function generateEvent(): Promise<void> {
   167→  const totalWeight = EVENT_TEMPLATES.reduce((sum, e) => sum + e.weight, 0);
   168→  let random = Math.random() * totalWeight;
   169→  let selected: EventTemplate | null = null;
   170→  for (const event of EVENT_TEMPLATES) {
   171→    random -= event.weight;
   172→    if (random <= 0) {
   173→      selected = event;
   174→      break;
   175→    }
   176→  }
   177→  if (!selected) selected = EVENT_TEMPLATES[0];
   178→
   179→  const now = new Date();
   180→  const endAt = new Date(now.getTime() + selected.durationDays * 60 * 1000);
   181→  await db.gameEvent.create({
   182→    data: {
   183→      title: selected.title,
   184→      description: selected.description,
   185→      type: selected.type,
   186→      effects: JSON.stringify(selected.effects),
   187→      active: true,
   188→      startsAt: now,
   189→      endsAt: endAt,
   190→    },
   191→  });
   192→}

   200→export async function updateMarketPrices(): Promise<void> {
   201→  const allProducts = Object.values(PRODUCTS).flat();
   202→  const cities = CITIES.map(c => c.id);
   203→
   204→  // Get active events for price modifiers
   205→  const activeEvents = await db.gameEvent.findMany({ where: { active: true } });
   206→  const categoryPriceMod: Record<string, number> = {};
   207→  let allPriceMod = 0;
   208→  for (const event of activeEvents) {
   209→    const effects = JSON.parse(event.effects) as Record<string, number>;
   210→    for (const [key, value] of Object.entries(effects)) {
   211→      if (key === 'all_price' || key === 'all_imported') allPriceMod += value;
   212→      if (key.endsWith('_price')) {
   213→        const cat = key.replace('_price', '');
   214→        categoryPriceMod[cat] = (categoryPriceMod[cat] || 0) + value;
   215→      }
   216→    }
   217→  }
   218→
   219→  const updates: { productName: string; city: string; priceMultiplier: number; demandMultiplier: number }[] = [];
   220→
   221→  for (const city of cities) {
   222→    for (const product of allProducts) {
   223→      // Get current market price (for retention)
   224→      const existing = await db.marketPrice.findUnique({
   225→        where: { productName_city: { productName: product.name, city } },
   226→      });
   227→      const previousPriceMult = existing?.priceMultiplier ?? 1;
   228→
   229→      // Calculate event price modifier for this product
   230→      const eventPriceMod = allPriceMod + (categoryPriceMod[product.category] || 0);
   231→
   232→      // Use Phase 1 retention-based update
   233→      const { priceMultiplier, demandMultiplier } = calculateMarketPriceUpdate(
   234→        previousPriceMult,
   235→        eventPriceMod
   236→      );
   237→
   238→      updates.push({ productName: product.name, city, priceMultiplier, demandMultiplier });
   239→    }
   240→  }
   241→
   242→  // Batch update in a transaction
   243→  await db.$transaction(async (tx) => {
   244→    for (const upd of updates) {
   245→      await tx.marketPrice.upsert({
   246→        where: { productName_city: { productName: upd.productName, city: upd.city } },
   247→        create: upd,
   248→        update: {
   249→          priceMultiplier: upd.priceMultiplier,
   250→          demandMultiplier: upd.demandMultiplier,
   251→          updatedAt: new Date(),
   252→        },
   253→      });
   254→    }
   255→  });
   256→}

   290→export async function simulateBusinessTick(businessId: string): Promise<void> {
   291→  const business = await db.business.findUnique({
   292→    where: { id: businessId },
   293→    include: {
   294→      inventories: true,
   295→      employees: true,
   296→      player: { select: { id: true, cash: true, netWorth: true } },
   297→    },
   298→  });
   299→
   300→  if (!business) return;
   301→
   302→  const businessType = getBusinessType(business.type);
   303→  const city = getCity(business.city);
   304→  if (!businessType || !city) return;
   305→
   306→  const economyConfig = getBusinessEconomyConfig(business.type);
   307→
   308→  // Get active event effects (combined)
   309→  const activeEvents = await db.gameEvent.findMany({
   310→    where: { active: true },
   311→    select: { effects: true },
   312→  });
   313→  const combinedEffects: Record<string, number> = {};
   314→  for (const event of activeEvents) {
   315→    const effects = JSON.parse(event.effects) as Record<string, number>;
   316→    for (const [key, value] of Object.entries(effects)) {
   317→      combinedEffects[key] = (combinedEffects[key] || 0) + value;
   318→    }
   319→  }
   320→
   321→  // Get market prices for this city
   322→  const marketPrices = await db.marketPrice.findMany({
   323→    where: { city: business.city },
   324→  });
   325→  const marketPriceMap: Record<string, { priceMultiplier: number; demandMultiplier: number }> = {};
   326→  for (const mp of marketPrices) {
   327→    marketPriceMap[mp.productName] = {
   328→      priceMultiplier: mp.priceMultiplier,
   329→      demandMultiplier: mp.demandMultiplier,
   330→    };
   331→  }
   332→
   333→  // ---- Step 0.5: CX Demand Modifier (from previous tick) ----
   334→  // Use the business's existing satisfaction/loyalty/reputation to calculate
   335→  // how CX affects THIS tick's customer count. This avoids circular dependency
   336→  // because we use previous-tick CX values to influence current-tick demand.
   337→  const previousCxDemandModifier = calculateCXDemandModifier(
   338→    business.satisfactionScore,
   339→    business.repeatCustomerRate,
   340→    getLoyaltyTier(business.loyaltyScore).multiplier,
   341→  );
   342→
   343→  // Calculate segment demand modifier from previous-tick data
   344→  // Segments react differently to price, quality, service, reputation
   345→  const previousCxPriceCompetitiveness = calculatePriceCompetitiveness(
   346→    business.inventories.map(inv => ({ productName: inv.productName, sellPrice: inv.sellPrice })),
   347→    (PRODUCTS[business.type] || []).map(p => ({ name: p.name, basePrice: p.basePrice, suggestedMarkup: p.suggestedMarkup })),
   348→    marketPriceMap,
   349→  );
   350→  const previousCxServiceQuality = calculateServiceQuality(
   351→    business.employees.length,
   352→    business.employees.length > 0
   353→      ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   354→      : 0,
   355→    2 + business.level,
   356→  );
   357→  const segmentDemands = calculateSegmentDemands({
   358→    priceCompetitiveness: previousCxPriceCompetitiveness,
   359→    productQuality: Math.max(0, Math.min(1, business.healthScore / 100)), // Use health as quality proxy from previous tick
   360→    serviceQuality: previousCxServiceQuality,
   361→    reputation: business.reputation,
   362→  });
   363→  // Aggregate segment demand: sum of (segment share × segment demand multiplier)
   364→  const segmentDemandModifier = segmentDemands.reduce(
   365→    (sum, seg) => sum + seg.share * seg.demandMultiplier, 0
   366→  );
   367→
   368→  // ---- Step 1: Calculate potential customers ----
   369→  const totalStock = business.inventories.reduce((sum, inv) => sum + inv.quantity, 0);
   370→  // Max stock capacity: sum of product maxStock values for this business type
   371→  const productDefs = PRODUCTS[business.type] || [];
   372→  const maxStockCapacity = productDefs.reduce((sum, p) => sum + p.maxStock, 0);
   373→  const avgEmployeeSkill = business.employees.length > 0
   374→    ? business.employees.reduce((sum, e) => sum + e.skill, 0) / business.employees.length
   375→    : 0;
   376→
   377→  const basePotentialCustomers = calculatePotentialCustomers({
   378→    baseCustomers: businessType.baseCustomers,
   379→    cityMultiplier: city.customerMultiplier,
   380→    level: business.level,
   381→    reputation: business.reputation,
   382→    totalStock,
   383→    maxStockCapacity,
   384→    employeeCount: business.employees.length,
   385→    avgEmployeeSkill,
   386→    eventCustomerEffect: combinedEffects['all_customers'] || 0,
   387→    businessDemandEffect: combinedEffects[`${businessType.id}_demand`] || 0,
   388→    businessTypeId: business.type,
   389→  });
   390→
   391→  // Apply CX demand modifier and segment demand modifier
   392→  // CX modifier: satisfaction/loyalty affect how many customers visit (0.2-2.0 range)
   393→  // Segment modifier: different customer segments find the business more/less attractive
   394→  const potentialCustomers = Math.max(0, Math.floor(
   395→    basePotentialCustomers * previousCxDemandModifier * segmentDemandModifier
   396→  ));
   397→
   398→  // ---- Step 2: Product-level sales simulation ----
   399→  let totalRevenue = 0;
   400→  let totalCOGS = 0;
   401→  const inventoryUpdates: { id: string; quantityDecrement: number }[] = [];
   402→  const salesResults: ProductSalesResult[] = [];
   403→
   404→  for (const inventory of business.inventories) {
   405→    if (inventory.quantity <= 0) continue;
   406→
   407→    const productDef = productDefs.find(p => p.name === inventory.productName);
   408→    if (!productDef) continue;
   409→
   410→    // Product demand config
   411→    const prodDemandConfig = getProductDemandConfig(inventory.productName);
   412→
   413→    // Market reference price = typical retail price (basePrice × (1 + suggestedMarkup) × priceMultiplier)
   414→    // This represents what the MARKET typically charges, not the wholesale cost.
   415→    // Players compare their price to market retail price, not to their own cost.
   416→    const mp = marketPriceMap[inventory.productName];
   417→    const typicalRetailPrice = productDef.basePrice * (1 + productDef.suggestedMarkup);
   418→    const marketReferencePrice = Math.round(
   419→      typicalRetailPrice * (mp?.priceMultiplier || 1)
   420→    );
   421→
   422→    // Market demand multiplier from events
   423→    const demandMult = mp?.demandMultiplier || 1;
   424→
   425→    // Event demand effect for this product's category
   426→    const eventDemandEffect = combinedEffects[`${productDef.category}_demand`] || 0;
   427→
   428→    // Run the complete sales pipeline
   429→    const salesResult = simulateProductSales({
   430→      inventoryId: inventory.id,
   431→      productName: inventory.productName,
   432→      quantity: inventory.quantity,
   433→      purchasePrice: inventory.purchasePrice,
   434→      sellPrice: inventory.sellPrice,
   435→      marketReferencePrice,
   436→      baseDemand: prodDemandConfig.baseDemand,
   437→      demandMultiplier: demandMult,
   438→      eventDemandEffect,
   439→      productPriceSensitivity: prodDemandConfig.priceSensitivity,
   440→      productVolatility: prodDemandConfig.volatility,
   441→      businessPriceSensitivity: economyConfig.priceSensitivity,
   442→      potentialCustomers,
   443→    });
   444→
   445→    if (salesResult.itemsSold > 0) {
   446→      totalRevenue += salesResult.revenue;
   447→      totalCOGS += salesResult.costOfGoodsSold;
   448→      inventoryUpdates.push({
   449→        id: inventory.id,
   450→        quantityDecrement: salesResult.itemsSold,
   451→      });
   452→    }
   453→
   454→    salesResults.push(salesResult);
   455→  }
   456→
   457→  // ---- Step 3: Calculate expenses ----
   458→  const totalMonthlySalaries = business.employees.reduce((sum, emp) => sum + emp.salary, 0);
   459→  const grossProfit = totalRevenue - totalCOGS;
   460→
   461→  const expenses = calculateBusinessExpenses({
   462→    baseRent: businessType.rent,
   463→    level: business.level,
   464→    cityRentMultiplier: city.rentMultiplier,
   465→    totalMonthlySalaries,
   466→    businessLevel: business.level,
   467→    businessTypeId: business.type,
   468→    revenue: totalRevenue,
   469→    grossProfit,
   470→  });
   471→
   472→  // ---- Step 4: Calculate net profit ----
   473→  const dailyProfit = calculateNetProfit(totalRevenue, totalCOGS, expenses.totalExpense);
   474→
   475→  // ---- Step 5: Calculate reputation change ----
   476→  const outOfStockRatio = business.inventories.length > 0
   477→    ? business.inventories.filter(inv => inv.quantity <= 0).length / business.inventories.length
   478→    : 0;
   479→  const managers = business.employees.filter(e => e.role === 'MANAGER');
   480→  const cleaners = business.employees.filter(e => e.role === 'CLEANER');
   481→
   482→  const reputationChange = calculateReputationChange({
   483→    dailyProfit,
   484→    outOfStockRatio,
   485→    managers: managers.map(m => ({ skill: m.skill })),
   486→    cleaners: cleaners.map(c => ({ skill: c.skill })),
   487→  });
   488→
   489→  const newReputation = Math.max(
   490→    ECONOMY_CONFIG.minReputation,
   491→    Math.min(ECONOMY_CONFIG.maxReputation, business.reputation + reputationChange)
   492→  );
   493→
   494→  // ---- Step 6: Calculate business health score ----
   495→  const healthResult = calculateBusinessHealth({
   496→    dailyProfit,
   497→    dailyRevenue: totalRevenue,
   498→    businessCash: business.cash + dailyProfit,
   499→    totalStock,
   500→    maxStockCapacity,
   501→    reputation: newReputation,
   502→    businessTypeId: business.type,
   503→  });
   504→
   505→  // ---- Step 6.5: Customer Experience (Phase 3) ----
   506→  // Calculate price competitiveness across all products
   507→  const cxPriceCompetitiveness = calculatePriceCompetitiveness(
   508→    business.inventories.map(inv => ({ productName: inv.productName, sellPrice: inv.sellPrice })),
   509→    productDefs.map(p => ({ name: p.name, basePrice: p.basePrice, suggestedMarkup: p.suggestedMarkup })),
   510→    marketPriceMap,
   511→  );
   512→
   513→  // Calculate service quality from employees
   514→  const idealEmployeeCount = 2 + business.level; // Base 2 + 1 per level
   515→  const cxServiceQuality = calculateServiceQuality(
   516→    business.employees.length,
   517→    avgEmployeeSkill,
   518→    idealEmployeeCount,
   519→  );
   520→
   521→  // Calculate product quality from margin (higher margin = better products)
   522→  const cxProductQuality = totalRevenue > 0
   523→    ? Math.min(1, Math.max(0, (totalRevenue - totalCOGS) / totalRevenue / 0.3))
   524→    : 0.5;
   525→
   526→  // Calculate satisfaction
   527→  const cxSatisfaction = calculateSatisfaction({
   528→    priceCompetitiveness: cxPriceCompetitiveness,
   529→    stockAvailability: maxStockCapacity > 0 ? totalStock / maxStockCapacity : 0,
   530→    serviceQuality: cxServiceQuality,
   531→    productQuality: cxProductQuality,
   532→    atmosphere: newReputation / 100,
   533→    currentSatisfaction: business.satisfactionScore,
   534→  });
   535→
   536→  // Determine if this was a positive experience
   537→  const wasPositiveExperience = cxSatisfaction.overall >= 50 && dailyProfit >= 0;
   538→
   539→  // Calculate loyalty
   540→  const cxLoyalty = calculateLoyalty({
   541→    currentLoyaltyScore: business.loyaltyScore,
   542→    currentRepeatRate: business.repeatCustomerRate,
   543→    wasPositiveExperience,
   544→    totalCustomers: potentialCustomers,
   545→    numberOfReviews: business.totalReviews,
   546→    averageRating: business.avgReviewRating,
   547→    loyaltyMultiplier: getLoyaltyTier(business.loyaltyScore).multiplier,
   548→  });
   549→
   550→  // ---- Step 7: Get game day for metrics ----
   551→  const gameDayState = await db.gameState.findUnique({ where: { key: 'gameDay' } });
   552→  const gameDay = parseInt(gameDayState?.value || '1', 10);
   553→
   554→  // ---- Step 8: Execute all updates in a single transaction ----

   612→      },
   613→      update: {
   614→        revenue: roundTaka(totalRevenue),
   615→        expenses: roundTaka(expenses.totalExpense),
   616→        profit: roundTaka(dailyProfit),
   617→        cogs: roundTaka(totalCOGS),
   618→        customers: potentialCustomers,

   620→        // Phase 3: CX metrics
   621→        satisfaction: cxSatisfaction.overall,

'Profit' : 'Loss'} ৳${Math.abs(roundTaka(dailyProfit)).toLocaleString()} | ${potentialCustomers} customers | Health ${healthResult.score} | Sat ${Math.round(cxSatisfaction.overall)} | Loy ${cxLoyalty.tier}`,
   635→        amount: dailyProfit,

   657→      const allReviews = await tx.customerReview.findMany({
   658→        where: { businessId: business.id },
   659→        select: { rating: true },
   660→        orderBy: { createdAt: 'desc' },
   661→        take: REVIEW_CONFIG.keepLast,
   662→      });
   663→      const npsResult = calculateNPS(allReviews);
   664→      const totalReviews = allReviews.length;
   665→      const avgRating = totalReviews > 0
   666→        ? allReviews.reduce((sum, r) => sum + r.rating, 0) / totalReviews
   667→        : 0;
   668→
   669→      await tx.business.update({
   670→        where: { id: business.id },
   671→        data: {
   672→          npsScore: npsResult.nps,
   673→          totalReviews,
   674→          avgReviewRating: Math.round(avgRating * 100) / 100,
   675→        },
   676→      });
   677→
   678→      // Prune old reviews (keep only last N)
   679→      const reviewCount = await tx.customerReview.count({
   680→        where: { businessId: business.id },
   681→      });
   682→      if (reviewCount > REVIEW_CONFIG.keepLast) {
   683→        const oldReviews = await tx.customerReview.findMany({
   684→          where: { businessId: business.id },
   685→          orderBy: { createdAt: 'asc' },
   686→          take: reviewCount - REVIEW_CONFIG.keepLast,
   687→          select: { id: true },
   688→        });
   689→        for (const old of oldReviews) {
   690→          await tx.customerReview.delete({ where: { id: old.id } });
   691→        }
   692→      }
   693→    }
   694→
   695→    // Recalculate net worth within the same transaction
   696→    await recalculateNetWorth(tx, business.playerId);
   697→  });
   698→}

   700→// ---- Loan Payment Processing ----
   701→
   702→async function processLoanPayments(): Promise<void> {
   703→  const activeLoans = await db.loan.findMany({
   704→    where: { status: 'ACTIVE' },
   705→  });
   706→
   707→  if (activeLoans.length === 0) return;
   708→
   709→  for (const loan of activeLoans) {
   710→    const payment = loan.dailyPayment;
   711→    const isPaidOff = loan.daysRemaining - 1 <= 0;
   712→    const actualPayment = isPaidOff ? loan.remainingDebt : payment;
   713→
   714→    await db.$transaction(async (tx) => {
   715→      const player = await tx.player.findUnique({
   716→        where: { id: loan.playerId },
   717→        select: { cash: true },
   718→      });
   719→      if (!player) {
   720→        console.error(`[GameEngine] processLoanPayments: Player ${loan.playerId} not found for loan ${loan.id}`);
   721→        return;
   722→      }
   723→
   724→      const deductAmount = Math.min(actualPayment, Math.max(0, player.cash));
   725→      const newRemainingDebt = Math.max(loan.remainingDebt - deductAmount, 0);
   726→
   727→      await tx.loan.update({
   728→        where: { id: loan.id },
   729→        data: {
   730→          remainingDebt: newRemainingDebt,
   731→          daysRemaining: Math.max(loan.daysRemaining - 1, 0),
   732→          status: newRemainingDebt <= 0 ? 'PAID_OFF' : 'ACTIVE',
   733→        },
   734→      });
   735→
   736→      await tx.player.update({
   737→        where: { id: loan.playerId },
   738→        data: { cash: { decrement: deductAmount } },
   739→      });
   740→
   741→      await tx.gameLog.create({
   742→        data: {
   743→          playerId: loan.playerId,
   744→          type: newRemainingDebt <= 0 ? 'LOAN_PAID' : 'LOAN_PAYMENT',
   745→          message: newRemainingDebt <= 0
   746→            ? `Loan of ৳${loan.amount.toLocaleString()} fully repaid! Final deduction: ৳${Math.round(deductAmount).toLocaleString()}`
   747→            : `Loan payment: ৳${Math.round(deductAmount).toLocaleString()} deducted. Remaining debt: ৳${Math.round(newRemainingDebt).toLocaleString()} (${loan.daysRemaining - 1} days left)`,
   748→          amount: -deductAmount,
   749→        },
   750→      });
   751→    });
   752→  }
   753→}

   757→export async function gameTick(): Promise<void> {

   791→    });

   793→    await tx.gameState.upsert({
   794→      where: { key: 'gameDay' },
   795→      create: { key: 'gameDay', value: '1' },
   796→      update: { value: String(parseInt(dayState?.value || '0') + 1) },
   797→    });

   800→  // Generate new events (20% chance)
   801→  if (Math.random() < 0.2) {

   803→      await generateEvent();

   805→      console.error('[GameEngine] Failed to generate event:', err);
   806→    }
   807→  }
   808→
   809→  // Update market prices (every 3 ticks)
   810→  // Phase 1: This now uses retention-based updates with event effects integrated
   811→  if (tickNum % 3 === 0) {
   812→    try {
   813→      await updateMarketPrices();
   814→    } catch (err) {
   815→      console.error('[GameEngine] Failed to update market prices:', err);
   816→    }
   817→    // applyEventEffects() is now a no-op — events handled in updateMarketPrices()
   818→  }
   819→
   820→  // 1. Simulate all businesses
   821→  const businesses = await db.business.findMany({ select: { id: true } });
   822→  for (const business of businesses) {
   823→    try {
   824→      await simulateBusinessTick(business.id);
   825→    } catch (err) {
   826→      console.error(`[GameEngine] Failed to simulate business ${business.id}:`, err);
   827→    }
   828→  }
   829→
   830→  // 2. Process loan payments
   831→  try {
   832→    await processLoanPayments();
   833→  } catch (err) {
   834→    console.error('[GameEngine] Failed to process loan payments:', err);
   835→  }
   836→
   837→  // 3. Simulate AI player activity (Phase 2: Real AI decisions)
   838→  try {
   839→    const gameDayState2 = await db.gameState.findUnique({ where: { key: 'gameDay' } });
   840→    const currentGameDay = parseInt(gameDayState2?.value || '1', 10);
   841→    await simulateAIPlayersTick(currentGameDay);
   842→  } catch (err) {
   843→    console.error('[GameEngine] Failed to simulate AI tick:', err);
   844→  }
   845→
   846→  // 4. Generate news (50% chance)
   847→  if (Math.random() < 0.5) {
   848→    try {
   849→      await generateNews();
   850→    } catch (err) {
   851→      console.error('[GameEngine] Failed to generate news:', err);
   852→    }
   853→  }
   854→
   855→  // Log tick completion
   856→  await db.gameLog.create({
   857→    data: {
   858→      type: 'GAME_TICK_COMPLETED',
   859→      message: `Game tick completed at ${new Date().toISOString()}`,
   860→    },
   861→  });
