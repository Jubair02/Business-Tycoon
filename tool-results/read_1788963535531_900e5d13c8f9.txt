   173→describe('AI Strategy - calculateAIPrice', () => {
   174→  const marketPrice = 100;
   175→
   176→  it('returns different values for different strategies', () => {
   177→    // Run multiple times since there's randomness
   178→    const results = new Set<number>();
   179→    const strategies: AIPricingStrategy[] = ['LOW_PRICE', 'MARKET_PRICE', 'PREMIUM', 'DYNAMIC'];
   180→    for (const strategy of strategies) {
   181→      // Run a few times to account for randomness
   182→      for (let i = 0; i < 5; i++) {
   183→        results.add(calculateAIPrice(marketPrice, 'BALANCED', strategy, 70, 0.5));
   184→      }
   185→    }
   186→    // With 4 different strategies and randomness, we should get multiple distinct values
   187→    expect(results.size).toBeGreaterThan(1);
   188→  });
   189→
   190→  it('LOW_PRICE strategy gives prices below market', () => {
   191→    // LOW_PRICE multiplier is 0.85 + random*0.13 → range 0.85-0.98
   192→    // With CONSERVATIVE personality modifier (*0.98), effective range: ~0.83-0.96
   193→    for (let i = 0; i < 20; i++) {
   194→      const price = calculateAIPrice(marketPrice, 'CONSERVATIVE', 'LOW_PRICE', 70, 0.5);
   195→      // Should be at or below market price (with small margin for rounding)
   196→      expect(price).toBeLessThanOrEqual(marketPrice);
   197→    }
   198→  });
   199→
   200→  it('PREMIUM strategy gives prices above market', () => {
   201→    // PREMIUM multiplier is 1.05 + random*0.15 → range 1.05-1.20
   202→    // With AGGRESSIVE personality modifier (*1.02), effective range: ~1.07-1.22
   203→    for (let i = 0; i < 20; i++) {
   204→      const price = calculateAIPrice(marketPrice, 'AGGRESSIVE', 'PREMIUM', 70, 0.5);
   205→      expect(price).toBeGreaterThanOrEqual(marketPrice);
   206→    }
   207→  });
   208→
   209→  it('MARKET_PRICE strategy gives prices near market', () => {
   210→    // MARKET_PRICE multiplier is 0.98 + random*0.07 → range 0.98-1.05
   211→    for (let i = 0; i < 20; i++) {
   212→      const price = calculateAIPrice(marketPrice, 'BALANCED', 'MARKET_PRICE', 70, 0.5);
   213→      // Should be within ~7% of market price
   214→      expect(price).toBeGreaterThanOrEqual(marketPrice * 0.95);
   215→      expect(price).toBeLessThanOrEqual(marketPrice * 1.1);
   216→    }
   217→  });
   218→
   219→  it('DYNAMIC strategy reacts to low health by cutting prices', () => {
   220→    let lowHealthTotal = 0;
   221→    let normalHealthTotal = 0;
   222→    const iterations = 30;
   223→    for (let i = 0; i < iterations; i++) {
   224→      lowHealthTotal += calculateAIPrice(marketPrice, 'BALANCED', 'DYNAMIC', 15, 0.5);
   225→      normalHealthTotal += calculateAIPrice(marketPrice, 'BALANCED', 'DYNAMIC', 80, 0.5);
   226→    }
   227→    // Low health should produce lower average prices than normal health
   228→    expect(lowHealthTotal / iterations).toBeLessThan(normalHealthTotal / iterations);
   229→  });

   231→  it('DYNAMIC strategy reacts to overstock by discounting', () => {
   232→    let overstockTotal = 0;
   233→    let normalStockTotal = 0;
   234→    const iterations = 30;
   235→    for (let i = 0; i < iterations; i++) {
   236→      overstockTotal += calculateAIPrice(marketPrice, 'BALANCED', 'DYNAMIC', 70, 0.9); // stockRatio > 0.8
   237→      normalStockTotal += calculateAIPrice(marketPrice, 'BALANCED', 'DYNAMIC', 70, 0.5);
   238→    }
   239→    // Overstocked should produce slightly lower average prices
   240→    expect(overstockTotal / iterations).toBeLessThanOrEqual(normalStockTotal / iterations);
   241→  });

   244→    let lowStockTotal = 0;
   245→    let normalStockTotal = 0;

   248→      lowStockTotal += calculateAIPrice(marketPrice, 'BALANCED', 'DYNAMIC', 70, 0.1); // stockRatio < 0.2
   249→      normalStockTotal += calculateAIPrice(marketPrice, 'BALANCED', 'DYNAMIC', 70, 0.5);

   630→interface MarketShareInput {
   631→  reputation: number;      // 0-100
   632→  level: number;           // 1-10
   633→  healthScore: number;     // 0-100
   634→  totalStock: number;
   635→  maxStockCapacity: number;
   636→  avgSellPrice: number;
   637→  dailyRevenue: number;
   638→}
   639→
   640→function calculateAttractiveness(
   641→  biz: MarketShareInput,
   642→  avgMarketPrice: number,
   643→  avgRevenue: number,
   644→): number {
   645→  const repFactor = 0.4 + (biz.reputation / 100) * 1.2;
   646→  const levelFactor = 1 + (biz.level - 1) * 0.12;
   647→  const healthFactor = 0.5 + (biz.healthScore / 100);
   648→  const stockRatio = biz.maxStockCapacity > 0 ? biz.totalStock / biz.maxStockCapacity : 0;
   649→  const inventoryFactor = 0.3 + stockRatio * 0.7;
   650→  const pricingFactor = Math.max(0.5, Math.min(1.5, avgMarketPrice / Math.max(biz.avgSellPrice, 1)));
   651→  const revenueFactor = 0.5 + Math.min(biz.dailyRevenue / Math.max(avgRevenue, 1), 2.0) * 0.25;
   652→  return repFactor * levelFactor * healthFactor * inventoryFactor * pricingFactor * revenueFactor;
   653→}
   654→
   655→function computeMarketShares(businesses: MarketShareInput[]): number[] {
   656→  if (businesses.length === 0) return [];
   657→
   658→  const avgMarketPrice = businesses.reduce((s, b) => s + b.avgSellPrice, 0) / businesses.length;
   659→  const avgRevenue = businesses.reduce((s, b) => s + b.dailyRevenue, 0) / businesses.length;
   660→
   661→  const scores = businesses.map(b => calculateAttractiveness(b, avgMarketPrice, avgRevenue));
   662→  const totalScore = scores.reduce((s, sc) => s + sc, 0);
   663→
   664→  return totalScore > 0 ? scores.map(sc => sc / totalScore) : scores.map(() => 0);
   665→}
