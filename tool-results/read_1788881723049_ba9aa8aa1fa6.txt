    69→describe('calculateCampaignReach', () => {
    70→  it('returns > 0 for valid inputs', () => {
    71→    // Run multiple times to account for random variation

    85→    expect(calculateCampaignReach('SOCIAL_MEDIA', -100, 1)).toBe(0);
    86→  });
    87→
    88→  it('higher spend generally produces more reach', () => {
    89→    // Test by averaging multiple runs to smooth out random variation
    90→    let lowSpendTotal = 0;
    91→    let highSpendTotal = 0;
    92→    const runs = 50;
    93→
    94→    for (let i = 0; i < runs; i++) {
    95→      lowSpendTotal += calculateCampaignReach('SOCIAL_MEDIA', 500, 1);
    96→      highSpendTotal += calculateCampaignReach('SOCIAL_MEDIA', 5000, 1);
    97→    }
    98→
    99→    const lowAvg = lowSpendTotal / runs;
   100→    const highAvg = highSpendTotal / runs;
   101→
   102→    // Higher spend should produce more reach
   103→    expect(highAvg).toBeGreaterThan(lowAvg);
   104→  });
   105→
   106→  it('diminishing returns exponent reduces the spend-ratio bonus', () => {
   107→    // The formula: reach = reachPerTaka * spend * (spend / baseCost)^exponent
   108→    // With exponent=0.6, the bonus multiplier for 10x spend = 10^0.6 ≈ 3.98
   109→    // If exponent were 1.0 (no diminishing returns), bonus = 10^1.0 = 10
   110→    // So the actual ratio should be less than what exponent=1.0 would give
   111→    //
   112→    // For SOCIAL_MEDIA: baseCost=500, exponent=0.6
   113→    // At spend 500:  ratio = 500/500 = 1, bonus = 1^0.6 = 1, reach ∝ 500*1 = 500
   114→    // At spend 5000: ratio = 5000/500 = 10, bonus = 10^0.6 ≈ 3.98, reach ∝ 5000*3.98 = 19900
   115→    // Ratio of reach = 19900/500 = 39.8
   116→    // If exponent=1: ratio would be 5000*10 / 500*1 = 100
   117→    //
   118→    // Key: with exponent < 1, the spend-ratio bonus grows sub-linearly
   119→    // Verify: 2x spend gives bonus of 2^0.6 ≈ 1.52, not 2^1.0 = 2
   120→    const bonusAt2x = Math.pow(2, 0.6);
   121→    expect(bonusAt2x).toBeLessThan(2.0); // Diminishing returns on the ratio bonus
   122→    expect(bonusAt2x).toBeCloseTo(1.52, 1);
   123→  });
   124→
   125→  it('higher business level produces more reach', () => {
   126→    let level1Total = 0;
   127→    let level5Total = 0;
   128→    const runs = 50;
   129→
   130→    for (let i = 0; i < runs; i++) {
   131→      level1Total += calculateCampaignReach('SOCIAL_MEDIA', 1000, 1);
   132→      level5Total += calculateCampaignReach('SOCIAL_MEDIA', 1000, 5);
   133→    }
   134→
   135→    // Level 5 should give more reach than level 1 (8% per level bonus)
   136→    expect(level5Total / runs).toBeGreaterThan(level1Total / runs);
   137→  });
   138→
   139→  it('SOCIAL_MEDIA has cheaper reach per taka than TV_MEDIA', () => {

   141→    // TV_MEDIA: reachPerTaka=0.8, baseDailyCost=20000
   142→    // At the same spend, SOCIAL_MEDIA should give more reach

   144→    let tvTotal = 0;
   145→    const runs = 50;

   147→    for (let i = 0; i < runs; i++) {
   148→      socialTotal += calculateCampaignReach('SOCIAL_MEDIA', 20000, 4);

   153→  });
   154→
