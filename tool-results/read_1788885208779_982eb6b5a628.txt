   120→// ---- Helpers ----
   121→function formatTk(n: number): string {
   122→  return `৳${n.toLocaleString()}`;

   124→
   125→function statusColor(status: CampaignStatus): string {
   126→  switch (status) {
   127→    case 'ACTIVE': return 'bg-green-100 text-green-800 border-green-300';
   128→    case 'PAUSED': return 'bg-yellow-100 text-yellow-800 border-yellow-300';
   129→    case 'COMPLETED': return 'bg-blue-100 text-blue-800 border-blue-300';
   130→    case 'CANCELLED': return 'bg-gray-100 text-gray-800 border-gray-300';
   131→  }
   132→}
   133→
   134→function roiColor(roi: number): string {
   135→  if (roi > 0) return 'text-green-600';
   136→  if (roi < 0) return 'text-red-600';
   137→  return 'text-gray-500';
   138→}

   141→interface MarketingViewProps {

   143→  businessLevel?: number;

   145→
   146→export default function MarketingView({ businessId, businessLevel = 1 }: MarketingViewProps) {
   147→  // Data state
   148→  const [campaigns, setCampaigns] = useState<Campaign[]>([]);
   149→  const [analytics, setAnalytics] = useState<AnalyticsData | null>(null);
   150→  const [loading, setLoading] = useState(true);
   151→  const [analyticsLoading, setAnalyticsLoading] = useState(true);
   152→
   153→  // Create campaign dialog
   154→  const [showCreateDialog, setShowCreateDialog] = useState(false);
   155→  const [createForm, setCreateForm] = useState({
   156→    name: '',
   157→    channel: 'SOCIAL_MEDIA' as MarketingChannel,
   158→    targetSegment: null as CampaignTargetSegment,
   159→    dailyBudget: 1000,
   160→    duration: 7,
   161→  });
   162→  const [createErrors, setCreateErrors] = useState<string[]>([]);
   163→  const [creating, setCreating] = useState(false);
   164→
   165→  // Action state
   166→  const [actioningId, setActioningId] = useState<string | null>(null);
   167→
   168→  // History expansion
   169→  const [expandedHistoryId, setExpandedHistoryId] = useState<string | null>(null);
   170→
   171→  // ---- Fetch campaigns ----
   172→  const fetchCampaigns = useCallback(async () => {
   173→    try {
   174→      const res = await fetch(`/api/businesses/${businessId}/campaigns`);
   175→      if (res.ok) {
   176→        const data = await res.json();
   177→        setCampaigns(Array.isArray(data) ? data : data.campaigns || []);
   178→      }
   179→    } catch {
   180→      // silent
   181→    } finally {
   182→      setLoading(false);
   183→    }
   184→  }, [businessId]);
   185→
   186→  // ---- Fetch analytics ----
   187→  const fetchAnalytics = useCallback(async () => {
   188→    try {
   189→      const res = await fetch(`/api/businesses/${businessId}/campaigns/analytics`);
   190→      if (res.ok) {
   191→        const data = await res.json();
   192→        setAnalytics(data.data || data);
   193→      }
   194→    } catch {
   195→      // silent
   196→    } finally {
   197→      setAnalyticsLoading(false);
   198→    }
   199→  }, [businessId]);
   200→
   201→  useEffect(() => {
   202→    setLoading(true);
   203→    setAnalyticsLoading(true);
   204→    fetchCampaigns();
   205→    fetchAnalytics();
   206→  }, [fetchCampaigns, fetchAnalytics]);
   207→
   208→  // ---- Campaign actions ----
   209→  const handleCampaignAction = async (campaignId: string, action: 'pause' | 'resume' | 'cancel') => {
   210→    setActioningId(campaignId);
   211→    try {
   212→      const res = await fetch(`/api/businesses/${businessId}/campaigns/${campaignId}`, {
   213→        method: 'PATCH',
   214→        headers: { 'Content-Type': 'application/json' },
   215→        body: JSON.stringify({ action }),
   216→      });
   217→      if (res.ok) {
   218→        toast.success(`Campaign ${action}d successfully`);
   219→        fetchCampaigns();
   220→        fetchAnalytics();
   221→      } else {
   222→        const err = await res.json();
   223→        toast.error(err.error || `Failed to ${action} campaign`);
   224→      }
   225→    } catch {
   226→      toast.error('Network error');
   227→    } finally {
   228→      setActioningId(null);
   229→    }
   230→  };
   231→
   232→  // ---- Create campaign ----
   233→  const handleCreateCampaign = async () => {
   234→    setCreateErrors([]);
   235→    const errors: string[] = [];
   236→
   237→    if (!createForm.name.trim()) errors.push('Campaign name is required');
   238→    if (createForm.name.trim().length > 50) errors.push('Campaign name too long (max 50 chars)');
   239→
   240→    const channelConf = CHANNEL_CONFIG[createForm.channel];
   241→    if (businessLevel < channelConf.minLevel) {
   242→      errors.push(`Business level ${businessLevel} too low for ${channelConf.name} (requires level ${channelConf.minLevel})`);
   243→    }
   244→
   245→    const activeCount = campaigns.filter(c => c.status === 'ACTIVE').length;
   246→    if (activeCount >= MAX_ACTIVE_CAMPAIGNS) {
   247→      errors.push(`Maximum ${MAX_ACTIVE_CAMPAIGNS} active campaigns allowed`);
   248→    }
   249→
   250→    if (errors.length > 0) {
   251→      setCreateErrors(errors);
   252→      return;
   253→    }
   254→
   255→    setCreating(true);
   256→    try {
   257→      const res = await fetch(`/api/businesses/${businessId}/campaigns`, {
   258→        method: 'POST',
   259→        headers: { 'Content-Type': 'application/json' },
   260→        body: JSON.stringify({
   261→          name: createForm.name.trim(),
   262→          channel: createForm.channel,
   263→          targetSegment: createForm.targetSegment,
   264→          dailyBudget: createForm.dailyBudget,
   265→          duration: createForm.duration,
   266→        }),
   267→      });
   268→      if (res.ok) {
   269→        toast.success('Campaign created!');
   270→        setShowCreateDialog(false);
   271→        setCreateForm({ name: '', channel: 'SOCIAL_MEDIA', targetSegment: null, dailyBudget: 1000, duration: 7 });
   272→        setCreateErrors([]);
   273→        fetchCampaigns();
   274→        fetchAnalytics();
   275→      } else {
   276→        const err = await res.json();
   277→        const msg = err.error || err.errors?.join(', ') || 'Failed to create campaign';
   278→        toast.error(msg);
   279→        if (err.errors) setCreateErrors(err.errors);
   280→      }
   281→    } catch {
   282→      toast.error('Network error');
   283→    } finally {
   284→      setCreating(false);
   285→    }
   286→  };
   287→
   288→  // ---- Derived data ----
   289→  const activeCampaigns = campaigns.filter(c => c.status === 'ACTIVE');
   290→  const pausedCampaigns = campaigns.filter(c => c.status === 'PAUSED');
   291→  const historyCampaigns = campaigns.filter(c => c.status === 'COMPLETED' || c.status === 'CANCELLED');
   292→  const brandAwareness = analytics?.brandAwareness ?? 0;
   293→  const brandAwarenessBonus = analytics?.brandAwarenessBonus ?? 1;
   294→  const combinedModifier = analytics?.combinedDemandModifier ?? 1;
   295→  const totalDailySpend = analytics?.totalDailySpend ?? activeCampaigns.reduce((s, c) => s + c.dailyBudget, 0);
   296→
   297→  // ---- Render ----
   298→  return (
   299→    <div className="space-y-4">
   300→      {/* ===== A. MARKETING OVERVIEW ===== */}
   301→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   302→        {/* Brand Awareness */}
   303→        <Card>
   304→          <CardContent className="p-4">
   305→            <div className="flex items-center gap-2 mb-2">
   306→              <Eye className="h-4 w-4 text-emerald-600" />
   307→              <span className="text-xs font-medium text-muted-foreground">Brand Awareness</span>
   308→            </div>
   309→            {analyticsLoading ? (
   310→              <Skeleton className="h-8 w-20 mb-1" />
   311→            ) : (
   312→              <div className="text-2xl font-bold">{brandAwareness.toFixed(0)}<span className="text-sm text-muted-foreground">/100</span></div>
   313→            )}
   314→            <Progress value={brandAwareness} className="h-2 mt-1" />
   315→            <p className="text-xs text-muted-foreground mt-1">Demand bonus: ×{brandAwarenessBonus.toFixed(2)}</p>
   316→          </CardContent>
   317→        </Card>
   318→
   319→        {/* Active Campaigns */}
   320→        <Card>
   321→          <CardContent className="p-4">
   322→            <div className="flex items-center gap-2 mb-2">
   323→              <Megaphone className="h-4 w-4 text-emerald-600" />
   324→              <span className="text-xs font-medium text-muted-foreground">Active Campaigns</span>
   325→            </div>
   326→            {loading ? (
   327→              <Skeleton className="h-8 w-10" />
   328→            ) : (
   329→              <div className="text-2xl font-bold">{activeCampaigns.length}<span className="text-sm text-muted-foreground">/{MAX_ACTIVE_CAMPAIGNS}</span></div>
   330→            )}
   331→            {pausedCampaigns.length > 0 && (
   332→              <p className="text-xs text-yellow-600 mt-1">{pausedCampaigns.length} paused</p>
   333→            )}
   334→          </CardContent>
   335→        </Card>
   336→
   337→        {/* Daily Spend */}
   338→        <Card>
   339→          <CardContent className="p-4">
   340→            <div className="flex items-center gap-2 mb-2">
   341→              <DollarSign className="h-4 w-4 text-emerald-600" />
   342→              <span className="text-xs font-medium text-muted-foreground">Daily Spend</span>
   343→            </div>
   344→            {analyticsLoading ? (
   345→              <Skeleton className="h-8 w-24" />
   346→            ) : (
   347→              <div className="text-2xl font-bold">{formatTk(totalDailySpend)}</div>
   348→            )}
   349→          </CardContent>
   350→        </Card>
   351→
   352→        {/* Demand Boost */}
   353→        <Card>
   354→          <CardContent className="p-4">
   355→            <div className="flex items-center gap-2 mb-2">
   356→              <Zap className="h-4 w-4 text-emerald-600" />
   357→              <span className="text-xs font-medium text-muted-foreground">Demand Boost</span>
   358→            </div>
   359→            {analyticsLoading ? (
   360→              <Skeleton className="h-8 w-16" />
   361→            ) : (
   362→              <div className="text-2xl font-bold">×{combinedModifier.toFixed(2)}</div>
   363→            )}
   364→            <p className="text-xs text-muted-foreground mt-1">
   365→              {combinedModifier > 1.1 ? '↑ Significant boost' : combinedModifier > 1.0 ? '↑ Moderate boost' : 'No active boost'}
   366→            </p>
   367→          </CardContent>
   368→        </Card>
   369→      </div>
   370→
   371→      {/* ===== B. ACTIVE CAMPAIGNS ===== */}
   372→      <div>
   373→        <div className="flex items-center justify-between mb-3">
   374→          <h3 className="text-sm font-semibold flex items-center gap-2">
   375→            <Activity className="h-4 w-4" /> Active Campaigns
   376→          </h3>
   377→          <Button
   378→            size="sm"
   379→            className="gap-1 text-white text-xs"
   380→            style={{ background: '#006a4e' }}
   381→            onClick={() => setShowCreateDialog(true)}
   382→            disabled={activeCampaigns.length >= MAX_ACTIVE_CAMPAIGNS}
   383→          >
   384→            <Plus className="h-3.5 w-3.5" /> New Campaign
   385→          </Button>
   386→        </div>
   387→
   388→        {loading ? (
   389→          <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   390→            {[1, 2].map(i => (
   391→              <Card key={i}><CardContent className="p-4"><Skeleton className="h-32 w-full" /></CardContent></Card>
   392→            ))}
   393→          </div>
   394→        ) : activeCampaigns.length === 0 && pausedCampaigns.length === 0 ? (
   395→          <Card>
   396→            <CardContent className="p-8 text-center">
   397→              <Megaphone className="h-10 w-10 mx-auto text-muted-foreground mb-2" />
   398→              <p className="text-sm text-muted-foreground">No active campaigns</p>
   399→              <p className="text-xs text-muted-foreground mb-3">Launch a campaign to boost customer demand</p>
   400→              <Button size="sm" className="gap-1 text-white" style={{ background: '#006a4e' }} onClick={() => setShowCreateDialog(true)}>
   401→                <Plus className="h-3.5 w-3.5" /> Create Campaign
   402→              </Button>
   403→            </CardContent>
   404→          </Card>
   405→        ) : (
   406→          <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   407→            {[...activeCampaigns, ...pausedCampaigns].map(campaign => {
   408→              const ch = CHANNEL_CONFIG[campaign.channel];
   409→              const IconComp = ch?.LucideIcon || Megaphone;
   410→              const progress = campaign.duration > 0 ? Math.min(100, (campaign.daysRun / campaign.duration) * 100) : 0;
   411→              const roi = campaign.roi ?? (campaign.totalSpend > 0 ? (campaign.revenueInfluenced - campaign.totalSpend) / campaign.totalSpend : 0);
   412→              const eff = campaign.effectiveness ?? 0;
   413→
   414→              return (
   415→                <motion.div key={campaign.id} initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
   416→                  <Card className="overflow-hidden">
   417→                    <CardContent className="p-4 space-y-3">
   418→                      {/* Header */}
   419→                      <div className="flex items-start justify-between">
   420→                        <div className="flex items-center gap-2">
   421→                          <div className="flex items-center justify-center w-8 h-8 rounded-lg bg-emerald-50">
   422→                            <IconComp className="h-4 w-4 text-emerald-600" />
   423→                          </div>
   424→                          <div>
   425→                            <p className="text-sm font-semibold leading-tight">{campaign.name}</p>
   426→                            <p className="text-xs text-muted-foreground">{ch?.name || campaign.channel}</p>
   427→                          </div>
   428→                        </div>
   429→                        <Badge variant="outline" className={statusColor(campaign.status)}>
   430→                          {campaign.status}
   431→                        </Badge>
   432→                      </div>
   433→
   434→                      {/* Progress */}
   435→                      <div>
   436→                        <div className="flex justify-between text-xs text-muted-foreground mb-1">
   437→                          <span>Day {campaign.daysRun} / {campaign.duration}</span>
   438→                          <span>{progress.toFixed(0)}%</span>
   439→                        </div>
   440→                        <Progress value={progress} className="h-2" />
   441→                      </div>
   442→
   443→                      {/* Metrics */}
   444→                      <div className="grid grid-cols-3 gap-2 text-center">
   445→                        <div>
   446→                          <p className="text-xs text-muted-foreground">Reach</p>
   447→                          <p className="text-sm font-semibold">{campaign.totalReach.toLocaleString()}</p>

   459→                      {/* Budget & ROI */}
   460→                      <div className="flex items-center justify-between text-xs">

   468→                      {/* Actions */}
   469→                      <div className="flex gap-2 pt-1">

   474→                              onClick={() => handleCampaignAction(campaign.id, 'pause')}

   478→                            </Button>
   479→                            <Button
   480→                              size="sm" variant="outline" className="text-xs gap-1 text-red-600 hover:text-red-700"
   481→                              onClick={() => handleCampaignAction(campaign.id, 'cancel')}
   482→                              disabled={actioningId === campaign.id}

   484→                              <XCircle className="h-3 w-3" /> Cancel
   485→                            </Button>
   486→                          </>
   487→                        )}
   488→                        {campaign.status === 'PAUSED' && (
   489→                          <>
   490→                            <Button
   491→                              size="sm" variant="outline" className="flex-1 text-xs gap-1 text-green-600 hover:text-green-700"
